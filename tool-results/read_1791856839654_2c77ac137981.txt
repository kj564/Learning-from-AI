    65→void RandomLevelSource::prepareHeights(int xOffs, int zOffs, unsigned char* blocks, /*Biome*/void* biomes, float* temperatures) {
    66→        
    67→        int xChunks = 16 / CHUNK_WIDTH;
    68→    int waterHeight = Level::DEPTH - 64;
    69→
    70→    int xSize = xChunks + 1;
    71→    int ySize = 128 / CHUNK_HEIGHT + 1;
    72→    int zSize = xChunks + 1;
    73→    buffer = getHeights(buffer, xOffs * xChunks, 0, zOffs * xChunks, xSize, ySize, zSize);
    74→
    75→    for (int xc = 0; xc < xChunks; xc++) {
    76→        for (int zc = 0; zc < xChunks; zc++) {
    77→            for (int yc = 0; yc < 128 / CHUNK_HEIGHT; yc++) {
    78→                float yStep = 1 / (float) CHUNK_HEIGHT;
    79→                float s0 = buffer[((xc + 0) * zSize + (zc + 0)) * ySize + (yc + 0)];
    80→                float s1 = buffer[((xc + 0) * zSize + (zc + 1)) * ySize + (yc + 0)];
    81→                float s2 = buffer[((xc + 1) * zSize + (zc + 0)) * ySize + (yc + 0)];
    82→                float s3 = buffer[((xc + 1) * zSize + (zc + 1)) * ySize + (yc + 0)];
    83→
    84→                float s0a = (buffer[((xc + 0) * zSize + (zc + 0)) * ySize + (yc + 1)] - s0) * yStep;
    85→                float s1a = (buffer[((xc + 0) * zSize + (zc + 1)) * ySize + (yc + 1)] - s1) * yStep;
    86→                float s2a = (buffer[((xc + 1) * zSize + (zc + 0)) * ySize + (yc + 1)] - s2) * yStep;
    87→                float s3a = (buffer[((xc + 1) * zSize + (zc + 1)) * ySize + (yc + 1)] - s3) * yStep;
    88→
    89→                for (int y = 0; y < CHUNK_HEIGHT; y++) {
    90→                    float xStep = 1 / (float) CHUNK_WIDTH;
    91→
    92→                    float _s0 = s0;
    93→                    float _s1 = s1;
    94→                    float _s0a = (s2 - s0) * xStep;
    95→                    float _s1a = (s3 - s1) * xStep;
    96→
    97→                    for (int x = 0; x < CHUNK_WIDTH; x++) {
    98→                        int offs = (x + xc * CHUNK_WIDTH) << 11 | (0 + zc * CHUNK_WIDTH) << 7 | (yc * CHUNK_HEIGHT + y);
    99→                        int step = 1 << 7;
   100→                        float zStep = 1 / (float) CHUNK_WIDTH;
   101→
   102→                        float val = _s0;
   103→                        float vala = (_s1 - _s0) * zStep;
   104→                        for (int z = 0; z < CHUNK_WIDTH; z++) {
   105→// + (zc * CHUNK_WIDTH + z)];
   106→                            float temp = temperatures[(xc * CHUNK_WIDTH + x) * 16 + (zc * CHUNK_WIDTH + z)];
   107→                            int tileId = 0;
   108→                            if (yc * CHUNK_HEIGHT + y < waterHeight) {
   109→                                if (temp < SNOW_CUTOFF && yc * CHUNK_HEIGHT + y >= waterHeight - 1) {
   110→                                    tileId = Tile::ice->id;
   111→                                } else {
   112→                                    tileId = Tile::calmWater->id;
   113→                                }
   114→                            }
   115→                            if (val > 0) {
   116→                                tileId = Tile::rock->id;
   117→                            } else {
   118→                            }
   119→
   120→                            blocks[offs] = (unsigned char) tileId;
   121→                            offs += step;
   122→                            val += vala;
   123→                        }
   124→                        _s0 += _s0a;
   125→                        _s1 += _s1a;
   126→                    }
   127→
   128→                    s0 += s0a;
   129→                    s1 += s1a;
   130→                    s2 += s2a;
   131→                    s3 += s3a;
   132→                }
   133→            }
   134→        }
   135→    }

   138→void RandomLevelSource::buildSurfaces(int xOffs, int zOffs, unsigned char* blocks, Biome** biomes) {
   139→    int waterHeight = Level::DEPTH - 64;
   140→
   141→    float s = 1 / 32.0f;
   142→    perlinNoise2.getRegion(sandBuffer, (float)(xOffs * 16), (float)(zOffs * 16), 0, 16, 16, 1, s, s, 1);
   143→    perlinNoise2.getRegion(gravelBuffer, (float)(xOffs * 16), 109.01340f, (float)(zOffs * 16), 16, 1, 16, s, 1, s);
   144→    perlinNoise3.getRegion(depthBuffer, (float)(xOffs * 16), (float)(zOffs * 16), 0, 16, 16, 1, s * 2, s * 2, s * 2);
   145→
   146→    for (int x = 0; x < 16; x++) {
   147→        for (int z = 0; z < 16; z++) {
   148→                        float temp = 1; // @todo: read temp from BiomeSource
   149→            Biome* b = biomes[x + z * 16];
   150→            bool sand = (sandBuffer[x + z * 16] + random.nextFloat() * 0.2f) > 0;
   151→            bool gravel = (gravelBuffer[x + z * 16] + random.nextFloat() * 0.2f) > 3;
   152→            int runDepth = (int) (depthBuffer[x + z * 16] / 3 + 3 + random.nextFloat() * 0.25f);
   153→
   154→            int run = -1;
   155→
   156→                        char top = b->topMaterial;
   157→            char material = b->material;
   158→
   159→            for (int y = 127; y >= 0; y--) {
   160→                int offs = (z * 16 + x) * 128 + y;
   161→
   162→                if (y <= 0 + random.nextInt(5)) {
   163→                    blocks[offs] = (char) Tile::unbreakable->id;
   164→                } else {
   165→                    int old = blocks[offs];
   166→
   167→                    if (old == 0) {
   168→                        run = -1;
   169→                    } else if (old == Tile::rock->id) {
   170→                        if (run == -1) {
   171→                            if (runDepth <= 0) {
   172→                                top = 0;
   173→                                material = (char) Tile::rock->id;
   174→                            } else if (y >= waterHeight - 4 && y <= waterHeight + 1) {
   175→                                top = b->topMaterial;
   176→                                                                material = b->material;
   177→                                                                
   178→                                                                //@attn: ?
   179→                                if (gravel) {
   180→                                                                        top = 0;
   181→                                                                        material = (char) Tile::gravel->id;
   182→                                                                }
   183→                                if (sand) {
   184→                                                                        top = (char) Tile::sand->id;
   185→                                                                        material = (char) Tile::sand->id;
   186→                                                                }
   187→                            }
   188→
   189→                            if (y < waterHeight && top == 0) {
   190→                                                                if (temp < 0.15f)
   191→                                                                        top = (char) Tile::ice->id;
   192→                                                                else
   193→                                                                        top = (char) Tile::calmWater->id;
   194→                                                        }
   195→
   196→                            run = runDepth;
   197→                            if (y >= waterHeight - 1) blocks[offs] = top;
   198→                            else blocks[offs] = material;
   199→                        } else if (run > 0) {
   200→                            run--;
   201→                            blocks[offs] = material;
   202→
   203→                            // place a few sandstone blocks beneath sand
   204→                            // runs
   205→                            if (run == 0 && material == Tile::sand->id) {
   206→                                run = random.nextInt(4);
   207→                                material = (char) Tile::sandStone->id;
   208→                            }
   209→                        }
   210→                    }
   211→                }
   212→            }
   213→        }
   214→    }

   488→LevelChunk* RandomLevelSource::getChunk(int xOffs, int zOffs) {

   494→                return it->second;
   495→
   496→    random.setSeed((long)(xOffs * 341872712l + zOffs * 132899541l)); //@fix
   497→

   513→/*private*/
   514→float* RandomLevelSource::getHeights(float* buffer, int x, int y, int z, int xSize, int ySize, int zSize) {
   515→        const int size = xSize * ySize * zSize;
   516→        if (size > MAX_BUFFER_SIZE) {
   517→                LOGI("RandomLevelSource::getHeights: TOO LARGE BUFFER REQUESTED: %d (max %d)\n", size, MAX_BUFFER_SIZE);
   518→        }
   519→
   520→    float s = 1 * 684.412f;
   521→    float hs = 1 * 684.412f;
   522→
   523→    float* temperatures = level->getBiomeSource()->temperatures;
   524→    float* downfalls = level->getBiomeSource()->downfalls;
   525→    sr = scaleNoise.getRegion(sr, x, z, xSize, zSize, 1.121f, 1.121f, 0.5f);
   526→    dr = depthNoise.getRegion(dr, x, z, xSize, zSize, 200.0f, 200.0f, 0.5f);
   527→
   528→    pnr = perlinNoise1.getRegion(pnr, (float)x, (float)y, (float)z, xSize, ySize, zSize, s / 80.0f, hs / 160.0f, s / 80.0f);
   529→    ar = lperlinNoise1.getRegion(ar, (float)x, (float)y, (float)z, xSize, ySize, zSize, s, hs, s);
   530→    br = lperlinNoise2.getRegion(br, (float)x, (float)y, (float)z, xSize, ySize, zSize, s, hs, s);
   531→
   532→    int p = 0;
   533→    int pp = 0;
   534→
   535→    int wScale = 16 / xSize;
   536→    for (int xx = 0; xx < xSize; xx++) {
   537→        int xp = xx * wScale + wScale / 2;
   538→
   539→        for (int zz = 0; zz < zSize; zz++) {
   540→            int zp = zz * wScale + wScale / 2;
   541→            float temperature = temperatures[xp * 16 + zp];
   542→            float downfall = downfalls[xp * 16 + zp] * temperature;
   543→            float dd = 1 - downfall;
   544→            dd = dd * dd;
   545→            dd = dd * dd;
   546→            dd = 1 - dd;
   547→
   548→            float scale = ((sr[pp] + 256.0f) / 512);
   549→            scale *= dd;
   550→            if (scale > 1) scale = 1;
   551→
   552→
   553→            float depth = (dr[pp] / 8000.0f);
   554→            if (depth < 0) depth = -depth * 0.3f;
   555→            depth = depth * 3.0f - 2.0f;
   556→
   557→            if (depth < 0) {
   558→                depth = depth / 2;
   559→                if (depth < -1) depth = -1;
   560→                depth = depth / 1.4f;
   561→                depth /= 2;
   562→                scale = 0;
   563→            } else {
   564→                if (depth > 1) depth = 1;
   565→                depth = depth / 8;
   566→            }
   567→
   568→            if (scale < 0) scale = 0;
   569→            scale = (scale) + 0.5f;
   570→            depth = depth * ySize / 16;
   571→
   572→            float yCenter = ySize / 2.0f + depth * 4;
   573→
   574→            pp++;
   575→
   576→            for (int yy = 0; yy < ySize; yy++) {
   577→                float val = 0;
   578→
   579→                float yOffs = (yy - (yCenter)) * 12 / scale;
   580→                if (yOffs < 0) yOffs *= 4;
   581→
   582→                float bb = ar[p] / 512;
   583→                float cc = br[p] / 512;
   584→
   585→                float v = (pnr[p] / 10 + 1) / 2;
   586→                if (v < 0) val = bb;
   587→                else if (v > 1) val = cc;
   588→                else val = bb + (cc - bb) * v;
   589→                val -= yOffs;
   590→
   591→                if (yy > ySize - 4) {
   592→                    float slide = (yy - (ySize - 4)) / (4 - 1.0f);
   593→                    val = val * (1 - slide) + -10 * slide;
   594→                }
   595→
   596→                buffer[p] = val;
   597→                p++;
   598→            }
   599→        }
   600→    }
   601→    return buffer;
   602→}

   605→void RandomLevelSource::calcWaterDepths(ChunkSource* parent, int xt, int zt) {
   606→    int xo = xt * 16;
   607→    int zo = zt * 16;
   608→    for (int x = 0; x < 16; x++) {
   609→        int y = level->getSeaLevel();
   610→        for (int z = 0; z < 16; z++) {
   611→            int xp = xo + x + 7;
   612→            int zp = zo + z + 7;
   613→            int h = level->getHeightmap(xp, zp);
   614→            if (h <= 0) {
   615→                if (level->getHeightmap(xp - 1, zp) > 0 || level->getHeightmap(xp + 1, zp) > 0 || level->getHeightmap(xp, zp - 1) > 0 || level->getHeightmap(xp, zp + 1) > 0) {
   616→                    bool hadWater = false;
   617→                    if (hadWater || (level->getTile(xp - 1, y, zp) == Tile::calmWater->id && level->getData(xp - 1, y, zp) < 7)) hadWater = true;
   618→                    if (hadWater || (level->getTile(xp + 1, y, zp) == Tile::calmWater->id && level->getData(xp + 1, y, zp) < 7)) hadWater = true;
   619→                    if (hadWater || (level->getTile(xp, y, zp - 1) == Tile::calmWater->id && level->getData(xp, y, zp - 1) < 7)) hadWater = true;
   620→                    if (hadWater || (level->getTile(xp, y, zp + 1) == Tile::calmWater->id && level->getData(xp, y, zp + 1) < 7)) hadWater = true;
   621→                    if (hadWater) {
   622→                        for (int x2 = -5; x2 <= 5; x2++) {
   623→                            for (int z2 = -5; z2 <= 5; z2++) {
   624→                                int d = (x2 > 0 ? x2 : -x2) + (z2 > 0 ? z2 : -z2);
   625→
   626→                                if (d <= 5) {
   627→                                    d = 6 - d;
   628→                                    if (level->getTile(xp + x2, y, zp + z2) == Tile::calmWater->id) {
   629→                                        int od = level->getData(xp + x2, y, zp + z2);
   630→                                        if (od < 7 && od < d) {
   631→                                            level->setData(xp + x2, y, zp + z2, d);
   632→                                        }
   633→                                    }
   634→                                }
   635→                            }
   636→                        }
   637→                        if (hadWater) {
   638→                            level->setTileAndDataNoUpdate(xp, y, zp, Tile::calmWater->id, 7);
   639→                            for (int y2 = 0; y2 < y; y2++) {
   640→                                level->setTileAndDataNoUpdate(xp, y2, zp, Tile::calmWater->id, 8);
   641→                            }
   642→                        }
   643→                    }
   644→                }
   645→            }
   646→        }
   647→    }
   648→}
