   269→void Minecraft::setLevel(Level* level, const std::string& message /* ="" */, LocalPlayer* forceInsertPlayer /* = NULL */) {
   270→        cameraTargetPlayer = NULL;
   271→        LOGI("Seed is %ld\n", level->getSeed());
   272→
   273→        if (level != NULL) {
   274→                level->raknetInstance = raknetInstance;
   275→        gameMode->initLevel(level);
   276→
   277→                if (!player && forceInsertPlayer)
   278→                {
   279→                        player = forceInsertPlayer;
   280→                        player->resetPos(false);
   281→                        //level->addEntity(forceInsertPlayer);
   282→                }
   283→                else if (player != NULL) {
   284→                        player->resetPos(false);
   285→                        if (level != NULL) {
   286→                                level->addEntity(player);
   287→                        }
   288→                }
   289→                this->level = level;
   290→                _hasSignaledGeneratingLevelFinished = false;
   291→#ifdef STANDALONE_SERVER
   292→                const bool threadedLevelCreation = false;
   293→#else
   294→                const bool threadedLevelCreation = true;
   295→#endif
   296→
   297→                if (threadedLevelCreation) {
   298→                        // Threaded
   299→                        // "Lock"
   300→                        isGeneratingLevel = true;
   301→                        generateLevelThread = new CThread(Minecraft::prepareLevel_tspawn, this);
   302→                } else {
   303→                        // Non-threaded
   304→                        generateLevel("Currently not used", level);
   305→                }
   306→    } else {
   307→        player = NULL;
   308→    }
   309→
   310→    this->lastTickTime = 0;
   311→        this->_running = true;
   312→}

   314→void Minecraft::leaveGame(bool renameLevel /*=false*/)
   315→{
   316→    if (isGeneratingLevel || !_hasSignaledGeneratingLevelFinished)
   317→        return;
   318→    
   319→        isGeneratingLevel = false;
   320→        bool saveLevel = level && (!level->isClientSide || renameLevel);
   321→
   322→        raknetInstance->disconnect();
   323→        if (saveLevel) {
   324→                // If server or wanting to save level as client, save all unsaved chunks!
   325→                level->getChunkSource()->saveAll(true);
   326→        }
   327→
   328→        LOGI("Clearing levels\n");
   329→
   330→        cameraTargetPlayer = NULL;
   331→#ifndef STANDALONE_SERVER
   332→        levelRenderer->setLevel(NULL);
   333→        particleEngine->setLevel(NULL);
   334→#endif
   335→        LOGI("Erasing callback\n");
   336→        delete netCallback;
   337→        netCallback = NULL;
   338→
   339→        LOGI("Erasing level\n");
   340→        if (level != NULL) {
   341→                delete level->getLevelStorage();
   342→                delete level;
   343→                level = NULL;
   344→        }
   345→        //delete player;
   346→        player = NULL;
   347→        cameraTargetPlayer = NULL;
   348→
   349→        _running = false;
   350→#ifndef STANDALONE_SERVER
   351→        if (renameLevel) {
   352→                setScreen(new RenameMPLevelScreen(LevelStorageSource::TempLevelId));
   353→        }
   354→        else
   355→                screenChooser.setScreen(SCREEN_STARTMENU);
   356→#endif
   357→}

   359→void Minecraft::prepareLevel(const std::string& title) {
   360→        LOGI("status: 1\n");
   361→        progressStageStatusId = 1;
   362→
   363→        Stopwatch A, B, C, D;
   364→        A.start();
   365→
   366→        Stopwatch L;
   367→
   368→        // Dont update lights if we load the level (ok, actually just with leveldata version=1.+(?))
   369→        if (!level->isNew())
   370→                level->setUpdateLights(false);
   371→
   372→        int Max = CHUNK_CACHE_WIDTH * CHUNK_CACHE_WIDTH;
   373→        int pp = 0;
   374→        for (int x = 8; x < (CHUNK_CACHE_WIDTH * CHUNK_WIDTH); x += CHUNK_WIDTH) {
   375→        for (int z = 8; z < (CHUNK_CACHE_WIDTH * CHUNK_WIDTH); z += CHUNK_WIDTH) {
   376→            progressStagePercentage = 100 * pp++ / Max;
   377→            //printf("level generation progress %d\n", progressStagePercentage);
   378→                        B.start();
   379→            level->getTile(x, 64, z);
   380→                        B.stop();
   381→                        L.start();
   382→                        if (level->isNew())
   383→                                while (level->updateLights())
   384→                                        ;
   385→                        L.stop();
   386→        }
   387→    }
   388→        A.stop();
   389→        level->setUpdateLights(true);
   390→
   391→        C.start();
   392→        for (int x = 0; x < CHUNK_CACHE_WIDTH; x++)
   393→        {
   394→                for (int z = 0; z < CHUNK_CACHE_WIDTH; z++)
   395→                {
   396→                        LevelChunk* chunk = level->getChunk(x, z);
   397→                        if (chunk && !chunk->createdFromSave)
   398→                        {
   399→                                chunk->unsaved = false;
   400→                                chunk->clearUpdateMap();
   401→                        }
   402→                }
   403→        }
   404→        C.stop();
   405→
   406→        LOGI("status: 3\n");
   407→        progressStageStatusId = 3;
   408→        if (level->isNew()) {
   409→                level->setInitialSpawn(); // @note: should obviously be called from Level itself
   410→                level->saveLevelData();
   411→                level->getChunkSource()->saveAll(false);
   412→                level->saveGame();
   413→        } else {
   414→                level->saveLevelData();
   415→                level->loadEntities();
   416→        }
   417→
   418→        progressStagePercentage = -1;
   419→        progressStageStatusId = 2;
   420→        LOGI("status: 2\n");
   421→
   422→        D.start();
   423→        level->prepare();
   424→        D.stop();
   425→
   426→        A.print("Generate level: ");
   427→        L.print(" - light: ");
   428→        B.print(" - getTl: ");
   429→        C.print(" - clear: ");
   430→        D.print(" - prepr: ");
   431→        progressStageStatusId = 0;
   432→}

   434→void Minecraft::update() {
   435→        //LOGI("Enter Update\n");
   436→
   437→        if (Options::debugGl)
   438→                LOGI(">>>>>>>>>>\n");
   439→
   440→        TIMER_PUSH("root");
   441→
   442→        //if (level) {
   443→        //      LOGI("numplayers: %d\n", level->players.size());
   444→        //      for (int i = 0; i < level->players.size(); ++i) {
   445→        //              Player* p = level->players[i];
   446→        //              bool inEnt = std::find(level->entities.begin(), level->entities.end(), p) != level->entities.end();
   447→        //              LOGI("  %p, %d, %d - in? %d\n", p, p->entityId, p->owner.ToUint32(p->owner), inEnt);
   448→        //      }
   449→        //}
   450→
   451→        if (pause && level != NULL) {
   452→                float lastA = timer.a;
   453→                timer.advanceTime();
   454→                timer.a = lastA;
   455→        } else {
   456→                timer.advanceTime();
   457→        }
   458→
   459→        if (raknetInstance) {
   460→                raknetInstance->runEvents(netCallback);
   461→        }
   462→
   463→        TIMER_PUSH("tick");
   464→        int toTick = timer.ticks;
   465→        for (int i = 0; i < toTick; ++i, ++ticks)
   466→                tick(i, toTick-1);
   467→
   468→        TIMER_POP_PUSH("updatelights");
   469→        if (level && !isGeneratingLevel) {
   470→                level->updateLights();
   471→        }
   472→        TIMER_POP();
   473→
   474→        #ifndef STANDALONE_SERVER
   475→                if (gameMode != NULL) gameMode->render(timer.a);
   476→                TIMER_PUSH("sound");
   477→                soundEngine->update(player, timer.a);
   478→                TIMER_POP_PUSH("render");
   479→                gameRenderer->render(timer.a);
   480→                TIMER_POP();
   481→        #else
   482→        CThread::sleep(1);
   483→        #endif
   484→#ifndef STANDALONE_SERVER
   485→        Multitouch::resetThisUpdate();
   486→#endif
   487→        TIMER_POP();
   488→#ifndef STANDALONE_SERVER
   489→        checkGlError("Update finished");
   490→
   491→        if (options.renderDebug) {
   492→                if (!PerfTimer::enabled) {
   493→                        PerfTimer::reset();
   494→                        PerfTimer::enabled = true;
   495→                }
   496→
   497→                //TIMER_PUSH("debugfps");
   498→                _perfRenderer->renderFpsMeter(1);
   499→                checkGlError("render debug");
   500→                //TIMER_POP();
   501→        } else {
   502→                PerfTimer::enabled = false;
   503→        }
   504→#endif
   505→        //LOGI("Exit Update\n");
   506→}

   508→void Minecraft::tick(int nTick, int maxTick) {
   509→        if (missTime > 0) missTime--;
   510→#ifndef STANDALONE_SERVER
   511→        if (!screen && player) {
   512→                if (player->health <= 0) {
   513→                        setScreen(new DeathScreen());
   514→                }
   515→        }
   516→#endif
   517→        TIMER_PUSH("gameMode");
   518→        if (level && !pause) {
   519→                gameMode->tick();
   520→        }
   521→
   522→        TIMER_POP_PUSH("commandServer");
   523→        if (level && _commandServer) {
   524→                _commandServer->tick();
   525→        }
   526→
   527→        TIMER_POP_PUSH("input");
   528→        tickInput();
   529→#ifndef STANDALONE_SERVER
   530→        TIMER_POP_PUSH("gui");
   531→        gui.tick();
   532→#endif
   533→        //
   534→        // Ongoing level generation in a (perhaps) different thread. When it's
   535→        // ready, _levelGenerated() is called once and any threads are deleted.
   536→        //
   537→        if (isGeneratingLevel) {
   538→                return;
   539→        } else if (!_hasSignaledGeneratingLevelFinished) {
   540→                if (generateLevelThread) {
   541→                        delete generateLevelThread;
   542→                        generateLevelThread = NULL;
   543→                }
   544→                _levelGenerated();
   545→        }
   546→
   547→        //
   548→        // Normal game loop, run before or efter level generation
   549→        //
   550→        if (level != NULL)
   551→        {
   552→                if (!pause) {
   553→#ifndef STANDALONE_SERVER
   554→                        TIMER_POP_PUSH("gameRenderer");
   555→                        gameRenderer->tick(nTick, maxTick);
   556→
   557→                        TIMER_POP_PUSH("levelRenderer");
   558→                        levelRenderer->tick();
   559→#endif
   560→                        level->difficulty = options.difficulty;
   561→                        if (level->isClientSide) level->difficulty = Difficulty::EASY;
   562→
   563→                        TIMER_POP_PUSH("level");
   564→                        level->tickEntities();
   565→                        level->tick();
   566→#ifndef STANDALONE_SERVER
   567→                        TIMER_POP_PUSH("animateTick");
   568→                        if (player) {
   569→                                level->animateTick(Mth::floor(player->x), Mth::floor(player->y), Mth::floor(player->z));
   570→                        }
   571→#endif
   572→                }
   573→        }
   574→#ifndef STANDALONE_SERVER
   575→        textures->loadAndBindTexture("terrain.png");
   576→        if (!pause && !(screen && !screen->renderGameBehind())) {
   577→                #if !defined(RPI)
   578→                        #ifdef __APPLE__
   579→                        if (isSuperFast())
   580→                        #endif
   581→                        {
   582→                        if (nTick == maxTick) {
   583→                                TIMER_POP_PUSH("textures");
   584→                                textures->tick(true);
   585→                        }
   586→                        }
   587→                #endif
   588→        }
   589→        TIMER_POP_PUSH("particles");
   590→        particleEngine->tick();
   591→        if (screen) {
   592→                screenMutex = true;
   593→                screen->tick();
   594→                screenMutex = false;
   595→        }
   596→
   597→        // @note: fix to keep "isPressed" and "isReleased" as long as necessary.
   598→        //      Most likely keyboard and mouse could/should be reset here as well.
   599→        Multitouch::reset();
   600→#endif
   601→        TIMER_POP();
   602→}

   614→void Minecraft::tickInput() {
   615→#ifndef STANDALONE_SERVER
   616→        InputRAII raiiInput;
   617→
   618→        if (screen && !screen->passEvents) {
   619→                screenMutex = true;
   620→                screen->updateEvents();
   621→                //screen->updateSetScreen();
   622→                screenMutex = false;
   623→                if (hasScheduledScreen) {
   624→                        setScreen(scheduledScreen);
   625→                        scheduledScreen = NULL;
   626→                        hasScheduledScreen = false;
   627→                }
   628→                return;
   629→        }
   630→
   631→        if (!player) {
   632→                return;
   633→        }
   634→
   635→#ifdef RPI
   636→        bool mouseDiggable = true;
   637→        bool allowGuiClicks = !mouseGrabbed;
   638→#else
   639→        bool mouseDiggable = !gui.isInside(Mouse::getX(), Mouse::getY());
   640→        bool allowGuiClicks = true;
   641→#endif
   642→
   643→        TIMER_PUSH("mouse");
   644→        while (Mouse::next()) {
   645→                //if (Mouse::getButtonState(MouseAction::ACTION_LEFT))
   646→                //      LOGI("mouse-down-at: %d, %d\n", Mouse::getX(), Mouse::getY());
   647→                int passedTime = getTimeMs() - lastTickTime;
   648→                if (passedTime > 200) continue; // @note: As long Mouse::clear CLEARS the whole buffer, it's safe to break here
   649→                // But since it might be rewritten anyway (and hopefully there aren't a lot of messages, we just continue.
   650→
   651→                const MouseAction& e = Mouse::getEvent();
   652→
   653→#ifdef RPI // If clicked when not having focus, get focus @keyboard
   654→                if (!mouseGrabbed) {
   655→                        if (!screen && e.data == MouseAction::DATA_DOWN) {
   656→                                grabMouse();

   953→void Minecraft::handleBuildAction(BuildActionIntention* action) {
   954→#ifndef STANDALONE_SERVER
   955→        if (action->isRemove()) {
   956→                if (missTime > 0) return;
   957→                player->swing();
   958→        }
   959→        if(player->isUsingItem())
   960→                return;
   961→    bool mayUse = true;
   962→
   963→        if (!hitResult.isHit()) {
   964→                if (action->isRemove() && !gameMode->isCreativeType()) {
   965→                        missTime = 10;
   966→                }
   967→    } else if (hitResult.type == ENTITY) {
   968→        if (action->isAttack()) {
   969→                        player->swing();
   970→                        //LOGI("attacking!\n");
   971→                        InteractPacket packet(InteractPacket::Attack, player->entityId, hitResult.entity->entityId);
   972→                        raknetInstance->send(packet);
   973→            gameMode->attack(player, hitResult.entity);
   974→                } else if (action->isInteract()) {
   975→                        if (hitResult.entity->interactPreventDefault())
   976→                                mayUse = false;
   977→                        //LOGI("interacting!\n");
   978→                        InteractPacket packet(InteractPacket::Interact, player->entityId, hitResult.entity->entityId);
   979→                        raknetInstance->send(packet);
   980→            gameMode->interact(player, hitResult.entity);
   981→        }
   982→    } else if (hitResult.type == TILE) {
   983→                int x = hitResult.x;
   984→        int y = hitResult.y;
   985→        int z = hitResult.z;
   986→        int face = hitResult.f;
   987→
   988→                int oldTileId = level->getTile(x, y, z);
   989→        Tile* oldTile = Tile::tiles[oldTileId];
   990→
   991→                //bool tryDestroyBlock = false;
   992→
   993→                if (action->isRemove()) {
   994→                        if (!oldTile)
   995→                                return;
   996→
   997→                        //LOGI("tile: %s - %d, %d, %d. b: %f - %f\n", oldTile->getDescriptionId().c_str(), x, y, z, oldTile->getBrightness(level, x, y, z), oldTile->getBrightness(level, x, y+1, z));
   998→            level->extinguishFire(x, y, z, hitResult.f);
   999→                        gameMode->startDestroyBlock(x, y, z, hitResult.f);
  1000→        }
  1001→                else {
  1002→                        ItemInstance* item = player->inventory->getSelected();
  1003→                        if (gameMode->useItemOn(player, level, item, x, y, z, face, hitResult.pos)) {
  1004→                            mayUse = false;
  1005→                            player->swing();
  1006→                        #ifdef RPI
  1007→                                } else if (item && item->id == ((Item*)Item::sword_iron)->id) {
  1008→                                        player->swing();
  1009→                        #endif
  1010→                        }
  1011→                        if (item && item->count <= 0) {
  1012→                                player->inventory->clearSlot(player->inventory->selected);
  1013→                        }
  1014→                        //} else if (item && item->count != oldCount) {
  1015→                        //        gameRenderer->itemInHandRenderer->itemPlaced();
  1016→                        //}
  1017→                }
  1018→        }
  1019→        if (mayUse && action->isInteract()) {
  1020→                ItemInstance* item = player->inventory->getSelected();
  1021→                if (item && !player->isUsingItem()) {
  1022→                        if (gameMode->useItem(player, level, item)) {
  1023→                                gameRenderer->itemInHandRenderer->itemUsed();
  1024→                        }
  1025→                        if (item && item->count <= 0) {
  1026→                                player->inventory->clearSlot(player->inventory->selected);
  1027→                        }
  1028→                }
  1029→        }
  1030→#endif
  1031→}
