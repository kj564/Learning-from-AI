 11343→void ParticleEngine::crack(int x, int y, int z, int face) {
 11344→    int tid = level->getTile(x, y, z);
 11345→    if (tid == 0) return;

 11350→    // Random position within the tile's box (xx0..xx1, yy0..yy1, zz0..zz1)

 11352→    float yp = y + random.nextFloat() * ((tile->yy1 - tile->yy0) - r * 2) + r + tile->yy0;
 11353→    float zp = z + random.nextFloat() * ((tile->zz1 - tile->zz0) - r * 2) + r + tile->zz0;
 11354→
 11355→    // Snap to the clicked face (offset by r outside the face)
 11356→    switch (face) {
 11357→        case 0: yp = y + tile->yy0 - r; break;     // DOWN
 11358→        case 1: yp = y + tile->yy1 + r; break;     // UP
 11359→        case 2: zp = z + tile->zz0 - r; break;     // NORTH
 11360→        case 3: zp = z + tile->zz1 + r; break;     // SOUTH
 11361→        case 4: xp = x + tile->xx0 - r; break;     // WEST
 11362→        case 5: xp = x + tile->xx1 + r; break;     // EAST
 11363→    }
 11364→    add((new TerrainParticle(level, xp, yp, zp, 0, 0, 0, tile, data))
 11365→        ->init(x, y, z)->setPower(0.2f)->scale(0.6f));
 11366→}
