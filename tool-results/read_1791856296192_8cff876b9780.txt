    75→    75→int Player::startSleepInBed( int x, int y, int z ) {
    76→    76→        if(!level->isClientSide) {
    77→    77→                if(isSleeping() || !isAlive()) {
    78→    78→                        return BedSleepingResult::OTHER_PROBLEM;
    79→    79→                }
    80→    80→                if(Mth::abs(this->x - x) > 3 || Mth::abs(this->y - y) > 4 || Mth::abs(this->z - z) > 3) {
    81→    81→                        return BedSleepingResult::TOO_FAR_AWAY;
    82→    82→                }
    83→    83→                if(level->dimension->isNaturalDimension()) {
    84→    84→                        return BedSleepingResult::NOT_POSSIBLE_HERE;
    85→    85→                }
    86→    86→                if(level->isDay()) {
    87→    87→                        return BedSleepingResult::NOT_POSSIBLE_NOW;
    88→    88→                }
    89→    89→                float hRange = 8;
    90→    90→                float vRange = 5;
    91→    91→                EntityList monsters;
    92→    92→                level->getEntitiesOfClass(MobTypes::BaseEnemy, AABB(x- hRange, y - vRange, z - hRange, x + hRange, y + vRange, z + hRange), monsters);
    93→    93→                if(!monsters.empty()) {
    94→    94→                        return BedSleepingResult::NOT_SAFE;
    95→    95→                }
    96→    96→        }
    97→    97→
    98→    98→        setSize(0.2f, 0.2f);
    99→    99→        heightOffset = 0.2f;
   100→   100→        if(level->hasChunkAt(x, y, z)) {
   101→   101→                int data = level->getData(x, y, z);
   102→   102→                int direction = BedTile::getDirection(data);
   103→   103→                float xo = 0.5f, zo = 0.5f;
   104→   104→                switch(direction) {
   105→   105→                case Direction::SOUTH:
   106→   106→                        zo = 0.9f;
   107→   107→                        break;
   108→   108→                case Direction::NORTH:
   109→   109→                        zo = 0.1f;
   110→   110→                        break;
   111→   111→                case Direction::WEST:
   112→   112→                        xo = 0.1f;
   113→   113→                        break;
   114→   114→                case Direction::EAST:
   115→   115→                        xo = 0.9f;
   116→   116→                        break;
   117→   117→                }
   118→   118→                setBedOffset(direction);
   119→   119→                setPos(x + xo, y + 15.0f / 16.0f, z + zo);
   120→   120→        } else {
   121→   121→                setPos(x + 0.5f, y + 1.0f / 16.0f, z + 0.5f);
   122→   122→        }
   123→   123→        playerIsSleeping = true;
   124→   124→        sleepCounter = 0;
   125→   125→        bedPosition = Pos(x, y, z);
   126→   126→        xd = zd = yd = 0;
   127→   127→        if(!level->isClientSide) {
   128→   128→                level->updateSleepingPlayerList();
   129→   129→        }
   130→   130→        entityData.set<Pos>(DATA_BED_POSITION_ID, bedPosition);
   131→   131→        entityData.setFlag<SharedFlagsInformation::SharedFlagsInformationType>(DATA_PLAYER_FLAGS_ID, PLAYER_SLEEP_FLAG);
   132→   132→        return BedSleepingResult::OK;
   133→   133→}

   135→   135→void Player::stopSleepInBed( bool forcefulWakeUp, bool updateLevelList, bool saveRespawnPoint ) {
   136→   136→        if(!isSleeping())
   137→   137→                return;
   138→   138→        setSize(0.6f, 1.8f);
   139→   139→        setDefaultHeadHeight();
   140→   140→        Pos standUp = bedPosition;
   141→   141→        if(level->getTile(int(bedPosition.x), int(bedPosition.y), int(bedPosition.z)) == Tile::bed->id) {
   142→   142→                BedTile::setOccupied(level, int(bedPosition.x), int(bedPosition.y), int(bedPosition.z), false);
   143→   143→                bool foundStandUpPosition = BedTile::findStandUpPosition(level, int(bedPosition.x), int(bedPosition.y), int(bedPosition.z), 0, standUp);
   144→   144→                if(!foundStandUpPosition) {
   145→   145→                        standUp = Pos(bedPosition.x, bedPosition.y, bedPosition.z);
   146→   146→                }
   147→   147→                setPos(standUp.x + 0.5f, standUp.y + heightOffset + 0.1f, standUp.z + 0.5f);
   148→   148→        }
   149→   149→        playerIsSleeping = false;
   150→   150→        if(!level->isClientSide && updateLevelList) {
   151→   151→                level->updateSleepingPlayerList();
   152→   152→        }
   153→   153→        if(forcefulWakeUp) {
   154→   154→                sleepCounter = 0;
   155→   155→        } else {
   156→   156→                sleepCounter = SLEEP_DURATION;
   157→   157→        }
   158→   158→        // Quick fix to make the spawn position always saved, not sure if we always want to save this position but I like it.
   159→   159→        if(true || saveRespawnPoint) {
   160→   160→                Pos newRespawnPos;
   161→   161→                BedTile::findStandUpPosition(level, bedPosition.x, bedPosition.y, bedPosition.z, 0, newRespawnPos);
   162→   162→                setRespawnPosition(newRespawnPos);
   163→   163→        }
   164→   164→        entityData.clearFlag<SharedFlagsInformation::SharedFlagsInformationType>(DATA_PLAYER_FLAGS_ID, PLAYER_SLEEP_FLAG);
   165→   165→        allPlayersSleeping = false;
   166→   166→}

   219→   219→void Player::tick() {
   220→   220→        bool shouldSleep = entityData.getFlag<SharedFlagsInformation::SharedFlagsInformationType>(DATA_PLAYER_FLAGS_ID, PLAYER_SLEEP_FLAG);
   221→   221→        if(shouldSleep != isSleeping()) {
   222→   222→                if(isSleeping()) {
   223→   223→                        stopSleepInBed(true, true, true);
   224→   224→                } else {
   225→   225→                        bedPosition = entityData.getPos(DATA_BED_POSITION_ID);
   226→   226→                        startSleepInBed(bedPosition.x, bedPosition.y, bedPosition.z);
   227→   227→                }
   228→   228→        }
   229→   229→        if(isSleeping()) {
   230→   230→                sleepCounter++;
   231→   231→                if(sleepCounter > SLEEP_DURATION) {
   232→   232→                        sleepCounter = SLEEP_DURATION;
   233→   233→                }
   234→   234→                if(!level->isClientSide) {
   235→   235→                        if(!checkBed()) {
   236→   236→                                stopSleepInBed(true, true, false);
   237→   237→                        } else if(level->isDay()) {
   238→   238→                                stopSleepInBed(false, true, true);
   239→   239→                        }
   240→   240→                } 
   241→   241→        } else if(sleepCounter > 0) {
   242→   242→                sleepCounter++;
   243→   243→                if(sleepCounter >= (SLEEP_DURATION + WAKE_UP_DURATION)) {
   244→   244→                        sleepCounter = 0;
   245→   245→                }
   246→   246→        }
   247→   247→    super::tick();
   248→   248→
   249→   249→        if (!level->isClientSide) {
   250→   250→                foodData.tick(this);
   251→   251→        //      if (containerMenu != NULL && !containerMenu->stillValid(this)) {
   252→   252→        //              closeContainer();
   253→   253→        //      }
   254→   254→        }
   255→   255→}

   272→   272→void Player::spawnEatParticles(const ItemInstance* useItem, int count) {
   273→   273→        if (useItem->getUseAnimation() == UseAnim::drink) {
   274→   274→                level->playSound(this, "random.drink", 0.5f, level->random.nextFloat() * 0.1f + 0.9f);
   275→   275→        }
   276→   276→        else if (useItem->getUseAnimation() == UseAnim::eat) {
   277→   277→                for (int i = 0; i < count; i++) {
   278→   278→                        const float xx = -xRot * Mth::PI / 180;
   279→   279→                        const float yy = -yRot * Mth::PI / 180;
   280→   280→                        Vec3 d((random.nextFloat() - 0.5f) * 0.1f, Mth::random() * 0.1f + 0.1f, 0);
   281→   281→                        d.xRot(xx);
   282→   282→                        d.yRot(yy);
   283→   283→                        Vec3 p((random.nextFloat() - 0.5f) * 0.3f, -random.nextFloat() * 0.6f - 0.3f, 0.6f);
   284→   284→                        p.xRot(xx);
   285→   285→                        p.yRot(yy);
   286→   286→                        p = p.add(x, y + getHeadHeight(), z);
   287→   287→                        level->addParticle(PARTICLETYPE(iconcrack), p.x, p.y, p.z, d.x, d.y + 0.05f, d.z, useItem->getItem()->id);
   288→   288→                }
   289→   289→                level->playSound(this, "random.eat", .5f + .5f * random.nextInt(2), (random.nextFloat() - random.nextFloat()) * 0.2f + 1.0f);
   290→   290→        }
   291→   291→}

   425→   425→void Player::aiStep() {
   426→   426→    if (level->difficulty == Difficulty::PEACEFUL && health < MAX_HEALTH) {
   427→   427→        if (tickCount % (12 * SharedConstants::TicksPerSecond) == 0) heal(1);
   428→   428→    }
   429→   429→    //inventory.tick();
   430→   430→    oBob = bob;
   431→   431→        // moved the super::aiStep() part to the local player
   432→   432→
   433→   433→    float tBob = (float) Mth::sqrt(xd * xd + zd * zd);
   434→   434→    float tTilt = (float) Mth::atan(-yd * 0.2f) * 15.f;
   435→   435→    if (tBob > 0.1f) tBob = 0.1f;
   436→   436→    if (!onGround || health <= 0) tBob = 0;
   437→   437→    if (onGround || health <= 0) tTilt = 0;
   438→   438→    bob += (tBob - bob) * 0.4f;
   439→   439→    tilt += (tTilt - tilt) * 0.8f;
   440→   440→
   441→   441→    if (health > 0) {
   442→   442→        EntityList& entities = level->getEntities(this, bb.grow(1, 0, 1));
   443→   443→        for (unsigned int i = 0; i < entities.size(); i++) {
   444→   444→            Entity* e = entities[i];
   445→   445→            if (!e->removed) {
   446→   446→                touch(e);
   447→   447→            }
   448→   448→        }
   449→   449→    }
   450→   450→}

   515→   515→void Player::drop(ItemInstance* item, bool randomly) {
   516→   516→    if (item == NULL || item->isNull())
   517→   517→        return;
   518→   518→
   519→   519→    ItemEntity* thrownItem = new ItemEntity(level, x, y - 0.3f + getHeadHeight(), z, *item);
   520→   520→        { //@todo:itementity
   521→   521→                delete item;
   522→   522→                item = NULL;
   523→   523→        }
   524→   524→    thrownItem->throwTime = 20 * 2;
   525→   525→
   526→   526→    float pow = 0.1f;
   527→   527→    if (randomly) {
   528→   528→        float _pow = random.nextFloat() * 0.5f;
   529→   529→        float dir = random.nextFloat() * Mth::PI * 2;
   530→   530→        thrownItem->xd = -Mth::sin(dir) * _pow;
   531→   531→        thrownItem->zd = Mth::cos(dir) * _pow;
   532→   532→        thrownItem->yd = 0.2f;
   533→   533→
   534→   534→    } else {
   535→   535→        pow = 0.3f;
   536→   536→        thrownItem->xd = -Mth::sin(yRot / 180 * Mth::PI) * Mth::cos(xRot / 180 * Mth::PI) * pow;
   537→   537→        thrownItem->zd = Mth::cos(yRot / 180 * Mth::PI) * Mth::cos(xRot / 180 * Mth::PI) * pow;
   538→   538→        thrownItem->yd = -Mth::sin(xRot / 180 * Mth::PI) * pow + 0.1f;
   539→   539→        pow = 0.02f;
   540→   540→
   541→   541→        float dir = random.nextFloat() * Mth::PI * 2;
   542→   542→        pow *= random.nextFloat();
   543→   543→        thrownItem->xd += Mth::cos(dir) * pow;
   544→   544→        thrownItem->yd += (random.nextFloat() - random.nextFloat()) * 0.1f;
   545→   545→        thrownItem->zd += Mth::sin(dir) * pow;
   546→   546→    }
   547→   547→
   548→   548→    reallyDrop(thrownItem);
   549→   549→}

   852→   852→                armor[i].save(tag);

   859→   859→        if (!listTag)
   860→   860→                return;
   861→   861→
   862→   862→        const int count = Mth::Min(4, listTag->size());
   863→   863→        for (int i = 0; i < count; ++i) {
   864→   864→                Tag* tag = listTag->get(i);
   865→   865→                if (tag->getId() != Tag::TAG_Compound) continue;
   866→   866→
   867→   867→                armor[i].load((CompoundTag*) tag);
   868→   868→        }
   869→   869→}
