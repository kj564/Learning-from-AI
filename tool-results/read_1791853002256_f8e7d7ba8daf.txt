   129→int Gui::getSlotIdAt(int x, int y) {
   130→        int screenWidth = (int)(minecraft->width * InvGuiScale);
   131→        int screenHeight = (int)(minecraft->height * InvGuiScale);
   132→        x = (int)(x * InvGuiScale);
   133→        y = (int)(y * InvGuiScale);
   134→
   135→        if (y < (screenHeight - 16 - 3) || y > screenHeight)
   136→                return -1;
   137→
   138→        int xBase = 2 + screenWidth / 2 - getNumSlots() * 10;
   139→        int xRel  = (x - xBase);
   140→        if (xRel < 0)
   141→                return -1;
   142→
   143→        int slot = xRel / 20;
   144→        return (slot >= 0 && slot < getNumSlots())? slot : -1;
   145→}

   172→
   173→        if (extendSide < 0)

   175→        if (extendSide > 0)
   176→                return RectangleArea(pCenterX-pHalfWidth, (float)minecraft->height-pHeight, (float)minecraft->width, (float)minecraft->height);
   177→        
   178→        return RectangleArea(pCenterX-pHalfWidth, (float)minecraft->height-pHeight, pCenterX+pHalfWidth+2, (float)minecraft->height);

   234→void Gui::addMessage(const std::string& _string) {
   235→        if (!minecraft->font)
   236→                return;
   237→
   238→        std::string string = _string;
   239→        while (minecraft->font->width(string) > MAX_MESSAGE_WIDTH) {
   240→                unsigned int i = 1;
   241→                while (i < string.length() && minecraft->font->width(string.substr(0, i + 1)) <= MAX_MESSAGE_WIDTH) {
   242→                        i++;
   243→                }
   244→                addMessage(string.substr(0, i));
   245→                string = string.substr(i);
   246→        }
   247→        GuiMessage message;
   248→        message.message = string;
   249→        message.ticks = 0;
   250→        guiMessages.insert(guiMessages.begin(), message);
   251→        while (guiMessages.size() > 30) {
   252→                guiMessages.pop_back();
   253→        }
   254→}

   338→void Gui::onConfigChanged( const Config& c ) {
   339→        Tesselator& t = Tesselator::instance;
   340→        t.begin();
   341→
   342→        //
   343→        // Create outer feedback circle
   344→        //
   345→#ifdef ANDROID
   346→        const float mm = 12;
   347→#else
   348→        const float mm = 12;
   349→#endif
   350→        const float maxRadius = minecraft->pixelCalcUi.millimetersToPixels(mm);
   351→        const float radius = Mth::Min(80.0f/2, maxRadius);
   352→        //LOGI("radius, maxradius: %f, %f\n", radius, maxRadius);
   353→        const float radiusInner = radius * 0.95f;
   354→
   355→        const int steps = 24;
   356→        const float fstep = Mth::TWO_PI / steps;
   357→        for (int i = 0; i < steps; ++i) {
   358→                float a = i * fstep;;
   359→                float b = a + fstep;
   360→
   361→                float aCos = Mth::cos(a);
   362→                float bCos = Mth::cos(b);
   363→                float aSin = Mth::sin(a);
   364→                float bSin = Mth::sin(b);
   365→                float x00 = radius * aCos;
   366→                float x01 = radiusInner * aCos;
   367→                float x10 = radius * bCos;
   368→                float x11 = radiusInner * bCos;
   369→                float y00 = radius * aSin;
   370→                float y01 = radiusInner * aSin;
   371→                float y10 = radius * bSin;
   372→                float y11 = radiusInner * bSin;
   373→
   374→                t.vertexUV(x01, y01, 0, 0, 1);
   375→                t.vertexUV(x11, y11, 0, 1, 1);
   376→                t.vertexUV(x10, y10, 0, 1, 0);
   377→                t.vertexUV(x00, y00, 0, 0, 0);
   378→        }
   379→        rcFeedbackOuter = t.end(true, rcFeedbackOuter.vboId);
   380→
   381→        //
   382→        // Create the inner feedback ring
   383→        //
   384→        t.begin(GL_TRIANGLE_FAN);
   385→        t.vertex(0, 0, 0);
   386→        for (int i = 0; i < steps + 1; ++i) {
   387→                float a = -i * fstep;
   388→                float xx = radiusInner * Mth::cos(a);
   389→                float yy = radiusInner * Mth::sin(a);
   390→                t.vertex(xx, yy, 0);
   391→                //LOGI("x,y: %f, %f\n", xx, yy);
   392→        }
   393→        rcFeedbackInner = t.end(true, rcFeedbackInner.vboId);
   394→
   395→        if (c.minecraft->useTouchscreen()) {
   396→                // I'll bump this up to 6.
   397→                int num = 6; // without "..." dots
   398→                if (!c.minecraft->options.isJoyTouchArea && c.width > 480) {
   399→                        while (num < Inventory::MAX_SELECTION_SIZE - 1) {
   400→                                int x0, x1, y;
   401→                                getSlotPos(0, x0, y);
   402→                                getSlotPos(num, x1, y);
   403→                                int width = x1 - x0;
   404→                                float leftoverPixels = c.width - c.guiScale*width;
   405→                                if (c.pixelCalc.pixelsToMillimeters(leftoverPixels) < 80)
   406→                                        break;
   407→                                num++;
   408→                        }
   409→                }
   410→                _numSlots = num;
   411→#if defined(__APPLE__)
   412→                _numSlots = Mth::Min(7, _numSlots);
   413→#endif
   414→        } else {
   415→                _numSlots = Inventory::MAX_SELECTION_SIZE; // Xperia Play
   416→        }
   417→        MAX_MESSAGE_WIDTH = c.guiWidth;
   418→}

   424→int Gui::itemCountItoa( char* buffer, int count )
   425→{
   426→        if (count < 0)
   427→                return 0;
   428→
   429→        if (count < 10) { // 1 digit
   430→                buffer[0] = '0' + count;
   431→                buffer[1] = 0;
   432→                return 1;
   433→        } else if (count < 100) { // 2 digits
   434→                int digit = count/10;
   435→                buffer[0] = '0' + digit;
   436→                buffer[1] = '0' + count - digit*10;
   437→                buffer[2] = 0;
   438→        } else { // 3 digits -> "99+"
   439→                buffer[0] = buffer[1] = '9';
   440→                buffer[2] = '+';
   441→                buffer[3] = 0;
   442→                return 3;
   443→        }
   444→        return 2;
   445→}

   447→void Gui::tickItemDrop()
   448→{
   449→        // Handle item drop
   450→        static bool isCurrentlyActive = false;
   451→        isCurrentlyActive = false;
   452→        if (Mouse::isButtonDown(MouseAction::ACTION_LEFT)) {
   453→                int slot = getSlotIdAt(Mouse::getX(), Mouse::getY());
   454→                if (slot >= 0 && slot < getNumSlots()-1) {
   455→                        if (slot != _currentDropSlot) {
   456→                                _currentDropTicks = 0;
   457→                                _currentDropSlot = slot;
   458→                        }
   459→                        isCurrentlyActive = true;
   460→                        if ((_currentDropTicks += 1.0f) >= DropTicks) {
   461→                                minecraft->player->inventory->dropSlot(slot, false);
   462→                                minecraft->level->playSound(minecraft->player, "random.pop", 0.3f, 1);
   463→                                isCurrentlyActive = false;
   464→                        }
   465→                }
   466→        }
   467→        if (!isCurrentlyActive) {
   468→                _currentDropSlot = -1;
   469→                _currentDropTicks = -1;
   470→        }
   471→}

   501→void Gui::renderProgressIndicator( const bool isTouchInterface, const int screenWidth, const int screenHeight, float a ) {
   502→        ItemInstance* currentItem = minecraft->player->inventory->getSelected();
   503→        bool bowEquipped = currentItem != NULL ? currentItem->getItem() == Item::bow : false;
   504→        bool itemInUse = currentItem != NULL ? currentItem->getItem() == minecraft->player->getUseItem()->getItem() : false;
   505→        if (!isTouchInterface || minecraft->options.isJoyTouchArea || (bowEquipped && itemInUse)) {
   506→                minecraft->textures->loadAndBindTexture("gui/icons.png");
   507→                glEnable(GL_BLEND);
   508→                glBlendFunc2(GL_ONE_MINUS_DST_COLOR, GL_ONE_MINUS_SRC_COLOR);
   509→                blit(screenWidth/2 - 8, screenHeight/2 - 8, 0, 0, 16, 16);
   510→                glDisable(GL_BLEND);
   511→        } else if(!bowEquipped) {
   512→                const float tprogress = minecraft->gameMode->destroyProgress;
   513→                const float alpha = Mth::clamp(minecraft->inputHolder->alpha, 0.0f, 1.0f);
   514→                //LOGI("alpha: %f\n", alpha);
   515→
   516→                if (tprogress <= 0 && minecraft->inputHolder->alpha >= 0) {
   517→                        glDisable2(GL_TEXTURE_2D);
   518→                        glEnable2(GL_BLEND);
   519→                        glBlendFunc2(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
   520→                        if (minecraft->hitResult.isHit())
   521→                                glColor4f2(1, 1, 1, 0.8f * alpha);
   522→                        else
   523→                                glColor4f2(1, 1, 1, Mth::Min(0.4f, alpha*0.4f));
   524→
   525→                        //LOGI("alpha2: %f\n", alpha);
   526→                        const float x = InvGuiScale * minecraft->inputHolder->mousex;
   527→                        const float y = InvGuiScale * minecraft->inputHolder->mousey;
   528→                        glTranslatef2(x, y, 0);
   529→                        drawArrayVT(rcFeedbackOuter.vboId, rcFeedbackOuter.vertexCount, 24);
   530→                        glTranslatef2(-x, -y, 0);
   531→
   532→                        glEnable2(GL_TEXTURE_2D);
   533→                        glDisable(GL_BLEND);
   534→                } else if (tprogress > 0) {
   535→                        const float oProgress = minecraft->gameMode->oDestroyProgress;
   536→                        const float progress = 0.5f * (oProgress + (tprogress - oProgress) * a);
   537→
   538→                        //static Stopwatch w;
   539→                        //w.start();
   540→
   541→                        glDisable2(GL_TEXTURE_2D);
   542→                        glColor4f2(1, 1, 1, 0.8f * alpha);
   543→                        glEnable(GL_BLEND);
   544→                        glBlendFunc2(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
   545→
   546→                        const float x = InvGuiScale * minecraft->inputHolder->mousex;
   547→                        const float y = InvGuiScale * minecraft->inputHolder->mousey;
   548→                        glPushMatrix2();
   549→                        glTranslatef2(x, y, 0);
   550→                        drawArrayVT(rcFeedbackOuter.vboId, rcFeedbackOuter.vertexCount, 24);

   567→void Gui::renderHearts() {
   568→        bool blink = (minecraft->player->invulnerableTime / 3) % 2 == 1;
   569→        if (minecraft->player->invulnerableTime < 10) blink = false;
   570→        int h = minecraft->player->health;
   571→        int oh = minecraft->player->lastHealth;
   572→        random.setSeed(tickCount * 312871);
   573→
   574→        int xx = 2;//screenWidth / 2 - getNumSlots() * 10;
   575→
   576→        int armor = minecraft->player->getArmorValue();
   577→        for (int i = 0; i < Player::MAX_HEALTH / 2; i++) {
   578→                int yo = 2;
   579→                int ip2 = i + i + 1;
   580→
   581→                if (armor > 0) {
   582→                    int xo = xx + 80 + i * 8 + 4;
   583→                    if (ip2 < armor) blit(xo, yo, 16 + 2 * 9, 9 * 1, 9, 9);
   584→                    else if (ip2 == armor) blit(xo, yo, 16 + 4 * 9, 9 * 1, 9, 9);
   585→                    else if (ip2 > armor) blit(xo, yo, 16 + 0 * 9, 9 * 1, 9, 9);
   586→                }
   587→
   588→                int bg = 0;
   589→                if (blink) bg = 1;
   590→                int xo = xx + i * 8;
   591→                if (h <= 4) {
   592→                        yo = yo + random.nextInt(2) - 1;
   593→                }
   594→                blit(xo, yo, 16 + bg * 9, 9 * 0, 9, 9);
   595→                if (blink) {
   596→                        if (ip2 < oh) blit(xo, yo, 16 + 6 * 9, 9 * 0, 9, 9);
   597→                        else if (ip2 == oh) blit(xo, yo, 16 + 7 * 9, 9 * 0, 9, 9);
   598→                }
   599→                if (ip2 < h) blit(xo, yo, 16 + 4 * 9, 9 * 0, 9, 9);
   600→                else if (ip2 == h) blit(xo, yo, 16 + 5 * 9, 9 * 0, 9, 9);
   601→        }
   602→}

   640→void Gui::renderSleepAnimation( const int screenWidth, const int screenHeight ) {

   647→
   648→        int color = (int) (220.0f * amount) << 24 | (0x101020);
   649→        fill(0, 0, screenWidth, screenHeight, color);
