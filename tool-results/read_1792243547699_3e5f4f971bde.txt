  8739→{
  8740→    TIMER_PUSH("ai");
  8741→
  8742→    if (fleeTime > 0) fleeTime--;
  8743→    holdGround = shouldHoldGround();
  8744→    float maxDist = 16;
  8745→
  8746→    Entity* attackTarget = NULL;
  8747→    if (attackTargetId == 0) {
  8748→        attackTarget = findAttackTarget();    // Animal::findAttackTarget returns NULL (no breeding AI)

  8750→            level->findPath(&path, this, attackTarget, maxDist, false, false);
  8751→            attackTargetId = attackTarget->entityId;
  8752→        }
  8753→    } else {
  8754→        attackTarget = level->getEntity(attackTargetId);
  8755→        if (!attackTarget || !attackTarget->isAlive()) {
  8756→            attackTargetId = 0;
  8757→            attackTarget = NULL;
  8758→        } else {
  8759→            attackTargetId = attackTarget->entityId;
  8760→            float d = attackTarget->distanceTo(this);
  8761→            if (canSee(attackTarget)) {
  8762→                checkHurtTarget(attackTarget, d);
  8763→            } else {
  8764→                checkCantSeeTarget(attackTarget, d);
  8765→            }
  8766→        }
  8767→    }
  8768→    TIMER_POP();
  8769→
  8770→    bool doStroll = false;
  8771→    if (!holdGround && (attackTarget != NULL && (path.isEmpty() || random.nextInt(20) == 0))) {
  8772→        level->findPath(&path, this, attackTarget, maxDist, false, false);
  8773→    } else if (!holdGround) {
  8774→        if (path.isEmpty() && (random.nextInt(180) == 0)) {
  8775→            doStroll = true;       // 1/180 chance per tick to pick a new stroll destination when idle
  8776→        } else {
  8777→            if (random.nextInt(120) == 0) doStroll = true;       // 1/120 chance
  8778→            else if (fleeTime > 0 && (fleeTime&7) == 1) doStroll = true;   // fleeing: every 8th tick of fleeTime
  8779→        }
  8780→    }
  8781→    if (doStroll) {
  8782→        if (noActionTime < SharedConstants::TicksPerSecond * 5) {   // only stroll if player nearby (5s no-action)
  8783→            findRandomStrollLocation();
  8784→        }
  8785→    }
  8786→
  8787→    int yFloor = Mth::floor(bb.y0 + .5f);
  8788→
  8789→    bool inWater = isInWater();
  8790→    bool inLava = isInLava();
  8791→    xRot = 0;
  8792→    if (path.isEmpty() || random.nextInt(100) == 0) {
  8793→        super::updateAi();    // Mob::updateAi (look-around / random turn) — see 16.13
  8794→        return;
  8795→    }
  8796→

  8804→            looping = false;
  8805→            path.destroy();
  8806→        } else target = path.currentPos(this);
  8807→    }

  8812→        float zd = target.z - z;

  8826→                float zd2 = attackTarget->z - z;

  8836→        }

  8850→**Wander-AI summary (PathfinderMob::updateAi):**
  8851→- `fleeTime` decrements toward 0 (set to 60 ticks when hurt).
  8852→- No target: 1/180 chance/tick to `findRandomStrollLocation` when idle (path empty); 1/120 chance when path is non-empty (occasional re-stroll).
  8853→- During flee (fleeTime > 0): every 8th tick (`fleeTime & 7 == 1`) re-strolls.
  8854→- `noActionTime` (set by `Mob::updateAi`) must be < 5 sec (`SharedConstants::TicksPerSecond * 5 = 100` ticks) or no stroll — this prevents idle animals far from players from wandering.
  8855→- When path is empty OR 1/100 chance: just call `super::updateAi` (look around).
  8856→- Otherwise, follow the path: turn toward the next node (max 30°/tick), set `yya = runSpeed`, jump if target is above current.
  8857→- Hold-ground: rotate toward attack target and strafe (`xxa`/`yya` re-projected).
  8858→- Swimming: 80% chance/tick to jump while in water/lava.
  8859→- Hit a wall while not path-finding: jump.
