   269→   269→void Minecraft::setLevel(Level* level, const std::string& message /* ="" */, LocalPlayer* forceInsertPlayer /* = NULL */) {
   270→   270→        cameraTargetPlayer = NULL;
   271→   271→        LOGI("Seed is %ld\n", level->getSeed());
   272→   272→
   273→   273→        if (level != NULL) {
   274→   274→                level->raknetInstance = raknetInstance;
   275→   275→        gameMode->initLevel(level);
   276→   276→
   277→   277→                if (!player && forceInsertPlayer)
   278→   278→                {
   279→   279→                        player = forceInsertPlayer;
   280→   280→                        player->resetPos(false);
   281→   281→                        //level->addEntity(forceInsertPlayer);
   282→   282→                }
   283→   283→                else if (player != NULL) {
   284→   284→                        player->resetPos(false);
   285→   285→                        if (level != NULL) {
   286→   286→                                level->addEntity(player);
   287→   287→                        }
   288→   288→                }
   289→   289→                this->level = level;
   290→   290→                _hasSignaledGeneratingLevelFinished = false;
   291→   291→#ifdef STANDALONE_SERVER
   292→   292→                const bool threadedLevelCreation = false;
   293→   293→#else
   294→   294→                const bool threadedLevelCreation = true;
   295→   295→#endif
   296→   296→
   297→   297→                if (threadedLevelCreation) {
   298→   298→                        // Threaded
   299→   299→                        // "Lock"
   300→   300→                        isGeneratingLevel = true;
   301→   301→                        generateLevelThread = new CThread(Minecraft::prepareLevel_tspawn, this);
   302→   302→                } else {
   303→   303→                        // Non-threaded
   304→   304→                        generateLevel("Currently not used", level);
   305→   305→                }
   306→   306→    } else {
   307→   307→        player = NULL;
   308→   308→    }
   309→   309→
   310→   310→    this->lastTickTime = 0;
   311→   311→        this->_running = true;
   312→   312→}

   314→   314→void Minecraft::leaveGame(bool renameLevel /*=false*/)
   315→   315→{
   316→   316→    if (isGeneratingLevel || !_hasSignaledGeneratingLevelFinished)
   317→   317→        return;
   318→   318→    
   319→   319→        isGeneratingLevel = false;
   320→   320→        bool saveLevel = level && (!level->isClientSide || renameLevel);
   321→   321→
   322→   322→        raknetInstance->disconnect();
   323→   323→        if (saveLevel) {
   324→   324→                // If server or wanting to save level as client, save all unsaved chunks!
   325→   325→                level->getChunkSource()->saveAll(true);
   326→   326→        }
   327→   327→
   328→   328→        LOGI("Clearing levels\n");
   329→   329→
   330→   330→        cameraTargetPlayer = NULL;
   331→   331→#ifndef STANDALONE_SERVER
   332→   332→        levelRenderer->setLevel(NULL);
   333→   333→        particleEngine->setLevel(NULL);
   334→   334→#endif
   335→   335→        LOGI("Erasing callback\n");
   336→   336→        delete netCallback;
   337→   337→        netCallback = NULL;
   338→   338→
   339→   339→        LOGI("Erasing level\n");
   340→   340→        if (level != NULL) {
   341→   341→                delete level->getLevelStorage();
   342→   342→                delete level;
   343→   343→                level = NULL;
   344→   344→        }
   345→   345→        //delete player;
   346→   346→        player = NULL;
   347→   347→        cameraTargetPlayer = NULL;
   348→   348→
   349→   349→        _running = false;
   350→   350→#ifndef STANDALONE_SERVER
   351→   351→        if (renameLevel) {
   352→   352→                setScreen(new RenameMPLevelScreen(LevelStorageSource::TempLevelId));
   353→   353→        }
   354→   354→        else
   355→   355→                screenChooser.setScreen(SCREEN_STARTMENU);
   356→   356→#endif
   357→   357→}

   359→   359→void Minecraft::prepareLevel(const std::string& title) {
   360→   360→        LOGI("status: 1\n");
   361→   361→        progressStageStatusId = 1;
   362→   362→
   363→   363→        Stopwatch A, B, C, D;
   364→   364→        A.start();
   365→   365→
   366→   366→        Stopwatch L;
   367→   367→
   368→   368→        // Dont update lights if we load the level (ok, actually just with leveldata version=1.+(?))
   369→   369→        if (!level->isNew())
   370→   370→                level->setUpdateLights(false);
   371→   371→
   372→   372→        int Max = CHUNK_CACHE_WIDTH * CHUNK_CACHE_WIDTH;
   373→   373→        int pp = 0;
   374→   374→        for (int x = 8; x < (CHUNK_CACHE_WIDTH * CHUNK_WIDTH); x += CHUNK_WIDTH) {
   375→   375→        for (int z = 8; z < (CHUNK_CACHE_WIDTH * CHUNK_WIDTH); z += CHUNK_WIDTH) {
   376→   376→            progressStagePercentage = 100 * pp++ / Max;
   377→   377→            //printf("level generation progress %d\n", progressStagePercentage);
   378→   378→                        B.start();
   379→   379→            level->getTile(x, 64, z);
   380→   380→                        B.stop();
   381→   381→                        L.start();
   382→   382→                        if (level->isNew())
   383→   383→                                while (level->updateLights())
   384→   384→                                        ;
   385→   385→                        L.stop();
   386→   386→        }
   387→   387→    }
   388→   388→        A.stop();
   389→   389→        level->setUpdateLights(true);
   390→   390→
   391→   391→        C.start();
   392→   392→        for (int x = 0; x < CHUNK_CACHE_WIDTH; x++)
   393→   393→        {
   394→   394→                for (int z = 0; z < CHUNK_CACHE_WIDTH; z++)
   395→   395→                {
   396→   396→                        LevelChunk* chunk = level->getChunk(x, z);
   397→   397→                        if (chunk && !chunk->createdFromSave)
   398→   398→                        {
   399→   399→                                chunk->unsaved = false;
   400→   400→                                chunk->clearUpdateMap();
   401→   401→                        }
   402→   402→                }
   403→   403→        }
   404→   404→        C.stop();
   405→   405→
   406→   406→        LOGI("status: 3\n");
   407→   407→        progressStageStatusId = 3;
   408→   408→        if (level->isNew()) {
   409→   409→                level->setInitialSpawn(); // @note: should obviously be called from Level itself
   410→   410→                level->saveLevelData();
   411→   411→                level->getChunkSource()->saveAll(false);
   412→   412→                level->saveGame();
   413→   413→        } else {
   414→   414→                level->saveLevelData();
   415→   415→                level->loadEntities();
   416→   416→        }
   417→   417→
   418→   418→        progressStagePercentage = -1;
   419→   419→        progressStageStatusId = 2;
   420→   420→        LOGI("status: 2\n");
   421→   421→
   422→   422→        D.start();
   423→   423→        level->prepare();
   424→   424→        D.stop();
   425→   425→
   426→   426→        A.print("Generate level: ");
   427→   427→        L.print(" - light: ");
   428→   428→        B.print(" - getTl: ");
   429→   429→        C.print(" - clear: ");
   430→   430→        D.print(" - prepr: ");
   431→   431→        progressStageStatusId = 0;
   432→   432→}

   508→   508→void Minecraft::tick(int nTick, int maxTick) {
   509→   509→        if (missTime > 0) missTime--;
   510→   510→#ifndef STANDALONE_SERVER
   511→   511→        if (!screen && player) {
   512→   512→                if (player->health <= 0) {
   513→   513→                        setScreen(new DeathScreen());
   514→   514→                }
   515→   515→        }
   516→   516→#endif
   517→   517→        TIMER_PUSH("gameMode");
   518→   518→        if (level && !pause) {
   519→   519→                gameMode->tick();
   520→   520→        }
   521→   521→
   522→   522→        TIMER_POP_PUSH("commandServer");
   523→   523→        if (level && _commandServer) {
   524→   524→                _commandServer->tick();
   525→   525→        }
   526→   526→
   527→   527→        TIMER_POP_PUSH("input");
   528→   528→        tickInput();
   529→   529→#ifndef STANDALONE_SERVER
   530→   530→        TIMER_POP_PUSH("gui");
   531→   531→        gui.tick();
   532→   532→#endif
   533→   533→        //
   534→   534→        // Ongoing level generation in a (perhaps) different thread. When it's
   535→   535→        // ready, _levelGenerated() is called once and any threads are deleted.
   536→   536→        //
   537→   537→        if (isGeneratingLevel) {
   538→   538→                return;
   539→   539→        } else if (!_hasSignaledGeneratingLevelFinished) {
   540→   540→                if (generateLevelThread) {
   541→   541→                        delete generateLevelThread;
   542→   542→                        generateLevelThread = NULL;
   543→   543→                }
   544→   544→                _levelGenerated();

   547→   547→        //
   548→   548→        // Normal game loop, run before or efter level generation
   549→   549→        //

   553→   553→#ifndef STANDALONE_SERVER

   568→   568→                        if (player) {
   569→   569→                                level->animateTick(Mth::floor(player->x), Mth::floor(player->y), Mth::floor(player->z));
   570→   570→                        }
   571→   571→#endif
   572→   572→                }
   573→   573→        }
   574→   574→#ifndef STANDALONE_SERVER
   575→   575→        textures->loadAndBindTexture("terrain.png");
   576→   576→        if (!pause && !(screen && !screen->renderGameBehind())) {
   577→   577→                #if !defined(RPI)
   578→   578→                        #ifdef __APPLE__
   579→   579→                        if (isSuperFast())
   580→   580→                        #endif
   581→   581→                        {
   582→   582→                        if (nTick == maxTick) {
   583→   583→                                TIMER_POP_PUSH("textures");
   584→   584→                                textures->tick(true);
   585→   585→                        }
   586→   586→                        }
   587→   587→                #endif
   588→   588→        }
   589→   589→        TIMER_POP_PUSH("particles");
   590→   590→        particleEngine->tick();
   591→   591→        if (screen) {
   592→   592→                screenMutex = true;
   593→   593→                screen->tick();
   594→   594→                screenMutex = false;
   595→   595→        }
   596→   596→
   597→   597→        // @note: fix to keep "isPressed" and "isReleased" as long as necessary.
   598→   598→        //      Most likely keyboard and mouse could/should be reset here as well.
   599→   599→        Multitouch::reset();
   600→   600→#endif
   601→   601→        TIMER_POP();
   602→   602→}

   953→   953→void Minecraft::handleBuildAction(BuildActionIntention* action) {
   954→   954→#ifndef STANDALONE_SERVER
   955→   955→        if (action->isRemove()) {
   956→   956→                if (missTime > 0) return;
   957→   957→                player->swing();
   958→   958→        }
   959→   959→        if(player->isUsingItem())
   960→   960→                return;
   961→   961→    bool mayUse = true;
   962→   962→
   963→   963→        if (!hitResult.isHit()) {
   964→   964→                if (action->isRemove() && !gameMode->isCreativeType()) {
   965→   965→                        missTime = 10;
   966→   966→                }
   967→   967→    } else if (hitResult.type == ENTITY) {
   968→   968→        if (action->isAttack()) {
   969→   969→                        player->swing();
   970→   970→                        //LOGI("attacking!\n");
   971→   971→                        InteractPacket packet(InteractPacket::Attack, player->entityId, hitResult.entity->entityId);
   972→   972→                        raknetInstance->send(packet);
   973→   973→            gameMode->attack(player, hitResult.entity);
   974→   974→                } else if (action->isInteract()) {
   975→   975→                        if (hitResult.entity->interactPreventDefault())
   976→   976→                                mayUse = false;
   977→   977→                        //LOGI("interacting!\n");
   978→   978→                        InteractPacket packet(InteractPacket::Interact, player->entityId, hitResult.entity->entityId);
   979→   979→                        raknetInstance->send(packet);
   980→   980→            gameMode->interact(player, hitResult.entity);
   981→   981→        }
   982→   982→    } else if (hitResult.type == TILE) {
   983→   983→                int x = hitResult.x;
