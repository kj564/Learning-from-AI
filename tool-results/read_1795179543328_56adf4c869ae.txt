   264→  async function waitFor(pred, timeout) {
   265→    const t0 = Date.now();
   266→    while (Date.now() - t0 < timeout) {
   267→      if (pred()) return true;
   268→      await sleep(120);
   269→    }
   270→    return false;
   271→  }

   273→  // Park until the AI tab is the FOREGROUND tab of its window again (or the user
   274→  // halts). Deliberately has NO time cap: a tab minimized/backgrounded for an
   275→  // hour must resume cleanly, not silently time out into a "could not send /
   276→  // not run" failure. (A plain waitFor with a finite timeout returned false on a
   277→  // long minimize, which broke the send loop and ended the agent loop.) Returns
   278→  // false ONLY if the user stopped while we were parked, so callers can break.
   279→  // EVENT-DRIVEN, not polled. A chained setTimeout loop is subject to Chrome's
   280→  // "intensive throttling": once a tab has been hidden >5 min, chained timers are
   281→  // clamped to ONE tick per minute, so a sleep(300) poll could take up to a full
   282→  // minute to notice the tab came back - the user sees the bar sit on "Paused"
   283→  // long after they returned. visibilitychange fires immediately on unhide, so we
   284→  // race it against a slow stop-poll (the stop path is not latency-critical).
   285→  async function waitVisible() {
   286→    if (!document.hidden || A.stop) return !A.stop;
   287→    // Multi-tab opt-in: a hidden tab may keep working. Read the flag through the
   288→    // awaitable so a session started immediately after load cannot race it.
   289→    if (await backgroundSessionsReady()) {
   290→      // Mark the state for the diagnostics only; never place a cover over a tab
   291→      // the user cannot see anyway.
   292→      diag("park.skipped.background", { configured: true });
   293→      return !A.stop;
   294→    }
   295→    A.parked = true;
   296→    try { ui.setStarting(); } catch {}
   297→    try {
   298→      await new Promise((resolve) => {
   299→        const done = () => {
   300→          document.removeEventListener("visibilitychange", onVis);
   301→          clearInterval(iv);
   302→          resolve();
   303→        };
   304→        const onVis = () => { if (!document.hidden) done(); };
   305→        document.addEventListener("visibilitychange", onVis);
   306→        // Safety net only: covers a stop click while parked, and the (unlikely)
   307→        // case of a missed visibilitychange event.
   308→        const iv = setInterval(() => { if (A.stop || !document.hidden) done(); }, 1000);
   309→      });
   310→    } finally {
   311→      A.parked = false;
   312→      try { ui.setStarting(); } catch {}
   313→    }
   314→    return !A.stop;
   315→  }

   318→  // can slide their timers forward by that amount. Without this, every deadline

   347→  async function submitAndGetBase(text, images) {
   348→    captureSendToken();
   349→    diag("send", { text: String(text).slice(0, 60), busy: P.isBusyNow() });
   350→    A.injecting = true;
   351→    ui.inputCover(true);
   352→    try {
   353→      // Quick 2-point settle: sample the previous response's stream length before
   354→      // and after a 200ms yield. A one-shot React batch flush (the common case)
   355→      // shows no second growth and costs only 200ms. A genuinely still-generating
   356→      // stream shows growth → fall back to the full idle wait.
   357→      const _settleItem = P.lastAssistant();
   358→      const _settleLen0 = _settleItem ? P.streamLen(_settleItem) : 0;
   359→      await sleep(200);
   360→      if (_settleItem && _settleItem === P.lastAssistant() &&
   361→          P.streamLen(_settleItem) > _settleLen0) {
   362→        await waitFor(() => !P.isGenerating(), 4000);
   363→      }
   364→      const base = P.assistantCount();
   365→      const preUser = P.userCount();
   366→      // Arm the optimistic pre-hide for the result turn we're about to inject:
   367→      // the very next NEW user turn is ours, so preHideWholeItems can mask it on
   368→      // creation instead of waiting for its "Output of '…'" caption to render
   369→      // (which lands a tick after the node - especially with an attached image -
   370→      // and would otherwise flash the raw output for the 200/700ms until a sweep
   371→      // nudge catches it). See preHideWholeItems.
   372→      A.injectPreUser = preUser;
   373→      A.injectHideUntil = Date.now() + 2500;
   374→      // "Landed" = a new turn appeared in the DOM. In long chats, list
   375→      // virtualisation can keep counts flat even when our message landed - the
   376→      // textarea-cleared signal below is the primary fast gate.
   377→      const landed = () => P.userCount() > preUser || P.assistantCount() > base;
   378→      // CRITICAL: never type/send while the tab is HIDDEN. Background tabs throttle
   379→      // rendering, which made the landed-check unreliable and caused the SAME
   380→      // feedback to be sent several times. Send ONLY while visible.
   381→      let tries = 0;
   382→      let messageSent = false;
   383→      const maxTries = Number.isInteger(P.sendRetryLimit) && P.sendRetryLimit > 0
   384→        ? P.sendRetryLimit
   385→        : 4;
   386→      while (!messageSent && !landed() && tries < maxTries && !A.stop) {
   387→        if (document.hidden) {
   388→          diag("send.waitVisible", { tries });
   389→          if (!(await waitVisible()) || A.stop) break; // park (no cap) until foreground; break only on user stop
   390→        }
   391→        await jitterBeforeSend();
   392→        diag("submit.typeAndSend", { hasImages: !!(images && images.length) });
   393→        await P.typeAndSend(text, images);
   394→        // Re-arm the pre-hide window NOW that typeAndSend has returned (the send
   395→        // was just clicked, so our result turn is about to render). The initial
   396→        // arm above can EXPIRE during an image upload - typeAndSend blocks ~3-6s
   397→        // uploading the capture before the turn appears, past the 2.5s window - so
   398→        // without this re-arm the raw "Output of…" + a still-loading (0-byte)
   399→        // thumbnail flash for image feedbacks until a sweep chip lands. Safe: the
   400→        // input is covered and the loop owns this send, so no user turn can slip
   401→        // into the window, and the pre-hide is one-shot (consumes the first turn).
   402→        A.injectHideUntil = Date.now() + 2500;
   403→        // The site clears the textarea as soon as the send is accepted - faster
   404→        // and more reliable than waiting for a DOM turn count change.
   405→        await waitFor(() => {
   406→          if (P.editorText().trim() === "") messageSent = true;
   407→          return messageSent || landed();
   408→        }, 3500);
   409→        tries++;
   410→      }
   411→      if (messageSent) diag("send.cleared", { tries });
   412→      // All retries exhausted with NO evidence the message landed (textarea never
   413→      // cleared, no new turn). Silently returning here left the loop waiting for
   414→      // a reply that will never come (~60s "empty" timeout) with zero explanation
   415→      // - the reported "the tool result just never gets injected" symptom. Tell
   416→      // the user what actually happened so they can nudge the conversation
   417→      // themselves instead of watching a stuck bar.
   418→      if (!messageSent && !landed() && !A.stop) {
   419→        diag("send.failed", { tries });
   420→        ui.banner("warn", "Message could not be sent",
   421→          `Send "continue" to resume.`);
   422→      }
   423→      return base;
   424→    } finally {
   425→      // During Starting Up / the agent loop, the bootstrap or loop owns the cover
   426→      // for the whole phase, so don't lift it here between an injection and the
   427→      // next waitForResponse - it stays up until the loop / bootstrap ends.
   428→      if (!A.starting && !A.running) ui.inputCover(false);
   429→      setTimeout(() => (A.injecting = false), 400);
   430→      // Camouflage the turn we just injected without waiting on the rAF observer
   431→      // (paused in a background tab). A couple of nudges cover the render.
   432→      setTimeout(scheduleSweep, 200);
   433→      setTimeout(scheduleSweep, 700);
   434→    }
   435→  }

   441→  // and agentLoop so the user sees one explanation, not two dialects.
   442→  const FALLBACK_TURN_TEXT =
   443→    "ChatGPT returned an empty placeholder reply for this conversation - open a new chat and Start again.";
   444→
   445→  async function waitForResponse(base) {
   446→    const t0 = Date.now();
   447→    // INACTIVITY timeout (not total-elapsed): the loop only gives up after this
   448→    // long with NO streaming AND no text change. lastActiveAt is refreshed every
   449→    // tick the model is generating or the reply text grows, so an arbitrarily
   450→    // LONG but still-active response never trips it (the old total-elapsed cap
   451→    // wrongly fired "No response" while the model was still writing past 300s).
   452→    const TIMEOUT = T.RESPONSE_TIMEOUT_MS;
   453→    let lastActiveAt = Date.now();
   454→    const STABLE_MS = T.STABLE_MS; // generating-flag stuck ON but text frozen → done
   455→    let started = false, doneSince = 0, lastLimitScan = 0;
   456→    let lastText = null, lastChangeAt = Date.now(), genFalseSince = 0;
   457→    // ── DIAG: finalisation-latency instrumentation (multi_edit "slow" probe) ──
   458→    // genOffFirstAt: the FIRST moment gen went false after streaming began (does
   459→    // NOT reset on flicker, unlike genFalseSince). genFlickers: how many times gen
   460→    // flipped back true after having been false - a high count means post-stop DOM
   461→    // churn (or a wedged stop button) is what keeps the watcher alive. waitedBlock/
   462→    // waitedFlicker: iterations spent waiting because effectiveBlock held vs because
   463→    // gen was (re)true. These pinpoint which gate causes any tail latency.
   464→    let genOffFirstAt = 0, genFlickers = 0, prevGen = null;
   465→    let waitedBlock = 0, waitedFlicker = 0;
   466→    const finalizeDiag = (kind) => {
   467→      const now = Date.now();
   468→      diag("stopGoneToResp", {
   469→        kind,
   470→        stopGoneToRespMs: genOffFirstAt ? now - genOffFirstAt : null,
   471→        genStableForMs: genFalseSince ? now - genFalseSince : null,
   472→        lastChangeAgoMs: now - lastChangeAt,
   473→        genFlickers, waitedBlock, waitedFlicker,
   474→        totalMs: now - t0,
   475→      });
   476→    };
   477→    let preStartSilent = 0; // nothing produced AND not generating
   478→    let curItem = null, sawContent = false, warmSince = 0; // per-turn "warming up"
   479→    // Last NON-EMPTY reply read for the CURRENT turn. Sites re-render a turn's
   480→    // subtree (React/Monaco churn) and a read can come back "" for a frame at
   481→    // the exact moment the watcher finalizes - the turn then ended as
   482→    // kind:"empty" even though a (possibly cut-off) command was sitting there a
   483→    // tick earlier, leaving a DEAD turn: no parse_error feedback, and the
   484→    // autoResume dedupe (zResume) blocks any later retry (validated live on a
   485→    // Qwen post-stop regenerate, 2026-07). Classify on this fallback instead of
   486→    // declaring empty. Reset whenever the turn NODE changes so a new turn can
   487→    // never inherit the previous turn's text.
   488→    let lastGoodReply = "";
   489→    let reasonSince = 0; // reasoning written but no answer yet (loading phase)
   490→    let noTurnSince = 0; // finalize attempted before this send's reply turn exists
   491→    let unsettledSince = 0; // command-shaped reply whose read is not yet stable
   492→    const WARMUP_MS = T.WARMUP_MS;
   493→    const REASON_NOREPLY_MS = T.REASON_NOREPLY_MS;
   494→    const NO_TURN_GRACE_MS = 30000;
   495→    // Upper bound on holding off a parse verdict while a provider reports its
   496→    // read is unsettled (Qwen A/B dual turn still landing). A genuinely stuck
   497→    // read still resolves after this and is parsed as-is.
   498→    const UNSETTLED_GRACE_MS = 8000;
   499→    // Once the generating flag has been OFF this long, the model has clearly
   500→    // stopped streaming - so an "open tool block" reading is a DOM-churn/parse
   501→    // artifact, not live output, and must not keep the watcher waiting. Provider
   502→    // -neutral: while a model is genuinely streaming, gen stays true and this is
   503→    // never reached.
   504→    const GEN_STOP_GRACE_MS = 2500;
   505→
   506→    while (Date.now() - lastActiveAt < TIMEOUT) {
   507→      if (A.stop) return { kind: "stopped" };
   508→      // SCROLL-AWAY RULE: while the user scrolls up, NOTHING is read or run.
   509→      // Hold here (and keep the inactivity clock alive, since this is the user
   510→      // browsing, not a stalled model) so no scroll-rendered history can be
   511→      // interpreted as a fresh command. Resumes as soon as the gesture window
   512→      // lapses; a genuine run that was already in flight is untouched.
   513→      if (scrollSuppressed()) {
   514→        lastActiveAt = Date.now();
   515→        await sleep(250);
   516→        continue;
   517→      }
   518→      // NEVER let a deadline expire while the tab is in the background. A hidden
   519→      // tab has no layout (innerText reads come back "", getBoundingClientRect is
   520→      // 0x0) and Chrome throttles its timers, so every read here is unreliable -
   521→      // and the site may legitimately keep streaming for as long as the user is
   522→      // away in Studio. Park until the tab is foreground again, then slide EVERY
   523→      // deadline forward by the time we were parked so nothing that was mid-flight
   524→      // when the user switched away expires the moment they come back. This is the
   525→      // fix for "I switched to Studio, came back and the command says 'not run'":
   526→      // the loop used to burn its 5-minute inactivity budget off-screen, end with
   527→      // "No response from <site>", and orphan the pending command.
   528→      if (document.hidden && !A.stop && !(await backgroundSessionsReady())) {
   529→        const parked = await parkHidden();
   530→        if (A.stop) return { kind: "stopped" };
   531→        if (parked) {
   532→          lastActiveAt += parked; lastChangeAt += parked;
   533→          if (doneSince) doneSince += parked;
   534→          if (genFalseSince) genFalseSince += parked;
   535→          if (preStartSilent) preStartSilent += parked;
   536→          if (warmSince) warmSince += parked;
   537→          if (reasonSince) reasonSince += parked;
   538→          if (noTurnSince) noTurnSince += parked;
   539→          if (unsettledSince) unsettledSince += parked;
   540→          if (genOffFirstAt) genOffFirstAt += parked;
   541→        }
   542→        continue; // re-read everything now that the tab has layout again
   543→      }
   544→      const gen = P.isGenerating();
   545→      if (gen) lastActiveAt = Date.now(); // actively generating ⇒ never time out
   546→      const d = P.readAssistant();
   547→      // Sites virtualize their lists, so the absolute assistant count can DROP
   548→      // even as a new reply is added. A count increase still proves a new turn
   549→      // appeared; the generating flag is the reliable "reply has begun" signal.
   550→      // A new reply turn exists. Prefer node IDENTITY (virtualization-proof) when
   551→      // the provider exposes it: the last assistant turn's id differs from the one
   552→      // captured at send time. Fall back to the count test otherwise. Without this,
   553→      // a provider's list virtualisation can keep assistantCount() <= base for a
   554→      // fresh reply, so the reliableCounts gate below waits out the full NO_TURN_GRACE
   555→      // (~30s) before finalising a multi_edit - the "input box stuck until I scroll
   556→      // up" symptom (scrolling re-attached old turns and bumped the count).
   557→      const curTok = P.lastAssistantId ? P.lastAssistantId() : undefined;
   558→      // A NULL token means the provider could not read an identity for the
   559→      // CURRENT last turn (not that the provider lacks ids - that's undefined).
   560→      // Treating null as "no new reply" wedged the watcher on Qwen: a
   561→      // REGENERATED turn is rebuilt WITHOUT the id attribute the normal turns
   562→      // carry, so curTok stayed null, `started` never latched, and the loop
   563→      // sat in the pre-start branch for the full 60s before ending "empty" -
   564→      // the regenerated command (complete in the net tap) was never run and
   565→      // zResume then blocked any retry (validated live via empty.why, 2026-07).
   566→      // Fall back to the count test instead, exactly as for a provider with no
   567→      // lastAssistantId at all.
   568→      const newReply = (curTok !== undefined && curTok !== null)
   569→        ? (curTok !== A.sendToken)
   570→        : P.assistantCount() > base;
   571→
   572→      // Track whether the CURRENT turn has produced anything. Reset when the
   573→      // turn node changes (the PREVIOUS turn's content never counts).
   574→      if (d.item !== curItem) { curItem = d.item; sawContent = false; warmSince = 0; lastGoodReply = ""; }
   575→      if ((d.reply && d.reply.length) || (d.thinking && d.thinking.length)) sawContent = true;
   576→      if (d.reply && d.reply.length) lastGoodReply = d.reply;
   577→
   578→      if (!started) {
   579→        // CRITICAL: a bare count increase is NOT enough - the empty turn
   580→        // CONTAINER can appear seconds before the first token. Require actual
   581→        // CONTENT (or the generating flag).
   582→        const hasText = !!((d.reply && d.reply.length) || (d.thinking && d.thinking.length));
   583→        if (gen || (newReply && hasText)) { started = true; }
   584→        else {
   585→          // The site can be slow to even CREATE the reply turn. Keep waiting -
   586→          // only give up after a long fully-silent window.
   587→          if (!preStartSilent) preStartSilent = Date.now();
   588→          // diag: WHICH empty-branch fired matters - a dead post-regenerate turn
   589→          // on Qwen kept ending "empty" with a complete command in the net tap,
   590→          // and without the branch name the cause was unfindable from the log.
   591→          if (Date.now() - preStartSilent > 60000) { diag("empty.why", { branch: "preStart", rep: (d.reply||"").length }); return { kind: "empty" }; }
   592→          await sleep(200);
   593→          continue;
   594→        }
   595→      }
   596→
   597→      // Track text stability (independent of the generating flag). Compare the
   598→      // NORMALISED reply (collapsed whitespace) so cosmetic re-renders of a large
   599→      // reply - React re-creating the hidden tool <pre>, syntax-highlight passes,
   600→      // copy-bar text churn - don't count as real "changes" and keep resetting
   601→      // lastChangeAt. A churn-poisoned lastChangeAt was stalling finalisation of
   602→      // big multi_edit blocks ~30s (stuckDone never fired); this can only ever
   603→      // reduce false changes, so short replies / other providers are unaffected.
   604→      const replyNorm = (d.reply || "").replace(/\s+/g, " ").trim();
   605→      if (replyNorm !== lastText) { lastText = replyNorm; lastChangeAt = Date.now(); lastActiveAt = Date.now(); }
   606→      // How long the generating flag has been OFF. A mid-stream flicker resets
   607→      // this the instant growth resumes and gen flips back on.
   608→      if (gen) genFalseSince = 0; else if (!genFalseSince) genFalseSince = Date.now();
   609→      // DIAG: first gen-off, and count flickers back to true after a gen-off.
   610→      if (started && !gen && !genOffFirstAt) genOffFirstAt = Date.now();
   611→      if (prevGen === false && gen && genOffFirstAt) genFlickers++;
   612→      prevGen = gen;
   613→
   614→      if (Date.now() - lastLimitScan > 1000) {
   615→        lastLimitScan = Date.now();
   616→        const ctx = P.scanError();
   617→        if (ctx) return { kind: "context_limit", detail: ctx };
   618→      }
   619→
   620→      // Keep waiting while a tool command is still being streamed (opener written
   621→      // but no end marker yet) so we never parse/finalize half a command.
   622→      const blockActive = ZSParse.hasOpenToolBlock(d.reply) && Date.now() - lastChangeAt < 6000;
   623→      // ...but once generation has clearly stopped (stop indicator gone past the
   624→      // grace window), stop honoring an "open block" - it is DOM churn, not live
   625→      // streaming. Lets a finished big block finalise in seconds instead of
   626→      // waiting out ~30s of re-render churn. Safe: real streaming keeps gen true.
   627→      const genStopped = !gen && genFalseSince && Date.now() - genFalseSince > GEN_STOP_GRACE_MS;
   628→      const effectiveBlock = blockActive && !genStopped;
   629→
   630→      // Fallback: generating flag stuck ON (e.g. a wedged stop button - seen
   631→      // live on Gemini after a mid-write halt) but the text has been frozen for
   632→      // a while → stop waiting and finalize. This must BYPASS the gen branch
   633→      // below entirely: falling through while gen stays true used to reset
   634→      // doneSince every iteration, so the watcher never finalized at all.
   635→      // ...but NEVER treat a still-OPEN command block as "done" while the site is
   636→      // genuinely still generating. A model writing a big command (a 3799-char
   637→      // execute_luau seen live on GLM) can pause >STABLE_MS between tokens - that
   638→      // is a mid-write gap, NOT a wedged stop button on a COMPLETE reply. Firing
   639→      // here parsed the half-written JSON and stamped a false "bad JSON" error
   640→      // while GLM was still typing. RESPONSE_TIMEOUT still bounds a truly stuck one.
   641→      const stuckDone = started && d.reply && Date.now() - lastChangeAt > STABLE_MS &&
   642→        !(gen && ZSParse.hasOpenToolBlock(d.reply));
   643→      if ((gen || effectiveBlock) && !stuckDone) {
   644→        // DIAG: attribute this wait. genOffFirstAt set ⇒ we are PAST first stop,
   645→        // so any wait here is tail latency: either gen flickered back on, or an
   646→        // (effective) open-block reading is holding us.
   647→        if (genOffFirstAt) { if (gen) waitedFlicker++; else if (effectiveBlock) waitedBlock++; }
   648→        doneSince = 0;
   649→        await sleep(160);
   650→        continue;
   651→      }
   652→      if (stuckDone && gen) log("generating flag stuck - falling back to text stability");
   653→
   654→      // On providers whose turn counts are RELIABLE (semantic elements, no
   655→      // list virtualisation - Gemini), never finalize before the reply turn
   656→      // for THIS send exists. The generating flag can flicker off in the gap
   657→      // between the send and the new <model-response> node spawning, and the
   658→      // watcher used to finalize on the PREVIOUS turn's stable text - a
   659→      // premature loop.end rescued only by autoResume 30-45s later (diag
   660→      // showed `response kind:text` ~2.4s after loop.start with rp unchanged).
   661→      // Bounded so a genuinely dead send still ends the turn.
   662→      if (P.reliableCounts && !newReply) {
   663→        if (!noTurnSince) noTurnSince = Date.now();
   664→        // [TRACE] This is the 30s NO_TURN_GRACE gate. If a Qwen tool turn sits here
   665→        // ~30s EVERY turn, newReply is wrongly stuck false: log the identity values
   666→        // that decide it so we can see whether curTok is null (id missing on the new
   667→        // turn -> count fallback) or equal to sendToken (last turn not advancing).
   668→        const _waited = Date.now() - noTurnSince;
   669→        if (_waited > 800 && (!A._noTurnLoggedAt || Date.now() - A._noTurnLoggedAt > 3000)) {
   670→          A._noTurnLoggedAt = Date.now();
   671→          diag("noTurnGrace.wait", {
   672→            waitedMs: _waited,
   673→            curTok: (P.lastAssistantId ? P.lastAssistantId() : undefined),
   674→            sendToken: A.sendToken,
   675→            assistantCount: P.assistantCount ? P.assistantCount() : undefined,
   676→            base, gen, started, replyLen: (d.reply || "").length });
   677→        }
   678→        if (Date.now() - noTurnSince < NO_TURN_GRACE_MS) { await sleep(200); continue; }
   679→      } else {
   680→        noTurnSince = 0;
   681→        A._noTurnLoggedAt = 0;
   682→      }
   683→
   684→      if (!doneSince) doneSince = Date.now();
   685→      if (Date.now() - doneSince < 500) {  // 500ms settle – DOM is stable
   686→        await sleep(120);
   687→        continue;
   688→      }
   689→
   690→      // A turn that has produced NOTHING yet is still warming up - never
   691→      // finalize it as empty/truncated/text (a premature retry interrupts it).
   692→      if (!sawContent) {
   693→        // 2026-11 ChatGPT: a saturated conversation answers with a PLACEHOLDER
   694→        // turn that never gains content. Its only marker is the search-unit key
   695→        //   data-content-search-unit-key="fallback-turn-0:2:assistant"
   696→        // and the bubble holds nothing but "ChatGPT said:". Measured live: the
   697→        // reply text was present in the page's own text but never rendered into
   698→        // the turn, so the loop waited the full WARMUP_MS (45 s) and then blamed
   699→        // the DOM ("empty"). Detect the placeholder immediately and say what it
   700→        // actually is, so the user is told to open a new chat instead of losing
   701→        // an hour to a stuck chip.
   702→        try {
   703→          const it = d && d.item;
   704→          const marker = it && it.querySelector
   705→            ? it.querySelector('[data-content-search-unit-key^="fallback-turn"]')
   706→            : null;
   707→          if (marker && !(d.reply || "").length) {
   708→            if (!warmSince) warmSince = Date.now();
   709→            if (Date.now() - warmSince > 2500) {
   710→              diag("empty.why", { branch: "fallbackTurn", rep: 0 });
   711→              return { kind: "fallback_turn" };
   712→            }
   713→            await sleep(200);
   714→            continue;
   715→          }
   716→        } catch {}
   717→        if (!warmSince) warmSince = Date.now();
   718→        if (Date.now() - warmSince < WARMUP_MS) { await sleep(200); continue; }
   719→        diag("empty.why", { branch: "warmup", rep: (d.reply||"").length, lastGood: lastGoodReply.length });
   720→        return { kind: "empty" };
   721→      }
   722→
   723→      // Still REASONING / loading: thinking written but no answer yet. Don't
   724→      // finalize - wait for the reply, bounded. A manually-stopped turn is
   725→      // exempt so a real stop still ends.
   726→      if (d.thinking && d.thinking.length && !(d.reply && d.reply.length) && !P.turnHalted(d.item)) {
   727→        if (!reasonSince) reasonSince = Date.now();
   728→        if (Date.now() - reasonSince < REASON_NOREPLY_MS) { await sleep(200); continue; }
   729→      } else {
   730→        reasonSince = 0;
   731→      }
   732→
   733→      // Blank-read guard: if THIS read came back empty but the same turn had
   734→      // real text a tick ago, classify that text - see lastGoodReply above.
   735→      let r = d.reply;
   736→      if (!r && lastGoodReply) { r = lastGoodReply; diag("reply.blankReadFallback", { len: r.length }); }
   737→      // "Conversation too long" / "server busy" notices are always SHORT system
   738→      // messages; gating on a short reply stops the model's own long output
   739→      // (which may quote those phrases) from tripping them.
   740→      if (r.length < 400 && P.isTooLongMsg(r)) return { kind: "too_long" };
   741→      // Hold off on any "unparseable command" verdict while the provider reports
   742→      // this turn's text is not yet a settled read. Qwen's A/B "dual" turn is the
   743→      // case: its network tap flips `done` the instant the SSE ends, but the
   744→      // candidate-1 DOM we parse can still be mid-render, so a real command looks
   745→      // half-written for a beat. Firing parse_error there sends an ERROR
   746→      // mid-generation and nags a model that did nothing wrong. Only guard when
   747→      // the reply already LOOKS like a command (so a plain-text answer is never
   748→      // delayed) and bound it with UNSETTLED_GRACE_MS. No-op on providers that
   749→      // don't implement replyUnsettled (DeepSeek/Gemini/GLM/Kimi/Arena).
   750→      const cmdShaped = P.replyUnsettled && (
   751→        ZSParse.hasToolSignature(r) ||
   752→        (ZSParse.LUA_END_RE.test(r) && !ZSParse.LUA_START_RE.test(r)) ||
   753→        (/"(?:datamodel_type|edits|old_string|new_string|file_path|target_file)"\s*:/.test(r) &&
   754→          !/"command"\s*:/.test(r))
   755→      );
   756→      if (cmdShaped && P.replyUnsettled(d.item)) {
   757→        if (!unsettledSince) unsettledSince = Date.now();
   758→        if (Date.now() - unsettledSince < UNSETTLED_GRACE_MS) { await sleep(250); continue; }
   759→      } else {
   760→        unsettledSince = 0;
   761→      }
   762→      // A/B "carousel" turn (Qwen): while it is unresolved the site REMOVES the
   763→      // composer from the DOM (validated live: getEditor() is null), so we can't
   764→      // send the tool result until a candidate is picked - and the read reply is a
   765→      // partial candidate, so a command there looks "cut off". Per the product rule
   766→      // we use the FIRST candidate: wait for BOTH candidates to finish generating
   767→      // (you can't select mid-stream), then auto-select Response 1. That collapses
   768→      // the carousel to a normal turn - composer returns - and the normal parse/run
   769→      // path below handles it. Never a parse_error here (the model didn't truncate).
   770→      // No-op for every provider except Qwen. RESPONSE_TIMEOUT still bounds a truly
   771→      // stuck carousel, so this cannot hang.
   772→      if (P.isComparisonTurn && P.isComparisonTurn(d.item)) {
   773→        if (P.isGenerating()) { await sleep(250); continue; }   // both still writing
   774→        if (P.resolveComparison && P.resolveComparison()) {
   775→          diag("carousel.resolved");
   776→          await sleep(400); continue;                            // let it collapse, re-read
   777→        }
   778→        await sleep(250); continue;                              // button not ready yet
   779→      }
   780→      if (ZSParse.hasToolSignature(r)) {
   781→        const calls = ZSParse.parseToolCalls(r);
   782→        if (calls.length) { finalizeDiag("tool"); return { kind: "tool", calls, item: d.item }; }
   783→        // A half-written command + the site's "Continue" button means the command
   784→        // was truncated mid-stream → resume it rather than reporting bad JSON.
   785→        if (P.findContinueBtn()) return { kind: "truncated", text: r, item: d.item };
   786→        // Only fire parse_error if explicit markers were present.
   787→        if (r.includes(ZSParse.START_M) || ZSParse.LUA_START_RE.test(r)) return { kind: "parse_error", reason: "malformed", raw: r, item: d.item };
   788→        // A command opener with no closer (a JSON object that never closed -
   789→        // the model was halted mid-write and there is no Continue affordance):
   790→        // ask the model to rewrite it instead of silently ending the turn.
   791→        // ...unless ONLY the trailing closers were lost (the model hit its
   792→        // output limit with the payload complete - seen live on Qwen: a big
   793→        // multi_edit missing exactly one final "}"). salvageCutOff auto-closes
   794→        // and runs it instead of burning a whole retry turn; it refuses any
   795→        // cut that amputated real content (mid-string / deep deficit), which
   796→        // still falls through to the parse_error feedback. Safe to run here:
   797→        // generation has ended (the open-block branch above kept waiting
   798→        // while it streamed).
   799→        if (ZSParse.hasOpenToolBlock(r)) {
   800→          const saved = ZSParse.salvageCutOff(r);
   801→          if (saved) {
   802→            diag("tool.salvaged", { name: saved.tool });
   803→            finalizeDiag("tool");
   804→            return { kind: "tool", calls: [saved], item: d.item };
   805→          }
   806→          return { kind: "parse_error", reason: "unclosed", raw: r, item: d.item };
   807→        }
   808→        // A closed-looking JSON command envelope that NAMES A REAL TOOL but failed
   809→        // to parse - typically an unescaped " inside a code/string param broke the
   810→        // JSON (seen live on Kimi's execute_blender_code: `name = "Camera_System"`
   811→        // mid-code). Unlike execute_luau there is NO ###LUA### fallback, so the
   812→        // command silently dropped and the loop finalized the turn as a plain-text
   813→        // answer with no result and no error - a dead turn. Fire a parse_error so
   814→        // the model can fix its JSON. GATED on a known command name so prose that
   815→        // merely quotes {"command":"..."} (a DeepSeek-style explanation, or a
   816→        // placeholder like "command_name") is NOT misread as a broken command and
   817→        // looped on - only a real tool name means a genuine failed call.
   818→        const nm = ZSParse.toolNameFromText(r);
   819→        if (nm && nm !== "command" && (A.toolNames.has(nm) || A.toolNames.has(bareToolName(nm)))) {
   820→          return { kind: "parse_error", reason: "malformed", raw: r, item: d.item };
   821→        }
   822→      }
   823→      // The model answered in its OWN native tool-call markup instead of a
   824→      // snapgentV3 command - DeepSeek's DSML invoke/parameter tags (reported by
   825→      // users, 2026-08; DeepSeek is the only provider seen doing it). It carries
   826→      // no "command"/"tool" key and no ###...### markers, so hasToolSignature is
   827→      // false and every guard below misses it too: the turn finalized as a
   828→      // plain-text answer, the tool never ran, and the loop ended on a dead agent.
   829→      // Checked FIRST among the fallthrough guards because the marker is
   830→      // unambiguous - unlike the guards below it needs no known-tool gate, which
   831→      // matters because the degenerate form seen in the wild carries no tool name
   832→      // at all (a bare tool_calls opener followed by prose).
   833→      if (ZSParse.DSML_RE.test(r)) {
   834→        diag("cmd.dsml", { len: r.length });
   835→        return { kind: "parse_error", reason: "dsml", raw: r, item: d.item };
   836→      }
   837→      // Malformed execute_luau: the model wrote the ###END_LUA### closer but
   838→      // FORGOT the ###LUA### opener, so hasToolSignature missed it and the block
   839→      // never ran (seen on Gemini). Don't silently treat it as a final answer -
   840→      // nudge a rewrite instead of leaving the user stuck on a dead turn.
   841→      if (ZSParse.LUA_END_RE.test(r) && !ZSParse.LUA_START_RE.test(r) && !r.includes(ZSParse.START_M)) {
   842→        return { kind: "parse_error", reason: "luaOpener", raw: r, item: d.item };
   843→      }
   844→      // Malformed command: the model emitted a tool's RAW ARGUMENTS as a bare JSON
   845→      // object (e.g. {"datamodel_type":...,"edits":[...],"file_path":...}) instead of
   846→      // the required {"command":...,"params":...} envelope - it treated the tool as a
   847→      // real callable function (seen on Gemini). Those argument keys never appear in a
   848→      // normal prose answer, so nudge a rewrite rather than ending the turn silently.
   849→      if (/"(?:datamodel_type|edits|old_string|new_string|file_path|target_file)"\s*:/.test(r) &&
   850→          !/"command"\s*:/.test(r)) {
   851→        return { kind: "parse_error", reason: "envelope", raw: r, item: d.item };
   852→      }
   853→      // Malformed command, function-calling flavour: the model named a REAL tool
   854→      // but under the WRONG KEY - {"toolName": "get_studio_state", "studio_id": …}
   855→      // instead of {"command": …, "params": {…}}. It reads as a deliberate call,
   856→      // yet hasToolSignature never fires (no "command" key, no markers) and the
   857→      // argument keys are the tool's own, so neither guard above catches it: the
   858→      // turn finalized as a plain-text answer and the loop ENDED with the user
   859→      // watching a dead agent (seen live on ChatGPT, long session, 2026-08).
   860→      // GATED on the value being a tool we actually have - same reasoning as the
   861→      // toolNameFromText gate above, so prose that merely mentions a tool name in
   862→      // some JSON example is never looped on.
   863→      const wrongKey = /"(?:toolName|tool_name|tool|name|function|action)"\s*:\s*"([A-Za-z0-9_.\/-]+)"/.exec(r);
   864→      if (wrongKey && !/"command"\s*:/.test(r)) {
   865→        const wk = wrongKey[1];
   866→        const hit = knownTool(wk);
   867→        // Log the miss too: the gate depends on how the bridge ADVERTISES names
   868→        // (they can be prefixed per server on a collision), so a silent no-match
   869→        // is exactly the case that is impossible to diagnose after the fact.
   870→        diag("cmd.wrongKey", { name: wk, known: hit, catalogue: A.toolNames.size });
   871→        if (hit) return { kind: "parse_error", reason: "toolKey", raw: r, item: d.item };
   872→      }
   873→      // NOTE: a site "server busy / something went wrong" notice is deliberately
   874→      // NOT special-cased. It falls through to kind:"text" below and simply ENDS
   875→      // the loop as a final answer - no auto-retry. Retrying risked an infinite
   876→      // re-answer loop when the model's OWN prose said "try again", and treating
   877→      // busy as a normal terminal turn is cleaner: the user just re-sends if the
   878→      // site actually hiccuped. (P.isBusyMsg stays on the provider interface,
   879→      // unused by the core, in case a future flow wants it.)
   880→      // The site caps output length and shows a native "Continue" button when it
   881→      // truncates. We try clicking it directly (same turn) in the loop.
   882→      if (P.findContinueBtn()) return { kind: "truncated", text: r, item: d.item };
   883→      if (r === "") { diag("empty.why", { branch: "finalBlank" }); return { kind: "empty" }; }
   884→      return { kind: "text", text: r };
   885→    }
   886→    return { kind: "timeout" };

   892→  // An ORPHANED content script: the extension was reloaded, updated or disabled
   893→  // while this page stayed open, so the script still running here belongs to a
   894→  // version of the extension that no longer exists. Chrome tears down its
   895→  // messaging port, and every chrome.runtime call then fails INSTANTLY with
   896→  // "Extension context invalidated".
   897→  //
   898→  // This must be told apart from a real bridge outage. They are opposite
   899→  // problems with opposite fixes: a bridge outage is fixed by start.bat and
   900→  // resolves itself, while this one can ONLY be fixed by reloading the page and
   901→  // never recovers on its own. Lumping them together (the old behaviour) told
   902→  // the model "the local snapgentV3 bridge is unreachable", which sent the user
   903→  // to check a bridge that was perfectly healthy. Diagnosed 2026-08-14 by the
   904→  // giveaway timing: a genuine outage takes 8s (background.js `send` waits via
   905→  // waitForConnection first), an invalidated context comes back in ~1ms.
   906→  //
   907→  // Every user meets this eventually - Chrome auto-updates extensions under
   908→  // open tabs - so it is worth its own message.
   909→  const isContextInvalidated = (m) =>
   910→    /Extension context invalidated|Receiving end does not exist|message port closed/i.test(m || "");

   912→  function bg(msg) {
   913→    return new Promise((resolve) => {
   914→      const fail = (m) => resolve({
   915→        ok: false,
   916→        kind: isContextInvalidated(m) ? "stale-extension" : "disconnected",
   917→        error: m,
   918→      });
   919→      try {
   920→        chrome.runtime.sendMessage(msg, (resp) => {
   921→          if (chrome.runtime.lastError) fail(chrome.runtime.lastError.message);
   922→          else resolve(resp || { ok: false, kind: "disconnected", error: "no response from background" });
   923→        });
   924→      } catch (e) {
   925→        fail(String(e && e.message || e));
   926→      }
   927→    });

   941→  const bareToolName = (name) => (name && name.includes("/") ? name.split("/").pop() : name) || "";

   943→  // a per-server prefix when two MCP servers expose the same tool (see

   945→  // plain `A.toolNames.has(bare)` misses, and so does bareToolName(bare), which
   946→  // cannot re-add a prefix it never had. Compare bare-to-bare in BOTH directions.
   947→  const bareKey = (n) => String(n || "").split("/").pop().split(".").pop();

   952→    for (const t of A.toolNames) if (bareKey(t) === b) return true;
   953→    return false;

  1026→  async function runTool(call) {
  1027→    const name = call.tool;
  1028→    const args = call.arguments || {};

  1031→    // through history nothing is executed - but the command is HELD, not
  1032→    // cancelled. It used to return "skipped: conversation scrolled up", which
  1033→    // dropped the call entirely: if the user happened to scroll just as a
  1034→    // command became ready, the model was told its work had not run and the
  1035→    // task stalled. The PR describes a hold, so wait the suppression window out
  1036→    // and then proceed. A Stop still wins, and the wait is bounded.
  1037→    if (scrollSuppressed()) {
  1038→      diag("tool.holdScrolling", { name });
  1039→      // Mark the hold so the busy surface keeps the Stop control visible while
  1040→      // we wait; scrollSuppressed() would otherwise hide it for the whole wait.
  1041→      A.held = true;
  1042→      renderBar();
  1043→      let waited = false;
  1044→      try {
  1045→        waited = await waitScrollWindow();
  1046→      } finally {
  1047→        A.held = false;
  1048→      }
  1049→      renderBar();
  1050→      if (!waited) {
  1051→        // Stopped by the user while parked - do not run the command.
  1052→        return "ERROR: the command was not run - stopped by the user.";
  1053→      }
  1054→      diag("tool.resumeAfterScroll", { name });
  1055→    }
  1056→    // ── Fast liveness probe ────────────────────────────────────────────────
  1057→    // A dead bridge used to cost the full tool budget before the loop learned it
  1058→    // was offline, so the chat felt far slower than on other providers for the
  1059→    // SAME work. A local bridge answers a ping in single-digit milliseconds when
  1060→    // the socket is WARM - but a COLD/reconnecting socket was measured at 4314ms,
  1061→    // so the budget must be generous (a false "offline" is far worse than a slow
  1062→    // probe) and one retry is allowed before giving up. The probe is paid only
  1063→    // when the cache is cold (>15s), so normal runs cost nothing.
  1064→    if (!A._bridgeProbeOkAt || Date.now() - A._bridgeProbeOkAt > 15000) {
  1065→      const probeOnce = (budget) => Promise.race([
  1066→        bg({ type: "ping" }),
  1067→        new Promise((res) => setTimeout(() => res({ ok: false, kind: "timeout" }), budget)),
  1068→      ]);
  1069→      let probe = await probeOnce(6000);
  1070→      let probeOk = probe && (probe.ok !== false) && probe.kind !== "disconnected" && probe.kind !== "timeout";
  1071→      if (!probeOk) {
  1072→        // One more chance: a socket that was mid-reconnect on the first attempt
  1073→        // is usually settled a couple of seconds later.
  1074→        probe = await probeOnce(8000);
  1075→        probeOk = probe && (probe.ok !== false) && probe.kind !== "disconnected" && probe.kind !== "timeout";
  1076→      }
  1077→      if (!probeOk) {
  1078→        diag("bridge.preflightFail", { name, kind: (probe && probe.kind) || "none" });
  1079→        A._bridgeProbeOkAt = 0;
  1080→        ui.banner("warn", "Bridge offline",
  1081→          "snapgentV3 bridge is unreachable - start it and keep the selected target app open.");
  1082→        return ZS.FEEDBACK.bridgeOffline;
  1083→      }
  1084→      A._bridgeProbeOkAt = Date.now();

  1087→    // choke point for ALL execution (agentLoop's tool dispatch AND the bootstrap's

  1202→            compact.push(`${k}${mark}:${ty}${v.description ? ` "${v.description}"` : ""}`);

  1254→    });

  1256→    clearInterval(stopTimer);

  1258→    if (!r) return ZS.FEEDBACK.bridgeOffline;

  1261→    // text instead of a result. Surface it as a proper environment ERROR so the

  1264→      ui.banner("warn", "Studio offline",

  1270→    // Re-shape those into a real ERROR so the model corrects the call instead
  1271→    // of misreading it as tool output.
  1272→    if (r.ok && r.text && /^[\w .'"-]{0,60}\bis (required|not available|invalid)\b[\w .'"-]{0,80}$/i.test(r.text.trim())) {

  1328→      // short settle before declaring the bridge offline - a false "offline"
  1329→      // makes the model stop working and shows "tool failed" to the user.
  1330→      //
  1331→      // BUT only when the request provably never left. The rule lives in
  1332→      // ZSGuard so the tested logic IS the running logic: `sent === false` means

  1336→      // write that did land is worse than reporting the ambiguity.
  1337→      if (!ZSGuard.mayRetryAfterDisconnect(r)) {

  1341→        return ZSGuard.inFlightFailure(name);

  1347→        hardCap,
  1348→      ]);
  1349→      // The retry itself has to be judged by the same rule. A second attempt
  1350→      // that goes out and then drops must NOT be reported as "bridge offline":
  1351→      // the tool may have run, and the model would retry it again.
  1352→      if (retry && retry.kind === "disconnected" && !ZSGuard.mayRetryAfterDisconnect(retry)) {

  1355→          `The bridge stopped answering while '${name}' was in flight. It may already have run - check the result before repeating it.`);
  1356→        return ZSGuard.inFlightFailure(name);
  1357→      }

  1359→        r = retry;
  1360→      } else {
  1361→        // Record WHY we are declaring the bridge offline. Every path (socket closed,
  1362→        // never opened, worker gave no answer) ends in the same message otherwise,
  1363→        // so an "unreachable while the bridge is open" report could not be traced.
  1364→        diag("bridge.offlineVerdict", {
  1365→          name,
  1366→          firstError: r && r.error,
  1367→          firstSent: r && r.sent,
  1368→          retryKind: retry ? retry.kind : null,
  1369→          retryError: retry ? retry.error : "no response",
  1370→          retrySent: retry ? retry.sent : null,
  1371→        });
  1372→        return ZS.FEEDBACK.bridgeOffline;
  1373→      }
  1374→    }
  1375→    if (r.kind === "timeout") {
  1376→      return `ERROR: tool '${name}' timed out after ${name === "execute_luau" ? 20 : 120}s.\n${r.error}\nTry a shorter/simpler call or check that the selected target app is open and responsive.`;
  1377→    }
  1378→    if (name === "execute_luau") {
  1379→      const err = r.error || "";
  1380→      // "Failed to parse command code" is StudioMCP's GENERIC parse rejection: an
  1381→      // empty/mis-marked block is only ONE of its causes. The others are ordinary
  1382→      // Luau syntax errors and - seen live on Meta AI 2026-08-13 - code that is
  1383→      // syntactically fine but too big for the parser: a `return 1+1+1+…` chain
  1384→      // ran at ~500 terms (1006 chars) and was rejected at ~1000 (2006 chars).
  1385→      // Telling the model its block was empty when we DID send it a full code
  1386→      // string sends it to fix something that isn't broken; it retries the same
  1387→      // payload and fails again (the reported spam of parse errors). Only give the
  1388→      // marker advice when the code we actually sent really was empty.
  1389→      const luaCode = args.code || "";
  1390→      const hint = err.includes("Failed to parse command code")
  1391→        ? !luaCode.trim()
  1392→          ? "Your code block was empty or the marker was wrong. Use exactly ###LUA### (three hashes) - never ###LUA---. The code must be between ###LUA### and ###END_LUA###."
  1393→          : `Roblox refused to PARSE the code (${luaCode.length} chars sent, so it was not empty). Check the syntax first; if it looks correct, split the work into several smaller calls.`
  1394→        : err.includes("attempt to") || err.includes("nil value")
  1395→          ? "Lua runtime error. Check that the API you are calling exists (use game:GetService() to access services). Make sure you use 'return' to output values, not 'print()'."
  1396→          : "Check your Lua syntax, make sure you use 'return' to output values (not 'print()'), and that all APIs you call exist in the current Roblox Studio context.";
  1397→      return `ERROR in execute_luau: ${err}\n\n${hint}\n\nFix the code and retry.`;
  1398→    }
  1399→    return `ERROR calling '${name}': ${r.error}\nRead the error carefully, fix the call or try a different approach.`;
  1400→  }

  1402→  function argSummary(call) {
  1403→    if (!call) return "";
  1404→    if (call.tool === "execute_luau") {
  1405→      const code = (call.arguments && call.arguments.code) || "";
  1406→      const first = code.split("\n").map((s) => s.trim()).filter(Boolean)[0] || "";
  1407→      return first.slice(0, 46);
  1408→    }
  1409→    const a = call.arguments || {};
  1410→    const k = Object.keys(a)[0];
  1411→    if (!k) return "";
  1412→    let v = String(a[k]);
  1413→    if (v.length > 34) v = v.slice(0, 31) + "…";
  1414→    return `${k}: ${v}`;
  1415→  }
  1416→
  1417→  // An MCP tool can report its OWN failure as a NORMAL result ("Output of '…':
  1418→  // Error executing code: …") instead of our ERROR wrapper - so a
  1419→  // startsWith("ERROR") test alone paints a FAILED call ✓ green and shows the
  1420→  // error as its summary (seen live on Blender's execute_blender_code, and it
  1421→  // will hit EVERY future MCP server the same way). Treat a result whose FIRST
  1422→  // line opens with an error lead-in as failed too. Deliberately PHRASE-based,
  1423→  // not the bare words "error"/"failed", so a genuine success line like
  1424→  // "Failed: 0" / "Error count: 0" is NOT misread as a failure.
  1425→  const BODY_ERR_RE =
  1426→    /^\s*(error executing|error:|erreur|exception|traceback|communication error|failed to|could ?not|cannot |unable to|fatal)\b/i;
  1427→  const stripOutputPrefix = (feedback) => feedback.replace(/^Output of '[^']*':\n?/, "");
  1428→  function bodyLooksFailed(feedback) {
  1429→    if (!feedback || feedback.startsWith("ERROR")) return false; // wrapper already flags it
  1430→    const first = stripOutputPrefix(feedback).split("\n").map((s) => s.trim()).find(Boolean) || "";
  1431→    return BODY_ERR_RE.test(first);
  1432→  }
  1433→  // True failure = OUR wrapper prefix OR an MCP tool's in-body error lead-in.
  1434→  const feedbackIsError = (feedback) => feedback.startsWith("ERROR") || bodyLooksFailed(feedback);
  1435→
  1436→  // Some tools answer with raw JSON on one line, and a 44-char slice of it makes
  1437→  // a chip that says nothing: list_roblox_studios read
  1438→  // `{"studios":[{"id":"8521cfad-f8d9-46f4-8cbe-2`. Summarise the SHAPE instead,
  1439→  // in the same spirit as the boot chip's "25 commands". Returns null when the
  1440→  // body isn't JSON, so plain-text outputs keep their first line unchanged.
  1441→  function jsonSummary(body) {
  1442→    if (!/^[[{]/.test(body) || body.length > 200000) return null;
  1443→    let v;
  1444→    try { v = JSON.parse(body); } catch { return null; }
  1445→    if (Array.isArray(v)) return `${v.length} item${v.length === 1 ? "" : "s"}`;
  1446→    if (!v || typeof v !== "object") return null;
  1447→    const keys = Object.keys(v);
  1448→    // The common MCP shape: one wrapper key holding the list. Its name is already
  1449→    // plural ("studios", "scripts"), so it reads correctly as-is - just drop the
  1450→    // trailing "s" when there is exactly one, so it says "1 studio".
  1451→    if (keys.length === 1 && Array.isArray(v[keys[0]])) {
  1452→      const n = v[keys[0]].length;
  1453→      const word = n === 1 ? keys[0].replace(/s$/, "") : keys[0];
  1454→      return `${n} ${word}`;
  1455→    }
  1456→    // Otherwise show the scalar fields - that's what a human would read off.
  1457→    const scalars = keys
  1458→      .filter((k) => v[k] === null || typeof v[k] !== "object")
  1459→      .map((k) => `${k}: ${v[k]}`);
  1460→    if (scalars.length) return scalars.join(", ").slice(0, 44);
  1461→    return `${keys.length} field${keys.length === 1 ? "" : "s"}`;
  1462→  }
  1463→
  1464→  function outSummary(feedback) {
  1465→    if (!feedback) return "";
  1466→    const isErr = feedbackIsError(feedback);
  1467→    const body = stripOutputPrefix(feedback).trim();
  1468→    if (!body) return "";
  1469→    // Errors are prose and read fine as-is; only reshape successful JSON.
  1470→    if (!isErr) { const j = jsonSummary(body); if (j) return j; }
  1471→    const all = body.split("\n").map((l) => l.trim()).filter(Boolean);
  1472→    const lines = all.length;
  1473→    // On SUCCESS, skip a leading non-fatal warning/note some MCP tools print
  1474→    // before the real status so the chip shows the useful line, not the noise.
  1475→    let first = all[0] || "";
  1476→    if (!isErr && lines > 1 && /^(warning|warn|note|deprecat|info)\b/i.test(first)) {
  1477→      first = all.find((l) => !/^(warning|warn|note|deprecat|info)\b/i.test(l)) || first;
  1478→    }
  1479→    first = first.slice(0, 44);
  1480→    if (isErr) return first;
  1481→    return lines > 1 ? `${first} · ${lines} lines` : first;
  1482→  }

  1486→    const a = call.arguments || {};

  1494→  async function agentLoop(base) {
  1495→    if (A.running) return;
  1496→    A.running = true;
  1497→    A.resumeArmed = false; // loop now owns the turn; drop the regenerate grace
  1498→    A.stop = false;
  1499→    A.stopping = false; // clean slate: never inherit a stale "Stopping…" from a
  1500→                        // Stop click that landed before this loop actually started
  1501→    A.loopKey = null; // pinned by syncSessionState once this chat has an id + content
  1502→    let truncCount = 0;
  1503→    const MAX_TRUNC = 6;
  1504→    // Re-send the command list after this many successful tool calls. Kept high
  1505→    // so the reminder does not bloat the context too often.
  1506→    const REMIND_TOOLS_EVERY = 20;
  1507→    ui.showStop(true);
  1508→    P.setInputLock(true); // prevent user from typing while the agent is active
  1509→    ui.inputCover(true);  // keep the "Agent is working" cover up for the WHOLE loop
  1510→    diag("loop.start", { base });
  1511→    // Watchdog: if the tab is visible, no tool is running and NOTHING has changed
  1512→    // (turn counts, stream length, generating flag) for WD_STALL_MS, the loop is
  1513→    // stuck waiting. Stop it so the cover and Stop button are released by the
  1514→    // finally block below, and tell the user instead of leaving a dead bar.
  1515→    const WD_STALL_MS = 180000;
  1516→    let _wdSig = "", _wdAt = Date.now();
  1517→    const _wd = setInterval(() => {
  1518→      if (!A.running) return;
  1519→      if (document.hidden || A.parked || A.toolRunning || A.injecting) { _wdAt = Date.now(); return; }
  1520→      const sig = [P.assistantCount(), P.userCount(), P.streamLen ? P.streamLen() : 0, P.isGenerating() ? 1 : 0].join("|");
  1521→      if (sig !== _wdSig) { _wdSig = sig; _wdAt = Date.now(); return; }
  1522→      if (Date.now() - _wdAt > WD_STALL_MS) {
  1523→        diag("loop.watchdog", { sig });
  1524→        ui.banner("warn", "Agent stopped waiting", "No progress for 3 min. Send continue to resume.");
  1525→        A.stop = true;
  1526→        _wdAt = Date.now();
  1527→      }
  1528→    }, 5000);
  1529→    try {
  1530→      while (!A.stop) {
  1531→        // Gate the WHOLE cycle on tab visibility. We only advance - read the
  1532→        // reply, PARSE it, EXECUTE a tool, inject the result - while the AI tab
  1533→        // is the FOREGROUND tab of its Edge window. document.visibilityState
  1534→        // (mirrored by document.hidden) is the right signal, NOT window focus:
  1535→        //  - Edge loses OS focus but the AI tab stays the active tab (user is
  1536→        //    working in Roblox Studio) -> still "visible" -> the agent keeps
  1537→        //    running, exactly as wanted.
  1538→        //  - The AI tab is backgrounded (another tab in front) or the window is
  1539→        //    minimized -> "hidden" -> pause here. Background tabs throttle
  1540→        //    rendering/timers, which made DOM reads unreliable (misparse,
  1541→        //    duplicate sends - see the send-side guard in submitAndGetBase).
  1542→        // Parking here means we never START a parse/exec cycle off-screen; the
  1543→        // send step re-checks too, so a switch-away mid-generation is covered.
  1544→        if (document.hidden && !A.stop) {
  1545→          diag("loop.waitVisible");
  1546→          ui.inputCover(true); // keep the "Agent is working" cover up while parked
  1547→          if (!(await waitVisible()) || A.stop) break; // park (no cap) until foreground; break only on user stop
  1548→          diag("loop.visibleAgain");
  1549→        }
  1550→        const res = await waitForResponse(base);
  1551→        diag("response", { kind: res.kind });
  1552→        if (A.stop || res.kind === "stopped") break;
  1553→
  1554→        if (res.kind === "context_limit") {
  1555→          ui.banner("limit", `${P.displayName} reached its context limit`,
  1556→            (res.detail || "") + " New chat.");
  1557→          break;
  1558→        }
  1559→        if (res.kind === "too_long") {
  1560→          ui.banner("limit", "Conversation too long",
  1561→            `Too long. Start a new chat.`);
  1562→          break;
  1563→        }
  1564→        if (res.kind === "timeout") {
  1565→          ui.banner("warn", `No response from ${P.displayName}`,
  1566→            `Timed out. Agent stopped.`);

  1817→  // virtualized list (DeepSeek/Qwen/GLM/Arena) scrolling up renders a different

  1829→  function rememberHalted(item) {
  1830→    try {
  1831→      if (!item || assistantIdx(item) < 0) return;
  1832→      const pref = (P.itemText(item) || "").slice(0, 60);
  1833→      // A stop during the REASONING phase leaves the answer text EMPTY - an
  1834→      // empty/short prefix would then startsWith-match ANY later turn at this
  1835→      // index (seen live: a fresh streaming command went red "stopped" on the
  1836→      // spot). Too little text to identify → rely on the dataset marker only.
  1837→      if (pref.trim().length < 12) return;
  1838→      halted.set(`${P.conversationKey()}|${turnKey(item)}`, pref);
  1839→    } catch {}
  1840→  }

  1842→    if (!item || !halted.size) return;

  1844→  }

  1851→      return pref != null && (txt || "").startsWith(pref);

  1900→  function installScrollDetector() {
  1901→    try {
  1902→      // Position must be tracked PER SCROLLABLE ELEMENT. The listener is on
  1903→      // `document` in the capture phase, so it sees the transcript, the sidebar,
  1904→      // code blocks and textareas alike. A single shared _lastScrollTop compared
  1905→      // a value from one element against a value from another: scrolling the
  1906→      // sidebar down to 900 and then the transcript (at 400) looked like an
  1907→      // upward move of 500px and suppressed execution for no reason. A WeakMap
  1908→      // keeps one last position per element and lets them be collected.
  1909→      const tracker = ZSGuard.createScrollTracker(24);
  1910→      // Is this target part of the CONVERSATION transcript? Both listeners must
  1911→      // answer the same question, otherwise the scroll and wheel paths disagree
  1912→      // about what counts as "scrolling history".
  1913→      //
  1914→      // `main` is deliberately NOT used as a fallback: it matches nearly the
  1915→      // whole page, which would let a sidebar or composer gesture through. The
  1916→      // provider's message selector is the real signal; a plain article is the
  1917→      // only generic fallback, and a composer/editor is always excluded.
  1918→      //
  1919→      // The two listeners get DIFFERENT kinds of target, so the test has to look
  1920→      // in both directions:
  1921→      //   wheel  -> the element under the cursor, a DESCENDANT of the transcript,
  1922→      //             so `closest()` walking up finds it;
  1923→      //   scroll -> the element that scrolled, normally the transcript CONTAINER
  1924→      //             (an ancestor of every message), so `closest()` finds nothing
  1925→      //             and the event has to be accepted when the target CONTAINS a
  1926→      //             message instead. Window-level scrolling reports `document`,
  1927→      //             which has no `closest` at all and is the provider's scroller
  1928→      //             for some layouts, so it is accepted as the transcript too.
  1929→      //
  1930→      // A filter that only walked up silently dropped scrollbar, keyboard and
  1931→      // touch scrolling while the wheel path kept working - a regression on the
  1932→      // exact gesture the suppression exists for.
  1933→      const inTranscript = (t) => {
  1934→        if (!t) return false;
  1935→        // The window/document scroller is the transcript for window-level
  1936→        // layouts; a composer or editor is never the transcript.
  1937→        if (t === document || t === document.documentElement || t === window) return true;
  1938→        if (!t.closest) return false;
  1939→        if (t.closest("textarea, input, .cm-editor, [contenteditable='true']")) return false;
  1940→        const msgSel = P.msgSel || "[data-message-author-role]";
  1941→        // Upward: the target sits inside a message (wheel, inner scrollables).
  1942→        if (t.closest(msgSel) || t.closest("article")) return true;
  1943→        // Downward: the target is the scroll container that holds messages.
  1944→        return !!(typeof t.querySelector === "function" && (t.querySelector(msgSel) || t.querySelector("article")));
  1945→      };
  1946→      const topOf = (t) => {
  1947→        if (t && typeof t.scrollTop === "number" && t !== document && t !== document.documentElement) {
  1948→          return { key: t, top: t.scrollTop };
  1949→        }
  1950→        return { key: document.documentElement, top: window.scrollY || document.documentElement.scrollTop || 0 };
  1951→      };
  1952→      const onScroll = (ev) => {
  1953→        const target = ev && ev.target;
  1954→        // The same transcript-only rule as the wheel listener below.
  1955→        if (!inTranscript(target)) return;
  1956→        const { key, top } = topOf(target);
  1957→        // Moving UP through a turn is the problematic gesture. The 24px
  1958→        // threshold ignores layout jitter and stream-driven auto-scroll (which
  1959→        // only ever moves DOWN as the reply grows).
  1960→        if (tracker.observe(key, top)) noteScrollUp();
  1961→      };
  1962→      // capture:true - ChatGPT scrolls inner containers, not just the window.
  1963→      document.addEventListener("scroll", onScroll, { capture: true, passive: true });
  1964→      // WHEEL: only a gesture over the transcript counts. Scrolling up inside the
  1965→      // sidebar, a code block or the composer is unrelated to replaying history,
  1966→      // and used to hold execution as if the user had scrolled the conversation.
  1967→      window.addEventListener("wheel", (ev) => {
  1968→        if (!(ev && ev.deltaY < 0)) return;
  1969→        if (!inTranscript(ev.target)) return;
  1970→        noteScrollUp();
  1971→      }, { passive: true });
  1972→    } catch {}
  1973→  }
