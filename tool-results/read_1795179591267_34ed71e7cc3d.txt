     3→// Exports the same ZSProvider interface as providers/deepseek.js and gemini.js;
     4→// the core (core/main.js) is provider-agnostic. To DISABLE GLM support, remove

     6→//
     7→// Z.ai (GLM) DOM notes (validated live, 2026-06):
     8→//  - Svelte app. One exchange = a `.user-message` div followed by a
     9→//    `.chat-assistant` div (the assistant reply root, also `.markdown-prose`).
    10→//    The assistant root is wrapped by `div.message-<uuid>`; that uuid is a
    11→//    stable per-turn identity we expose as lastAssistantId (virtualization-proof).
    12→//  - Reasoning ("Thought Process") renders as a `.thinking-chain-container`
    13→//    INSIDE `.chat-assistant`, so all reply-text extraction strips that subtree
    14→//    (tool blocks drafted inside reasoning are never detected or executed).
    15→//  - The composer is a real <textarea id="chat-input">: set .value via the native
    16→//    prototype setter + input event (React/Svelte both read it), WAIT for the
    17→//    send button to re-enable, then click it. Typing + clicking in the same tick
    18→//    fails because `#send-message-button` is still `disabled` for a frame.
    19→//  - Idle: `button#send-message-button` is present (disabled when the box is
    20→//    empty). While generating it is REPLACED by a round black stop button
    21→//    (`button.rounded-full.bg-black` holding a small square <span>), so
    22→//    "send button gone / stop button present" == generating for the WHOLE
    23→//    stream (thinking included) - no indicatorless reasoning phase like DeepSeek.
    24→//  - Fenced code = an atomic wrapper `div[class*="rounded-xl"]` carrying a
    25→//    `.copy-code-button`; the command JSON survives in textContent.
    26→//  - Conversation URL is /c/<uuid>; a fresh chat is exactly "/".

   107→      for (const c of n.childNodes) walk(c);
   108→    };
   109→    walk(root);

   252→  let _streamMax = -1, _streamAt = 0, _streamItem = null;
   253→  function sampleStream() {
   254→    const item = lastAssistant();

   268→  // measured text length PLATEAUS mid-stream. The growth guard then wrongly

   311→  async function waitFor(pred, timeout) {
   312→    const t0 = Date.now();
   313→    while (Date.now() - t0 < timeout) {
   314→      if (pred()) return true;
   315→      await sleep(120);
   316→    }
   317→    return false;
   318→  }

   320→  // ── Sending ─────────────────────────────────────────────────────────────────
   321→  // Real <textarea> driven by Svelte: set .value via the native prototype setter
   322→  // so the framework's input handler fires, dispatch an input event, then click
   323→  // the send button ONCE it re-enables (a fresh value disables it for a frame).
   324→  function setTextareaValue(el, v) {
   325→    const proto = window.HTMLTextAreaElement && window.HTMLTextAreaElement.prototype;
   326→    const setter = proto && Object.getOwnPropertyDescriptor(proto, "value");
   327→    if (setter && setter.set) setter.set.call(el, v);
   328→    else el.value = v;
   329→    el.dispatchEvent(new Event("input", { bubbles: true }));
   330→  }

   339→  async function typeAndSend(text, images) {
   340→    const editor = getEditor();
   341→    if (!editor) throw new Error("GLM input box not found");
   342→    editor.focus();
   343→    setTextareaValue(editor, text);
   344→    // Attach images LAST, right before the send click - see gemini.js's
   345→    // typeAndSend for why (attaching before retyping the text can sever the
   346→    // site's binding between the pending upload and the message being sent).
   347→    // submitAndGetBase RETRIES this function; only attach if nothing is staged
   348→    // yet, else each retry uploads ANOTHER duplicate copy.
   349→    if (images && images.length && !hasPendingAttachment()) { try { await attachImages(images); } catch {} }
   350→    // Wait for Svelte to re-enable the send button (proof it registered the text),
   351→    // then click the instant it's clickable. In a LONG conversation the message
   352→    // list is heavy, so Svelte's reactivity to the input event can take well over
   353→    // a second - the old 1500ms cap expired with the button still `disabled`, the
   354→    // click no-opped, and only the core's retry loop eventually landed it (the
   355→    // user's "message sits in the input then sends after a long time"). We now wait
   356→    // up to 8s AND re-dispatch the input event every ~700ms so a slow/queued Svelte
   357→    // cycle keeps getting nudged until it re-enables the button.
   358→    let lastNudge = Date.now();
   359→    const enabled = await waitFor(() => {
   360→      const b = sendButton();
   361→      if (b && !b.disabled) return true;
   362→      if (Date.now() - lastNudge > 700) {
   363→        lastNudge = Date.now();
   364→        // Re-assert the value (a heavy re-render can drop it) and re-fire input.
   365→        if (editorText() !== text) setTextareaValue(editor, text);
   366→        else editor.dispatchEvent(new Event("input", { bubbles: true }));
   367→      }
   368→      return false;
   369→    }, 8000);
   370→    diag("glm.send", { enabled, busy: isBusyNow() });
   371→    if (!clickSendButton() && !isBusyNow()) {
   372→      const o = { key: "Enter", code: "Enter", keyCode: 13, which: 13, bubbles: true, cancelable: true };
   373→      editor.dispatchEvent(new KeyboardEvent("keydown", o));
   374→      editor.dispatchEvent(new KeyboardEvent("keyup", o));
   375→    }
   376→  }

   406→  // ── Image attachment (best effort) ────────────────────────────────────────
   407→  function fileFromImage(img, i) {
   408→    const mime = img.mimeType || "image/jpeg";
   409→    const bin = atob(img.data);
   410→    const arr = new Uint8Array(bin.length);
   411→    for (let j = 0; j < bin.length; j++) arr[j] = bin.charCodeAt(j);
   412→    const ext = mime.includes("png") ? "png" : "jpg";
   413→    return new File([arr], `snapgentV3_${Date.now()}_${i}.${ext}`, { type: mime });
   414→  }
   415→  // The PENDING attachment chip. z.ai mounts each staged upload as a
   416→  // `.chip-scroll > button` inside the composer card (barAnchor). A SENT image
   417→  // renders separately in its `.user-message` turn, so scoping the probe to the
   418→  // composer card excludes history - which stops a leftover chip from a PREVIOUS
   419→  // capture reading as "already pending" and skipping the next attach (the
   420→  // 2nd-capture-never-attaches bug seen on Kimi). Validated live on chat.z.ai.
   421→  const pendingChip = () => {
   422→    const host = barAnchor() || document;
   423→    return host.querySelector(".chip-scroll > button");
   424→  };
   425→  const hasPendingAttachment = () => !!pendingChip();
   426→  // Upload done = the chip's <img data-cy="image"> src has flipped from its local
   427→  // placeholder to the remote CDN url (https://z-cdn-media.chatglm.cn/…). Sending
   428→  // before that goes out text-only, so we WAIT for the http(s) src.
   429→  const attachmentReady = () => {
   430→    const host = barAnchor() || document;
   431→    const im = host.querySelector('.chip-scroll img[data-cy="image"]') || host.querySelector(".chip-scroll img");
   432→    return !!(im && /^https?:\/\//.test(im.getAttribute("src") || ""));
   433→  };
   434→  async function attachImages(images) {
   435→    if (!images || !images.length) return false;
   436→    const dt = new DataTransfer();
   437→    images.forEach((img, i) => { try { dt.items.add(fileFromImage(img, i)); } catch {} });
   438→    if (!dt.items.length) return false;
   439→    // z.ai keeps a single always-mounted <input type=file> in the composer
   440→    // (accepts png/jpg among others); setting .files + change stages the upload -
   441→    // no "+" menu to open (unlike Kimi). A synthetic paste is NOT needed.
   442→    const fileInput = document.querySelector('input[type="file"]');
   443→    if (!fileInput) { diag("attach.noFileInput"); return false; }
   444→    diag("attach.setFiles", { count: dt.items.length });

   470→  function installSendHooks(handlers) {
   471→    document.addEventListener(
   472→      "keydown",
   473→      (e) => {
   474→        if (e.key !== "Enter" || e.shiftKey || e.isComposing) return;
   475→        const ed = getEditor();
   476→        if (!ed || !ed.contains(e.target)) return;
   477→        if (editorText().trim() === "") return;
   478→        if (handlers.isBlocked()) return;
   479→        if (!handlers.isStarted()) {
   480→          if (!chatIsEmpty()) return; // existing conversation → not ours to gate
   481→          handlers.onBlockedAttempt(); // nudge only; never block plain chat
   482→          return;
   483→        }
   484→        handlers.onUserMessage(assistantCount());
   485→      },
   486→      true
   487→    );
   488→
   489→    document.addEventListener(
   490→      "click",
   491→      (e) => {
   492→        if (!getEditor()) return;
   493→        const t = e.target;
   494→        // Stop button (round, in the composer) = a native stop intent.
   495→        const stop = t && t.closest && t.closest("button");
   496→        if (stop && composerFrame() && composerFrame().contains(stop) &&
   497→            stop.id !== "send-message-button" &&
   498→            /rounded-full/.test(stop.className) && /bg-black|bg-white/.test(stop.className) &&
   499→            !stop.querySelector("svg")) { // exclude the scroll-to-bottom arrows (see stopButton)
   500→          handlers.onNativeStop();
   501→          return;
   502→        }
   503→        const btn = t && t.closest && t.closest(S.sendBtn);
   504→        if (!btn) return;
   505→        if (btn.disabled) return;
   506→        if (handlers.isBlocked()) return;
   507→        if (!handlers.isStarted()) {
   508→          if (!chatIsEmpty()) return;
   509→          handlers.onBlockedAttempt(); // nudge only; never block plain chat
   510→          return;
   511→        }
   512→        handlers.onUserMessage(assistantCount());
   513→      },
   514→      true
   515→    );
   516→  }

   518→  // ── Tool-block location for camouflage ────────────────────────────────────
   519→  // GLM wraps each fenced code block in an atomic `div[class*="rounded-xl"]`
   520→  // carrying a `.copy-code-button` (markers + JSON survive in textContent), and
   521→  // Svelte re-renders the markdown subtree on stream updates / next send. So,
   522→  // like Gemini: hide every code wrapper in the reply whose text carries a
   523→  // command shape AND mark the reply root (`.chat-assistant`, which keeps its
   524→  // identity) with .zs-cmd-mask so a CSS rule re-hides recreated wrappers with
   525→  // zero flash. Also catch a stray bare inline command paragraph.
   526→  const CMD_SHAPE = /"(?:command|tool)"\s*:\s*"|###\s*lua|###mcp_tool###/i;
   527→  function findToolBlockSpot(item /*, chip */) {
   528→    if (!item) return null;
   529→    let hidAny = null;
   530→    // 1. Fenced code blocks carrying a command (skip ones inside the reasoning).
   531→    for (const cc of item.querySelectorAll(S.copyCodeBtn)) {
   532→      if (cc.closest(S.thinking)) continue;
   533→      const wrap = cc.closest(S.codeWrap);
   534→      if (!wrap || wrap.closest(".zs-chip")) continue;
   535→      if (CMD_SHAPE.test(wrap.textContent || "")) {
   536→        wrap.classList.add("zs-tool-hide");
   537→        item.classList.add("zs-cmd-mask");
   538→        hidAny = hidAny || { parent: wrap.parentElement, ref: wrap };
   539→      }
   540→    }
   541→    // 2. Bare top-level blocks with an inline command (no code wrapper inside).
   542→    const reply = item.querySelector(".markdown-prose") || item;
   543→    [...reply.children].forEach((el) => {
   544→      if (el.classList.contains("zs-chip") || el.closest(S.thinking) ||
   545→          el.querySelector(S.copyCodeBtn)) return;
   546→      const t = el.textContent || "";
   547→      if (t.length < 600 && CMD_SHAPE.test(t)) {
   548→        el.classList.add("zs-tool-hide");
   549→        hidAny = hidAny || { parent: el.parentElement, ref: el };
   550→      }
   551→    });
   552→    return hidAny;
   553→  }
