   148→void Mob::baseTick()
   149→{
   150→        oAttackAnim = attackAnim;
   151→        super::baseTick();
   152→
   153→        TIMER_PUSH("mobBaseTick");
   154→
   155→        if (((ambientSoundTime++ & 15) == 0) && random.nextInt(2000) < ambientSoundTime) {
   156→            ambientSoundTime = -getAmbientSoundInterval();
   157→            playAmbientSound();
   158→        }
   159→
   160→        if (isAlive() && isInWall()) {
   161→                hurt(NULL, 1);
   162→        }
   163→
   164→        //if (fireImmune || level.isOnline) onFire = 0;
   165→
   166→        if (isAlive() && isUnderLiquid(Material::water) && !isWaterMob()) {
   167→                airSupply--;
   168→                if (airSupply == -20) {
   169→                        airSupply = 0;
   170→                        for (int i = 0; i < 8; i++) {
   171→                                float xo = random.nextFloat() - random.nextFloat();
   172→                                float yo = random.nextFloat() - random.nextFloat();
   173→                                float zo = random.nextFloat() - random.nextFloat();
   174→                                level->addParticle(PARTICLETYPE(bubble), x + xo, y + yo, z + zo, xd, yd, zd);
   175→                        }
   176→                        hurt(NULL, 2);
   177→                }
   178→                //onFire = 0;
   179→        } else {
   180→                airSupply = airCapacity;
   181→        }
   182→
   183→        oTilt = tilt;
   184→        if (attackTime > 0) attackTime--;
   185→        if (hurtTime > 0) hurtTime--;
   186→        if (invulnerableTime > 0) invulnerableTime--;
   187→
   188→        if (health <= 0) {
   189→            deathTime++;
   190→            if (deathTime > SharedConstants::TicksPerSecond) {
   191→                beforeRemove();
   192→                remove();
   193→                for (int i = 0; i < 20; i++) {
   194→                    float xa = (2.0f * random.nextFloat() - 1.0f) * (2.0f * random.nextFloat() - 1.0f) * 0.02f;
   195→                                float ya = (2.0f * random.nextFloat() - 1.0f) * (2.0f * random.nextFloat() - 1.0f) * 0.02f;
   196→                                float za = (2.0f * random.nextFloat() - 1.0f) * (2.0f * random.nextFloat() - 1.0f) * 0.02f;
   197→                    level->addParticle(PARTICLETYPE(explode), x + random.nextFloat() * bbWidth * 2 - bbWidth, y + random.nextFloat() * bbHeight, z + random.nextFloat() * bbWidth * 2 - bbWidth, xa, ya, za);
   198→                }
   199→            }
   200→        }
   201→
   202→        animStepO = animStep;
   203→
   204→        yBodyRotO = yBodyRot;
   205→        yRotO = yRot;
   206→        xRotO = xRot;
   207→
   208→        TIMER_POP();
   209→
   210→        if (!level->isClientSide) {
   211→                if (autoSendPosRot) {
   212→                        if (autoSendPosRot && (std::abs(x - sentX) > .1f || std::abs(y - sentY) > .05f || std::abs(z - sentZ) > .1f || std::abs(sentRotX - xRot) > 1 || std::abs(sentRotY - yRot) > 1)) {
   213→                                MoveEntityPacket_PosRot packet(this);
   214→                                level->raknetInstance->send(packet);
   215→                                sentX = x;
   216→                                sentY = y;
   217→                                sentZ = z;
   218→                                sentRotX = xRot;
   219→                                sentRotY = yRot;
   220→                        }
   221→                }
   222→
   223→                float ddx = std::abs(xd - sentXd);
   224→                float ddy = std::abs(yd - sentYd);
   225→                float ddz = std::abs(zd - sentZd);
   226→                const float max = 0.02f;
   227→                const float diff = ddx*ddx + ddy*ddy + ddz*ddz;
   228→                if (diff > max*max || (diff > 0 && xd == 0 && yd == 0 && zd == 0)) {
   229→                        sentXd = xd;
   230→                        sentYd = yd;
   231→                        sentZd = zd;
   232→                        SetEntityMotionPacket packet(this);
   233→                        level->raknetInstance->send(packet);
   234→                        //LOGI("Motion-packet: %d\n", entityId);
   235→                }
   236→        }
   237→}

   270→void Mob::tick()
   271→{
   272→        super::tick();
   273→
   274→        if (arrowCount > 0) {
   275→                if (removeArrowTime <= 0) {
   276→                        removeArrowTime = 20 * 3;
   277→                }
   278→                removeArrowTime--;
   279→                if (removeArrowTime <= 0) {
   280→                        arrowCount--;
   281→                }
   282→        }
   283→
   284→        if (lSteps > 0) {
   285→                float xt = x + (lx - x) / lSteps;
   286→                float yt = y + (ly - y) / lSteps;
   287→                float zt = z + (lz - z) / lSteps;
   288→
   289→                float yrd = lyr - yRot;
   290→                while (yrd < -180)
   291→                        yrd += 360;
   292→                while (yrd >= 180)
   293→                        yrd -= 360;
   294→
   295→                yRot += (yrd) / lSteps;
   296→                xRot += (lxr - xRot) / lSteps;
   297→
   298→                lSteps--;
   299→                this->setPos(xt, yt, zt);
   300→                this->setRot(yRot, xRot);
   301→        }
   302→
   303→        aiStep();
   304→
   305→        updateWalkAnim();
   306→
   307→        float xd = x - xo;
   308→        float zd = z - zo;
   309→
   310→        float sideDist = (float) Mth::sqrt(xd * xd + zd * zd);
   311→
   312→        float yBodyRotT = yBodyRot;
   313→
   314→        float walkSpeed = 0;
   315→        oRun = run;
   316→        float tRun = 0;
   317→        if (sideDist <= 0.05f) {
   318→                // animStep = 0;
   319→        } else {
   320→                tRun = 1;
   321→                walkSpeed = sideDist * 3;
   322→                yBodyRotT = Mth::atan2(zd, xd) * Mth::RADDEG - 90;
   323→        }
   324→        if (attackAnim > 0) {
   325→                yBodyRotT = yRot;
   326→        }
   327→        if (!onGround) {
   328→                tRun = 0;
   329→        }
   330→        run = run + (tRun - run) * 0.3f;
   331→
   332→        /*
   333→        * float yBodyRotD = yRot-yBodyRot; while (yBodyRotD < -180) yBodyRotD
   334→        * += 360; while (yBodyRotD >= 180) yBodyRotD -= 360; yBodyRot +=
   335→        * yBodyRotD * 0.1f;
   336→        */
   337→
   338→        float yBodyRotD = yBodyRotT - yBodyRot;
   339→        while (yBodyRotD < -180)
   340→                yBodyRotD += 360;
   341→        while (yBodyRotD >= 180)
   342→                yBodyRotD -= 360;
   343→        yBodyRot += yBodyRotD * 0.3f;
   344→
   345→        float headDiff = yRot - yBodyRot;
   346→        while (headDiff < -180)
   347→                headDiff += 360;
   348→        while (headDiff >= 180)
   349→                headDiff -= 360;
   350→        bool behind = headDiff < -90 || headDiff >= 90;
   351→        if (headDiff < -75) headDiff = -75;
   352→        if (headDiff >= 75) headDiff = +75;
   353→        yBodyRot = yRot - headDiff;
   354→        if (headDiff * headDiff > 50 * 50) {
   355→                yBodyRot += headDiff * 0.2f;
   356→        }
   357→
   358→        if (behind) {
   359→                walkSpeed *= -1;
   360→        }
   361→        while (yRot - yRotO < -180)
   362→                yRotO -= 360;
   363→        while (yRot - yRotO >= 180)
   364→                yRotO += 360;
   365→
   366→        while (yBodyRot - yBodyRotO < -180)
   367→                yBodyRotO -= 360;
   368→        while (yBodyRot - yBodyRotO >= 180)
   369→                yBodyRotO += 360;
   370→
   371→        while (xRot - xRotO < -180)
   372→                xRotO -= 360;
   373→        while (xRot - xRotO >= 180)
   374→                xRotO += 360;
   375→        animStep += walkSpeed;
   376→}

   391→bool Mob::hurt( Entity* source, int dmg )
   392→{
   393→        if (level->isClientSide) return false;
   394→        noActionTime = 0;
   395→        if (health <= 0) return false;
   396→
   397→        this->walkAnimSpeed = 1.5f;
   398→
   399→        bool sound = true;
   400→        if (invulnerableTime > invulnerableDuration / 2.0f) {
   401→                if (dmg <= lastHurt) return false;
   402→                actuallyHurt(dmg - lastHurt);
   403→                lastHurt = dmg;
   404→                sound = false;
   405→        } else {
   406→                lastHurt = dmg;
   407→                lastHealth = health;
   408→                invulnerableTime = invulnerableDuration;
   409→                actuallyHurt(dmg);
   410→                hurtTime = hurtDuration = 10;
   411→                hurtDuration = 10;
   412→        }
   413→
   414→        hurtDir = 0;
   415→
   416→        if (sound) {
   417→                level->broadcastEntityEvent(this, EntityEvent::HURT);
   418→                markHurt();
   419→                if (source != NULL) {
   420→                        float xd = source->x - x;
   421→                        float zd = source->z - z;
   422→                        while (xd * xd + zd * zd < 0.0001) {
   423→                                xd = (Mth::random() - Mth::random()) * 0.01f;
   424→                                zd = (Mth::random() - Mth::random()) * 0.01f;
   425→                        }
   426→                        hurtDir = (float) (std::atan2(zd, xd) * Mth::RADDEG) - yRot;
   427→                        knockback(source, dmg, xd, zd);
   428→                } else {
   429→                        hurtDir = (int) (Mth::random() * 2.0f) * 180.0f;
   430→                }
   431→        }
   432→
   433→        if (health <= 0) {
   434→                if (sound) level->playSound(this, getDeathSound(), getSoundVolume(), getVoicePitch());
   435→                die(source);
   436→        } else {
   437→                if (sound) level->playSound(this, getHurtSound(), getSoundVolume(), getVoicePitch());
   438→        }
   439→
   440→        return true;
   441→}

   471→void Mob::knockback( Entity* source, int dmg, float xd, float zd )
   472→{
   473→        float dd = Mth::invSqrt(xd * xd + zd * zd);
   474→        float pow = 0.4f;
   475→
   476→        this->xd *= 0.5f;
   477→        this->yd *= 0.5f;
   478→        this->zd *= 0.5f;
   479→
   480→        this->xd -= xd * dd * pow;
   481→        this->yd += 0.4f;
   482→        this->zd -= zd * dd * pow;
   483→
   484→        if (this->yd > 0.4f) this->yd = 0.4f;
   485→}

   558→void Mob::travel( float xa, float ya )
   559→{
   560→        if (isInWater()) {
   561→                float yo = y;
   562→                moveRelative(xa, ya, 0.02f);
   563→                move(xd, yd, zd);
   564→
   565→                xd *= 0.80f;
   566→                yd *= 0.80f;
   567→                zd *= 0.80f;
   568→                yd -= 0.02f;
   569→
   570→                if (horizontalCollision && isFree(xd, yd + 0.6f - y + yo, zd)) {
   571→                        yd = 0.3f;
   572→                }
   573→        } else if (isInLava()) {
   574→                float yo = y;
   575→                moveRelative(xa, ya, 0.02f);
   576→                move(xd, yd, zd);
   577→                xd *= 0.50f;
   578→                yd *= 0.50f;
   579→                zd *= 0.50f;
   580→                yd -= 0.02f;
   581→
   582→                if (horizontalCollision && isFree(xd, yd + 0.6f - y + yo, zd)) {
   583→                        yd = 0.3f;
   584→                }
   585→        } else {
   586→                float friction = 0.91f;
   587→                if (onGround) {
   588→                        friction = 0.6f * 0.91f;
   589→                        int t = level->getTile(Mth::floor(x), Mth::floor(bb.y0 - 0.5f), Mth::floor(z));
   590→                        if (t > 0) {
   591→                                friction = Tile::tiles[t]->friction * 0.91f;
   592→                        }
   593→                }
   594→
   595→                float friction2 = (0.6f * 0.6f * 0.91f * 0.91f * 0.6f * 0.91f) / (friction * friction * friction);
   596→                moveRelative(xa, ya, (onGround ? walkingSpeed * friction2 : flyingSpeed));
   597→
   598→                friction = 0.91f;
   599→                if (onGround) {
   600→                        friction = 0.6f * 0.91f;
   601→                        int t = level->getTile(Mth::floor(x), Mth::floor(bb.y0 - 0.5f), Mth::floor(z));
   602→                        if (t > 0) {
   603→                                friction = Tile::tiles[t]->friction * 0.91f;
   604→                        }
   605→                }
   606→                //@todo: make it easier to climb ladders
   607→                if (onLadder()) {
   608→                        this->fallDistance = 0;
   609→                        if (yd < -0.15) yd = -0.15f;
   610→                        if (isSneaking() && yd < 0) yd = 0;
   611→                }
   612→
   613→                move(xd, yd, zd);
   614→
   615→                //@todo: make it easier to climb ladders
   616→                if (horizontalCollision && onLadder()) {
   617→                        yd = 0.2f;
   618→                }
   619→
   620→                yd -= 0.08f;
   621→                yd *= 0.98f;
   622→                xd *= friction;
   623→                zd *= friction;
   624→        }
   625→

   688→void Mob::aiStep()
   689→{
   690→        //@todo? 30 lines of code here in java version
   691→
   692→        TIMER_PUSH("ai");
   693→        if (isImmobile()) {
   694→                jumping = false;
   695→                xxa = 0;
   696→                yya = 0;
   697→                yRotA = 0;
   698→        } else {
   699→                if (!interpolateOnly()) {
   700→                        if (useNewAi()) {
   701→                                TIMER_PUSH("newAi");
   702→                                newServerAiStep();
   703→                                TIMER_POP();
   704→                        }
   705→                        else {
   706→                                TIMER_PUSH("oldAi");
   707→                                updateAi();
   708→                                TIMER_POP();
   709→                        }
   710→                }
   711→        }
   712→        TIMER_POP_PUSH("move");
   713→
   714→        bool inWater = isInWater();
   715→        bool inLava = isInLava();
   716→
   717→        if (jumping) {
   718→                if (inWater) {
   719→                        yd += 0.04f;
   720→                } else if (inLava) {
   721→                        yd += 0.04f;
   722→                } else if (onGround) {
   723→                        jumpFromGround();
   724→                }
   725→        }
   726→
   727→        xxa *= 0.98f;
   728→        yya *= 0.98f;
   729→        yRotA *= 0.9f;
   730→
   731→        float normalSpeed = walkingSpeed;
   732→        float normalAirControllSpeed = flyingSpeed;
   733→        walkingSpeed *= getWalkingSpeedModifier();
   734→        flyingSpeed  *= getWalkingSpeedModifier();
   735→        travel(xxa, yya);
   736→        walkingSpeed = normalSpeed;
   737→        flyingSpeed = normalAirControllSpeed;
   738→
   739→        TIMER_POP_PUSH("push");
   740→        Player* player = NULL;
   741→        if(isPlayer()) {
   742→                player = (Player*)this;
   743→        }
   744→        if(player == NULL || !player->isSleeping())  {
   745→                EntityList& entities = level->getEntities(this, this->bb.grow(0.2f, 0, 0.2f));
   746→                for (unsigned int i = 0; i < entities.size(); i++) {
   747→                        Entity* e = entities[i];
   748→                        if(e->isPlayer()) {
   749→                                Player* p = (Player*)e;
   750→                                if(p->isSleeping())
   751→                                        continue;
   752→                        }
   753→                        if (e->isPushable()) e->push(this);
   754→                }
   755→        }
   756→        TIMER_POP();
   757→}

   760→void Mob::updateAi() {
   761→    noActionTime++;
   762→
   763→        Player* player = level->getNearestPlayer(this, -1);
   764→        checkDespawn(player);
   765→
   766→    xxa = 0;
   767→    yya = 0;
   768→
   769→    float lookDistance = 8;
   770→    if (random.nextFloat() < 0.02f) {
   771→        player = level->getNearestPlayer(this, lookDistance);
   772→        if (player != NULL) {
   773→            lookingAtId = player->entityId;
   774→            lookTime = 10 + random.nextInt(20);
   775→        } else {
   776→            yRotA = (random.nextFloat() - 0.5f) * 20;
   777→        }
   778→    }
   779→
   780→    if (lookingAtId != 0) {
   781→                Entity* lookingAt = level->getEntity(lookingAtId);
   782→                lookingAtId = 0;
   783→                if (lookingAt != NULL) {
   784→                        lookAt(lookingAt, 10, (float)getMaxHeadXRot());
   785→                        if (lookTime-- <= 0 || lookingAt->removed || lookingAt->distanceToSqr(this) > lookDistance * lookDistance) {
   786→                    lookingAtId = 0;
   787→                    } else {
   788→                                lookingAtId = lookingAt->entityId;
   789→                        }
   790→                }
   791→    } else {
   792→        if (random.nextFloat() < 0.05f) {
   793→            yRotA = (random.nextFloat() - 0.5f) * 20;
   794→        }
   795→        yRot += yRotA;
   796→        xRot = defaultLookAngle;
   797→    }
   798→
   799→    bool inWater = isInWater();
   800→    bool inLava = isInLava();
   801→    if (inWater || inLava) jumping = random.nextFloat() < 0.8f;
   802→}

   860→{
   861→        float xd = e->x - x;

   890→float Mob::rotlerp( float a, float b, float max )
   891→{
   892→        float diff = b - a;
   893→        while (diff < -180)
   894→                diff += 360;
   895→        while (diff >= 180)
   896→                diff -= 360;
   897→        if (diff > max) {
   898→                diff = max;
   899→        }
   900→        if (diff < -max) {
   901→                diff = -max;
   902→        }
   903→        return a + diff;
   904→}

   963→        float xSin = (float) Mth::sin(-xRot * Mth::DEGRAD);
   964→
   965→        return Vec3(ySin * xCos, xSin, yCos * xCos); // newTemp

  1049→void Mob::checkDespawn(Mob* nearestBlocking) {
  1050→        if (nearestBlocking != NULL) {
  1051→                const bool removeIfFar = removeWhenFarAway();

  1053→                float yd = nearestBlocking->y - y;
  1054→                float zd = nearestBlocking->z - z;
  1055→                float sd = xd * xd + yd * yd + zd * zd;
  1056→
  1057→                if (removeIfFar && sd > 96 * 96) {
  1058→                        //LOGI("removed some angry mob more than 10 meters away!\n");
  1059→                        remove();
  1060→                }
  1061→
  1062→                if (noActionTime > 30 * SharedConstants::TicksPerSecond && random.nextInt(800) == 0 && removeIfFar && sd > 32 * 32)
  1063→                        remove();
  1064→                else
  1065→                        noActionTime = 0;
  1066→        }
  1067→}
