   100→   100→void LevelRenderer::generateSky() {
   101→   101→        Tesselator& t = Tesselator::instance;
   102→   102→        float yy;
   103→   103→        int s = 128;
   104→   104→        int d = (256 / s) + 2;
   105→   105→        yy = (float) (16);
   106→   106→        t.begin();
   107→   107→
   108→   108→        skyVertexCount = 0;
   109→   109→        for (int xx = -s * d; xx <= s * d; xx += s) {
   110→   110→                for (int zz = -s * d; zz <= s * d; zz += s) {
   111→   111→                        t.vertex((float) xx + 0,  yy, (float) zz + 0);
   112→   112→                        t.vertex((float)(xx + s), yy, (float) zz + 0);
   113→   113→                        t.vertex((float)(xx + s), yy, (float)(zz + s));
   114→   114→                        t.vertex((float) xx + 0,  yy, (float)(zz + s));
   115→   115→                        //LOGI("x, z: %d, %d\n", xx, zz);
   116→   116→                        skyVertexCount += 4;
   117→   117→                }
   118→   118→        }
   119→   119→
   120→   120→        t.end(true, skyBuffer);
   121→   121→        //LOGI("skyvertexcount: %d\n", skyVertexCount);
   122→   122→        //glEndList();
   123→   123→}

   148→   148→void LevelRenderer::allChanged()
   149→   149→{
   150→   150→        deleteChunks();
   151→   151→
   152→   152→        Tile::leaves->setFancy(mc->options.fancyGraphics);
   153→   153→        Tile::leaves_carried->setFancy(mc->options.fancyGraphics);
   154→   154→        lastViewDistance = mc->options.viewDistance;
   155→   155→
   156→   156→        int dist = (512 >> 3) << (3 - lastViewDistance);
   157→   157→        if (lastViewDistance <= 2 && mc->isPowerVR())
   158→   158→                dist = (int)((float)dist * 0.8f);
   159→   159→        LOGI("last: %d, power: %d\n", lastViewDistance, mc->isPowerVR());
   160→   160→
   161→   161→        #if defined(RPI)
   162→   162→                dist *= 0.6f;
   163→   163→        #endif
   164→   164→
   165→   165→        if (dist > 400) dist = 400;
   166→   166→        /*
   167→   167→        * if (Minecraft.FLYBY_MODE) { dist = 512 - CHUNK_SIZE * 2; }
   168→   168→        */
   169→   169→        xChunks = (dist / LevelRenderer::CHUNK_SIZE) + 1;
   170→   170→        yChunks = (128 /  LevelRenderer::CHUNK_SIZE);
   171→   171→        zChunks = (dist / LevelRenderer::CHUNK_SIZE) + 1;
   172→   172→        chunksLength = xChunks * yChunks * zChunks;
   173→   173→        LOGI("chunksLength: %d. Distance: %d\n", chunksLength, dist);
   174→   174→
   175→   175→        chunks = new Chunk*[chunksLength];
   176→   176→        sortedChunks = new Chunk*[chunksLength];
   177→   177→
   178→   178→        int id = 0;
   179→   179→        int count = 0;
   180→   180→
   181→   181→        xMinChunk = 0;
   182→   182→        yMinChunk = 0;
   183→   183→        zMinChunk = 0;
   184→   184→        xMaxChunk = xChunks;
   185→   185→        yMaxChunk = yChunks;
   186→   186→        zMaxChunk = zChunks;
   187→   187→        dirtyChunks.clear();
   188→   188→        //renderableTileEntities.clear();
   189→   189→
   190→   190→        for (int x = 0; x < xChunks; x++) {
   191→   191→                for (int y = 0; y < yChunks; y++) {
   192→   192→                        for (int z = 0; z < zChunks; z++) {
   193→   193→                                const int c = getLinearCoord(x, y, z);
   194→   194→                                Chunk* chunk = new Chunk(level, x * CHUNK_SIZE, y * CHUNK_SIZE, z * CHUNK_SIZE, CHUNK_SIZE, chunkLists + id, &chunkBuffers[id]);
   195→   195→
   196→   196→                                if (occlusionCheck) {
   197→   197→                                        chunk->occlusion_id = 0;//occlusionCheckIds.get(count);
   198→   198→                                }
   199→   199→                                chunk->occlusion_querying = false;
   200→   200→                                chunk->occlusion_visible = true;
   201→   201→                                chunk->visible = true;
   202→   202→                                chunk->id = count++;
   203→   203→                                chunk->setDirty();
   204→   204→
   205→   205→                                chunks[c] = chunk;
   206→   206→                                sortedChunks[c] = chunk;
   207→   207→                                dirtyChunks.push_back(chunk);
   208→   208→
   209→   209→                                id += 3;
   210→   210→                        }
   211→   211→                }
   212→   212→        }
   213→   213→
   214→   214→        if (level != NULL) {
   215→   215→                Entity* player = mc->cameraTargetPlayer;
   216→   216→                if (player != NULL) {
   217→   217→                        this->resortChunks(Mth::floor(player->x), Mth::floor(player->y), Mth::floor(player->z));
   218→   218→                        DistanceChunkSorter distanceSorter(player);
   219→   219→                        std::sort(sortedChunks, sortedChunks + chunksLength, distanceSorter);
   220→   220→                }
   221→   221→        }
   222→   222→        noEntityRenderFrames = 2;
   223→   223→}

   225→   225→void LevelRenderer::deleteChunks()
   226→   226→{
   227→   227→        for (int z = 0; z < zChunks; ++z)
   228→   228→        for (int y = 0; y < yChunks; ++y)
   229→   229→        for (int x = 0; x < xChunks; ++x) {
   230→   230→                int c = getLinearCoord(x, y, z);
   231→   231→                delete chunks[c];
   232→   232→        }
   233→   233→
   234→   234→        delete[] chunks;
   235→   235→        chunks = NULL;
   236→   236→
   237→   237→        delete[] sortedChunks;
   238→   238→        sortedChunks = NULL;

   241→   241→void LevelRenderer::resortChunks( int xc, int yc, int zc )
   242→   242→{
   243→   243→        xc -= CHUNK_SIZE / 2;
   244→   244→        //yc -= CHUNK_SIZE / 2;
   245→   245→        zc -= CHUNK_SIZE / 2;
   246→   246→        xMinChunk = INT_MAX;
   247→   247→        yMinChunk = INT_MAX;
   248→   248→        zMinChunk = INT_MAX;
   249→   249→        xMaxChunk = INT_MIN;
   250→   250→        yMaxChunk = INT_MIN;
   251→   251→        zMaxChunk = INT_MIN;
   252→   252→
   253→   253→        int dirty = 0;
   254→   254→
   255→   255→        int s2 = xChunks * CHUNK_SIZE;
   256→   256→        int s1 = s2 / 2;
   257→   257→
   258→   258→        for (int x = 0; x < xChunks; x++) {
   259→   259→                int xx = x * CHUNK_SIZE;
   260→   260→
   261→   261→                int xOff = (xx + s1 - xc);
   262→   262→                if (xOff < 0) xOff -= (s2 - 1);
   263→   263→                xOff /= s2;
   264→   264→                xx -= xOff * s2;
   265→   265→
   266→   266→                if (xx < xMinChunk) xMinChunk = xx;
   267→   267→                if (xx > xMaxChunk) xMaxChunk = xx;
   268→   268→
   269→   269→                for (int z = 0; z < zChunks; z++) {
   270→   270→                        int zz = z * CHUNK_SIZE;
   271→   271→                        int zOff = (zz + s1 - zc);
   272→   272→                        if (zOff < 0) zOff -= (s2 - 1);
   273→   273→                        zOff /= s2;
   274→   274→                        zz -= zOff * s2;
   275→   275→
   276→   276→                        if (zz < zMinChunk) zMinChunk = zz;
   277→   277→                        if (zz > zMaxChunk) zMaxChunk = zz;
   278→   278→
   279→   279→                        for (int y = 0; y < yChunks; y++) {
   280→   280→                                int yy = y * CHUNK_SIZE;
   281→   281→                                if (yy < yMinChunk) yMinChunk = yy;
   282→   282→                                if (yy > yMaxChunk) yMaxChunk = yy;
   283→   283→
   284→   284→                                Chunk* chunk = chunks[(z * yChunks + y) * xChunks + x];
   285→   285→                                bool wasDirty = chunk->isDirty();
   286→   286→                                chunk->setPos(xx, yy, zz);
   287→   287→                                if (!wasDirty && chunk->isDirty()) {
   288→   288→                                        dirtyChunks.push_back(chunk);
   289→   289→                                        ++dirty;
   290→   290→                                }
   291→   291→                        }
   292→   292→                }
   293→   293→        }
   294→   294→}

   296→   296→int LevelRenderer::render( Mob* player, int layer, float alpha )
   297→   297→{
   298→   298→        if (mc->options.viewDistance != lastViewDistance) {
   299→   299→                allChanged();
   300→   300→        }
   301→   301→
   302→   302→        TIMER_PUSH("sortchunks");
   303→   303→        for (int i = 0; i < 10; i++) {
   304→   304→                chunkFixOffs = (chunkFixOffs + 1) % chunksLength;
   305→   305→                Chunk* c = chunks[chunkFixOffs];
   306→   306→                if (c->isDirty() && std::find(dirtyChunks.begin(), dirtyChunks.end(), c) == dirtyChunks.end()) {
   307→   307→                        dirtyChunks.push_back(c);
   308→   308→                }
   309→   309→        }
   310→   310→
   311→   311→        if (layer == 0) {
   312→   312→                totalChunks = 0;
   313→   313→                offscreenChunks = 0;
   314→   314→                occludedChunks = 0;
   315→   315→                renderedChunks = 0;
   316→   316→                emptyChunks = 0;
   317→   317→        }
   318→   318→
   319→   319→        float xOff = player->xOld + (player->x - player->xOld) * alpha;
   320→   320→        float yOff = player->yOld + (player->y - player->yOld) * alpha;
   321→   321→        float zOff = player->zOld + (player->z - player->zOld) * alpha;
   322→   322→
   323→   323→        float xd = player->x - xOld;
   324→   324→        float yd = player->y - yOld;
   325→   325→        float zd = player->z - zOld;
   326→   326→        if (xd * xd + yd * yd + zd * zd > 4 * 4) {
   327→   327→                xOld = player->x;
   328→   328→                yOld = player->y;
   329→   329→                zOld = player->z;
   330→   330→
   331→   331→                resortChunks(Mth::floor(player->x), Mth::floor(player->y), Mth::floor(player->z));
   332→   332→                DistanceChunkSorter distanceSorter(player);
   333→   333→                std::sort(sortedChunks, sortedChunks + chunksLength, distanceSorter);
   334→   334→        }
   335→   335→
   336→   336→        int count = 0;
   337→   337→        if (occlusionCheck && !mc->options.anaglyph3d && layer == 0) {
   338→   338→                int from = 0;
   339→   339→                int to = 16;
   340→   340→                //checkQueryResults(from, to);
   341→   341→                for (int i = from; i < to; i++) {
   342→   342→                        sortedChunks[i]->occlusion_visible = true;
   343→   343→                }
   344→   344→
   345→   345→                count += renderChunks(from, to, layer, alpha);
   346→   346→
   347→   347→                do {
   348→   348→                        from = to;
   349→   349→                        to = to * 2;
   350→   350→                        if (to > chunksLength) to = chunksLength;
   351→   351→
   352→   352→                        glDisable2(GL_TEXTURE_2D);
   353→   353→                        glDisable2(GL_LIGHTING);
   354→   354→                        glDisable2(GL_ALPHA_TEST);
   355→   355→                        glDisable2(GL_FOG);
   356→   356→
   357→   357→                        glColorMask(false, false, false, false);
   358→   358→                        glDepthMask(false);
   359→   359→                        //checkQueryResults(from, to);
   360→   360→                        glPushMatrix2();
   361→   361→                        float xo = 0;
   362→   362→                        float yo = 0;
   363→   363→                        float zo = 0;
   364→   364→                        for (int i = from; i < to; i++) {
   365→   365→                                if (sortedChunks[i]->isEmpty()) {
   366→   366→                                        sortedChunks[i]->visible = false;
   367→   367→                                        continue;
   368→   368→                                }
   369→   369→                                if (!sortedChunks[i]->visible) {
   370→   370→                                        sortedChunks[i]->occlusion_visible = true;
   371→   371→                                }
   372→   372→
   373→   373→                                if (sortedChunks[i]->visible && !sortedChunks[i]->occlusion_querying) {
   374→   374→                                        float dist = Mth::sqrt(sortedChunks[i]->distanceToSqr(player));
   375→   375→
   376→   376→                                        int frequency = (int) (1 + dist / 128);
   377→   377→
   378→   378→                                        if (ticks % frequency == i % frequency) {
   379→   379→                                                Chunk* chunk = sortedChunks[i];
   380→   380→                                                float xt = (float) (chunk->x - xOff);
   381→   381→                                                float yt = (float) (chunk->y - yOff);
   382→   382→                                                float zt = (float) (chunk->z - zOff);
   383→   383→                                                float xdd = xt - xo;
   384→   384→                                                float ydd = yt - yo;
   385→   385→                                                float zdd = zt - zo;
   386→   386→
   387→   387→                                                if (xdd != 0 || ydd != 0 || zdd != 0) {
   388→   388→                                                        glTranslatef2(xdd, ydd, zdd);
   389→   389→                                                        xo += xdd;
   390→   390→                                                        yo += ydd;
   391→   391→                                                        zo += zdd;
   392→   392→                                                }
   393→   393→
   394→   394→                                                sortedChunks[i]->renderBB();
   395→   395→                                                sortedChunks[i]->occlusion_querying = true;
   396→   396→                                        }
   397→   397→                                }
   398→   398→                        }
   399→   399→                        glPopMatrix2();
   400→   400→                        glColorMask(true, true, true, true);
   401→   401→                        glDepthMask(true);
   402→   402→                        glEnable2(GL_TEXTURE_2D);
   403→   403→                        glEnable2(GL_ALPHA_TEST);
   404→   404→                        glEnable2(GL_FOG);
   405→   405→
   406→   406→                        count += renderChunks(from, to, layer, alpha);
   407→   407→
   408→   408→                } while (to < chunksLength);
   409→   409→
   410→   410→        } else {
   411→   411→                TIMER_POP_PUSH("render");
   412→   412→                count += renderChunks(0, chunksLength, layer, alpha);
   413→   413→        }
   414→   414→
   415→   415→        TIMER_POP();
   416→   416→        return count;
   417→   417→}

   593→   593→bool LevelRenderer::updateDirtyChunks( Mob* player, bool force )
   594→   594→{
   595→   595→        bool slow = false;
   596→   596→
   597→   597→        if (slow) {
   598→   598→                DirtyChunkSorter dirtySorter(player);
   599→   599→                std::sort(dirtyChunks.begin(), dirtyChunks.end(), dirtySorter);
   600→   600→                int s = dirtyChunks.size() - 1;
   601→   601→                int amount = dirtyChunks.size();
   602→   602→                for (int i = 0; i < amount; i++) {
   603→   603→                        Chunk* chunk = dirtyChunks[s-i];
   604→   604→                        if (!force) {
   605→   605→                                if (chunk->distanceToSqr(player) > 16 * 16) {
   606→   606→                                        if (chunk->visible) {
   607→   607→                                                if (i >= MAX_VISIBLE_REBUILDS_PER_FRAME) return false;
   608→   608→                                        } else {
   609→   609→                                                if (i >= MAX_INVISIBLE_REBUILDS_PER_FRAME) return false;
   610→   610→                                        }
   611→   611→                                }
   612→   612→                        } else {
   613→   613→                                if (!chunk->visible) continue;
   614→   614→                        }
   615→   615→                        chunk->rebuild();
   616→   616→
   617→   617→                        dirtyChunks.erase( std::find(dirtyChunks.begin(), dirtyChunks.end(), chunk) ); // @q: s-i?
   618→   618→                        chunk->setClean();
   619→   619→                }
   620→   620→
   621→   621→                return dirtyChunks.size() == 0;
   622→   622→        } else {
   623→   623→                const int count = 3;
   624→   624→
   625→   625→                DirtyChunkSorter dirtyChunkSorter(player);
   626→   626→                Chunk* toAdd[count] = {NULL};
   627→   627→                std::vector<Chunk*>* nearChunks = NULL;
   628→   628→
   629→   629→                int pendingChunkSize = dirtyChunks.size();
   630→   630→                int pendingChunkRemoved = 0;
   631→   631→
   632→   632→                for (int i = 0; i < pendingChunkSize; i++) {
   633→   633→                        Chunk* chunk = dirtyChunks[i];
   634→   634→
   635→   635→                        if (!force) {
   636→   636→                                if (chunk->distanceToSqr(player) > 1024.0f) {
   637→   637→                                        int index;
   638→   638→
   639→   639→                                        // is this chunk in the closest <count>?
   640→   640→                                        for (index = 0; index < count; index++) {
   641→   641→                                                if (toAdd[index] != NULL && dirtyChunkSorter(toAdd[index], chunk) == false) {
   642→   642→                                                        break;
   643→   643→                                                }
   644→   644→                                        }
   645→   645→
   646→   646→                                        index--;
   647→   647→
   648→   648→                                        if (index > 0) {
   649→   649→                                                int x = index;
   650→   650→                                                while (--x != 0) {
   651→   651→                                                        toAdd[x - 1] = toAdd[x];
   652→   652→                                                }
   653→   653→                                                toAdd[index] = chunk;
   654→   654→                                        }
   655→   655→
   656→   656→                                        continue;
   657→   657→                                }
   658→   658→                        } else if (!chunk->visible) {
   659→   659→                                continue;
   660→   660→                        }
   661→   661→
   662→   662→                        // chunk is very close -- always render
   663→   663→
   664→   664→                        if (nearChunks == NULL) {
   665→   665→                                nearChunks = new std::vector<Chunk*>();
   666→   666→                        }
   667→   667→
   668→   668→                        pendingChunkRemoved++;
   669→   669→                        nearChunks->push_back(chunk);
   670→   670→                        dirtyChunks[i] = NULL;
   671→   671→                }
   672→   672→
   673→   673→                // if there are nearby chunks that need to be prepared for
   674→   674→                // rendering, sort them and then process them
   675→   675→                static const float MaxFrameTime = 1.0f / 100.0f;
   676→   676→                Stopwatch chunkWatch;
   677→   677→                chunkWatch.start();
   678→   678→
   679→   679→                if (nearChunks != NULL) {
   680→   680→                        if (nearChunks->size() > 1) {
   681→   681→                                std::sort(nearChunks->begin(), nearChunks->end(), dirtyChunkSorter);
   682→   682→                        }
   683→   683→
   684→   684→                        for (int i = nearChunks->size() - 1; i >= 0; i--) {
   685→   685→                                Chunk* chunk = (*nearChunks)[i];
   686→   686→                                chunk->rebuild();
   687→   687→                                chunk->setClean();
   688→   688→                        }
   689→   689→                        delete nearChunks;
   690→   690→                }
   691→   691→
   692→   692→                // render the nearest <count> chunks (farther than 1024 units away)
   693→   693→                int secondaryRemoved = 0;
   694→   694→
   695→   695→                for (int i = count - 1; i >= 0; i--) {
   696→   696→                        Chunk* chunk = toAdd[i];
   697→   697→                        if (chunk != NULL) {
   698→   698→
   699→   699→                                float ttt = chunkWatch.stopContinue();
   700→   700→                                if (ttt >= MaxFrameTime) {
   701→   701→                                        //LOGI("Too much work, I quit2!\n");
   702→   702→                                        break;
   703→   703→                                }
   704→   704→
   705→   705→                                if (!chunk->visible && i != count - 1) {
   706→   706→                                        // escape early if chunks aren't ready
   707→   707→                                        toAdd[i] = NULL;
   708→   708→                                        toAdd[0] = NULL;
   709→   709→                                        break;
   710→   710→                                }
   711→   711→                                toAdd[i]->rebuild();
   712→   712→                                toAdd[i]->setClean();
   713→   713→                                secondaryRemoved++;
   714→   714→                        }
   715→   715→                }
   716→   716→
   717→   717→                // compact by removing nulls
   718→   718→                int cursor = 0;
   719→   719→                int target = 0;
   720→   720→                int arraySize = dirtyChunks.size();
   721→   721→                while (cursor != arraySize) {
   722→   722→                        Chunk* chunk = dirtyChunks[cursor];
   723→   723→                        if (chunk != NULL) {
   724→   724→                                bool remove = false;
   725→   725→                for (int i = 0; i < count && !remove; i++)
   726→   726→                    if (chunk == toAdd[i]) {
   727→   727→                        remove = true;
   728→   728→                    }
   729→   729→
   730→   730→                if (!remove) {
   731→   731→                                //if (chunk == toAdd[0] || chunk == toAdd[1] || chunk == toAdd[2]) {
   732→   732→                                //      ; // this chunk was rendered and should be removed
   733→   733→                                //} else {
   734→   734→                                        if (target != cursor) {
   735→   735→                                                dirtyChunks[target] = chunk;
   736→   736→                                        }
   737→   737→                                        target++;
   738→   738→                                }
   739→   739→                        }
   740→   740→                        cursor++;
   741→   741→                }
   742→   742→
   743→   743→                // trim
   744→   744→                if (cursor > target)
   745→   745→                        dirtyChunks.erase(dirtyChunks.begin() + target, dirtyChunks.end());
   746→   746→
   747→   747→                return pendingChunkSize == (pendingChunkRemoved + secondaryRemoved);
   748→   748→        }
   749→   749→}

   842→   842→{
   843→   843→        int _x0 = Mth::intFloorDiv(x0, CHUNK_SIZE);

   850→   850→        for (int x = _x0; x <= _x1; x++) {
   851→   851→                int xx = x % xChunks;
   852→   852→                if (xx < 0) xx += xChunks;
   853→   853→                for (int y = _y0; y <= _y1; y++) {
   854→   854→                        int yy = y % yChunks;
   855→   855→                        if (yy < 0) yy += yChunks;
   856→   856→                        for (int z = _z0; z <= _z1; z++) {
   857→   857→                                int zz = z % zChunks;
   858→   858→                                if (zz < 0) zz += zChunks;
   859→   859→
   860→   860→                                int p = ((zz) * yChunks + (yy)) * xChunks + (xx);
   861→   861→                                Chunk* chunk = chunks[p];
   862→   862→                                if (!chunk->isDirty()) {
   863→   863→                                        dirtyChunks.push_back(chunk);
   864→   864→                                        chunk->setDirty();
   865→   865→                                }
   866→   866→                        }
   867→   867→                }
   868→   868→        }
