   101→### Mob::travel(float xa, float ya) — THE movement physics
   102→**Verified formula (line-by-line):**
   103→```cpp
   104→void Mob::travel(float xa, float ya) {
   105→    // xa = strafe input, ya = forward input (from aiStep)
   106→
   107→    if (this->isAlive()) {
   108→        // 1. If no horizontal input, damp run animation
   109→        if (xa == 0 && ya == 0 && this->isControlledByLocalInstance()) {
   110→            this->walkAnimSpeed = 0;
   111→            // oBob = bob = 0
   112→        }
   113→
   114→        // 2. Get walking speed multiplier
   115→        float speed = this->getWalkingSpeedModifier();
   116→
   117→        // 3. Movement in water/lava handled differently (skip for port —
   118→        //    most mobs walk on land)
   119→
   120→        // 4. Apply movement input scaled by walking speed
   121→        float forward = ya;   // forward direction
   122→        float strafe = xa;     // strafe direction
   123→
   124→        // 5. If sneaking, clamp magnitude
   125→        if (this->isSneaking()) {
   126→            // clamp total horizontal speed to 0.3
   127→            float mag = sqrtf(forward*forward + strafe*strafe);
   128→            if (mag > 0.3f) {
   129→                forward = forward / mag * 0.3f;
   130→                strafe = strafe / mag * 0.3f;
   131→            }
   132→        }
   133→
   134→        // 6. Convert local (forward,strafe) to world (xd, zd) using yRot
   135→        float yaw = this->yRot * M_PI / 180.0f;  // degrees → radians
   136→        // sin/cos of yaw for the rotation
   137→        float sinYaw = sinf(yaw);
   138→        float cosYaw = cosf(yaw);
   139→        float wishXd = -sinYaw * forward + cosYaw * strafe;
   140→        float wishZd = -cosYaw * forward - sinYaw * strafe;
   141→        // (signs depend on Minecraft's coordinate convention)
   142→
   143→        // 7. Flying override (Player::travel overrides this)
   144→        if (this->abilities && this->abilities.flying) {
   145→            this->yd *= 0.6;       // damp vertical when flying
   146→            // use flying speed instead of walk speed
   147→            // skip gravity & jump
   148→            this->move(wishXd * flySpeed, this->yd, wishZd * flySpeed);
   149→            return;
   150→        }
   151→
   152→        // 8. Apply movement + gravity
   153→        this->move(wishXd * speed, this->yd, wishZd * speed);
   154→        this->yd -= 0.08;            // gravity per tick (NOT per second)
   155→        this->yd *= 0.98;            // air drag
   156→
   157→        // 9. If on ground, more friction
   158→        if (this->onGround) {
   159→            this->yd *= 0.6;          // ground friction on vertical
   160→            // horizontal friction handled by move()/collision
   161→        }
   162→    }
   163→}
   164→```

   166→### Mob::aiStep()
   167→```
   168→void Mob::aiStep() {
   169→    // 1. Decay hurt time
   170→    if (this->hurtTime > 0) this->hurtTime--;
   171→    if (this->attackTime > 0) this->attackTime--;
   172→
   173→    // 2. Decay invulnerable time
   174→    if (this->invulnerableTime > 0) this->invulnerableTime--;
   175→
   176→    // 3. Swing animation decay
   177→    if (this->swinging) {
   178→        this->swingTime++;
   179→        if (this->swingTime >= SWING_DURATION) {  // SWING_DURATION = 8
   180→            this->swinging = false;
   181→            this->swingTime = 0;
   182→        }
   183→    }
   184→
   185→    // 4. Read input from moveControl (set by aiStep in subclass)
   186→    this->xxa = this->moveControl->xxa;   // strafe
   187→    this->yya = this->moveControl->yya;   // forward
   188→    this->jumping = this->moveControl->jumping;
   189→
   190→    // 5. Update attack animation interpolation
   191→    this->oAttackAnim = this->attackAnim;
   192→    this->oRun = this->run;
   193→
   194→    // 6. Walk animation
   195→    this->walkAnimSpeedO = this->walkAnimSpeed;
   196→    float dist = sqrtf((this->x - this->xOld)*(this->x - this->xOld) +
   197→                       (this->z - this->zOld)*(this->z - this->zOld));
   198→    float speedFactor = Math::clamp(dist * 4 - 0.1f, 0, 1);
   199→    this->walkAnimSpeed += (speedFactor - this->walkAnimSpeed) * 0.4f;
   200→    this->walkAnimPos += this->walkAnimSpeed;
   201→    this->run += (dist - this->run) * 0.1f;
   202→
   203→    // 7. Body bob (player only)
   204→    this->bob += (sqrtf(dist*dist + (this->y - this->yOld)*(this->y - this->yOld)) * 0.5f - this->bob) * 0.1f;
   205→
   206→    // 8. Hurt tilt interpolation
   207→    this->oTilt = this->tilt;
   208→    if (this->hurtTime > 0 || this->deathTime > 0) {
   209→        this->tilt = Math::clamp(...);
   210→    } else {
   211→        this->tilt *= 0.4f;  // decay
   212→    }
   213→
   214→    // 9. Jump logic
   215→    if (this->jumping && this->onGround) {
   216→        this->jumpFromGround();
   217→    }
   218→
   219→    // 10. Apply input to movement: travel(xxa, yya)
   220→    this->travel(this->xxa, this->yya);
   221→
   222→    // 11. Walk animation step (legs/arms)
   223→    // ... (model animation)
   224→
   225→    // 12. xOld = x; yOld = y; zOld = z;  // for interpolation
   226→}

  1813→static float linearTransform(float value, float deadZone, float scale = 1.0f, bool limit1 = false) {
  1814→    float deadSigned = value >= 0 ? deadZone : -deadZone;
  1815→    if (std::abs(deadSigned) >= std::abs(value)) return 0;     // inside deadzone
  1816→    float ret = (value - deadSigned) * scale;
  1817→    if (limit1 && std::abs(ret) > 1) ret /= std::abs(ret);     // optional clamp to ±1
  1818→    return ret;
  1819→}
  1820→```
