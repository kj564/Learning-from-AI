   904→/**
   905→ * Wrapper that subscribes to ChatContext and passes stable individual values
   906→ * to the memo'd ChatForm. This prevents ChatForm from re-rendering on every
   907→ * streaming chunk: it only re-renders when the specific values it uses change.
   908→ */
   909→function ChatFormWrapper({
   910→  index = 0,
   911→  placeholder,
   912→  project,
   913→  routePending = false,
   914→  isLandingPage,
   915→  /** Defaults to the atom's own default (`atomWithLocalStorage('enterToSend',
   916→   *  true)`) so call sites that predate this prop, mainly tests, keep their
   917→   *  prior behavior without passing it explicitly. */
   918→  enterToSend = true,
   919→  /** Same reason, from `atomWithLocalStorage('autoSendText', -1)`. */
   920→  autoSendText = -1,
   921→  /** Same reason, from the `speechSettingsInitialized` atom's `false`. */
   922→  speechSettingsInitialized = false,
   923→  footerBelow,
   924→  centerFormOnLanding,
   925→}: {
   926→  index?: number;
   927→  placeholder?: string;
   928→  project?: TChatProject;
   929→  routePending?: boolean;
   930→  enterToSend?: boolean;
   931→  autoSendText?: number;
   932→  speechSettingsInitialized?: boolean;
   933→  isLandingPage: boolean;
   934→  footerBelow: boolean;
   935→  centerFormOnLanding: boolean;
   936→}) {
   937→  const {
   938→    files,
   939→    setFiles,
   940→    conversation,
   941→    setConversation,
   942→    isSubmitting,
   943→    setFilesLoading,
   944→    newConversation,
   945→    handleStopGenerating,
   946→    stopGenerating,
   947→  } = useChatContext();
   948→
   949→  /**
   950→   * Stabilize conversation reference: only update when rendering-relevant fields change,
   951→   * not on every metadata update (e.g., title generation during streaming).
   952→   */
   953→  const hasMessages = (conversation?.messages?.length ?? 0) > 0;
   954→  const stableConversation = useMemo(
   955→    () => conversation,
   956→    // eslint-disable-next-line react-hooks/exhaustive-deps
   957→    [
   958→      conversation?.conversationId,
   959→      conversation?.endpoint,
   960→      conversation?.endpointType,
   961→      conversation?.agent_id,
   962→      conversation?.assistant_id,
   963→      conversation?.spec,
   964→      conversation?.useResponsesApi,
   965→      conversation?.model,
   966→      conversation?.maxContextTokens,
   967→      conversation?.codeApprovalMode,
   968→      conversation?.codeEnvironmentMode,
   969→      conversation?.codeWorkspaces,
   970→      /** The composer's reasoning control reads its configured value off this
   971→       *  object. Leaving these out held the old level on screen after a
   972→       *  Parameters-panel edit, while the next send already used the new one. */
   973→      conversation?.reasoning_effort,
   974→      conversation?.effort,
   975→      conversation?.thinkingLevel,
   976→      conversation?.thinkingBudget,
   977→      /** The attach picker resolves the upload route from it, and a Responses
   978→       *  routing policy can send web-search turns to a different route. */
   979→      conversation?.web_search,
   980→      hasMessages,
   981→    ],
   982→  );
   983→
   984→  /** Stabilize function refs so they never trigger ChatForm re-renders */
   985→  const handleStopRef = useRef(handleStopGenerating);
   986→  handleStopRef.current = handleStopGenerating;
   987→  const stableHandleStop = useCallback(
   988→    (e: React.MouseEvent<HTMLButtonElement>) => handleStopRef.current(e),
   989→    [],
   990→  );
   991→
   992→  const newConvoRef = useRef(newConversation);

   995→    (...args: Parameters<ConvoGenerator>): ReturnType<ConvoGenerator> =>
   996→      newConvoRef.current(...args),
   997→    [],
   998→  );
   999→
  1000→  const stopRef = useRef(stopGenerating);
  1001→  stopRef.current = stopGenerating;
  1002→  const stableStop = useCallback(() => {
  1003→    void stopRef.current();
  1004→  }, []);
  1005→
  1006→  return (
  1007→    <ChatForm
  1008→      index={index}
  1009→      enterToSend={enterToSend}
  1010→      autoSendText={autoSendText}

  1019→      setFiles={setFiles}
  1020→      conversation={stableConversation}
  1021→      setConversation={setConversation}
  1022→      isSubmitting={isSubmitting}
  1023→      setFilesLoading={setFilesLoading}
  1024→      newConversation={stableNewConversation}
  1025→      handleStopGenerating={stableHandleStop}
  1026→      stopGenerating={stableStop}
  1027→    />
  1028→  );
  1029→}
