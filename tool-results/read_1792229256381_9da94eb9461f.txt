  1812→```cpp
  1813→static float linearTransform(float value, float deadZone, float scale = 1.0f, bool limit1 = false) {
  1814→    float deadSigned = value >= 0 ? deadZone : -deadZone;
  1815→    if (std::abs(deadSigned) >= std::abs(value)) return 0;     // inside deadzone
  1816→    float ret = (value - deadSigned) * scale;
  1817→    if (limit1 && std::abs(ret) > 1) ret /= std::abs(ret);     // optional clamp to ±1
  1818→    return ret;
  1819→}

  2420→        float v = (1 - i / (float) (16));
