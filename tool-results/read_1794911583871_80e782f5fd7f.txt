    43→function renderMarkdown(text: string): string {
    44→  return text
    45→    .replace(/```(\w*)\n([\s\S]*?)```/g, '<pre class="bg-zinc-800/80 rounded-lg p-3 my-2 overflow-x-auto text-xs font-mono"><code>$2</code></pre>')
    46→    .replace(/`([^`]+)`/g, '<code class="bg-zinc-700/60 px-1.5 py-0.5 rounded text-xs text-indigo-300">$1</code>')
    47→    .replace(/\*\*([^*]+)\*\*/g, '<strong class="font-semibold text-white">$1</strong>')
    48→    .replace(/^### (.+)$/gm, '<h3 class="text-sm font-semibold mt-3 mb-1 text-zinc-200">$1</h3>')
    49→    .replace(/^## (.+)$/gm, '<h2 class="text-base font-semibold mt-4 mb-1 text-zinc-100">$1</h2>')
    50→    .replace(/^# (.+)$/gm, '<h1 class="text-lg font-bold mt-4 mb-2 text-white">$1</h1>')
    51→    .replace(/^\- (.+)$/gm, '<li class="ml-5 list-disc text-zinc-300">$1</li>')
    52→    .replace(/^\d+\. (.+)$/gm, '<li class="ml-5 list-decimal text-zinc-300">$1</li>')
    53→    .replace(/\n\n/g, '</p><p class="mt-2">')
    54→    .replace(/\n/g, '<br/>')
    55→}
