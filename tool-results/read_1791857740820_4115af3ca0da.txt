   100→void LevelRenderer::generateSky() {
   101→        Tesselator& t = Tesselator::instance;
   102→        float yy;
   103→        int s = 128;
   104→        int d = (256 / s) + 2;
   105→        yy = (float) (16);
   106→        t.begin();
   107→
   108→        skyVertexCount = 0;
   109→        for (int xx = -s * d; xx <= s * d; xx += s) {
   110→                for (int zz = -s * d; zz <= s * d; zz += s) {
   111→                        t.vertex((float) xx + 0,  yy, (float) zz + 0);
   112→                        t.vertex((float)(xx + s), yy, (float) zz + 0);
   113→                        t.vertex((float)(xx + s), yy, (float)(zz + s));
   114→                        t.vertex((float) xx + 0,  yy, (float)(zz + s));
   115→                        //LOGI("x, z: %d, %d\n", xx, zz);
   116→                        skyVertexCount += 4;
   117→                }
   118→        }
   119→
   120→        t.end(true, skyBuffer);
   121→        //LOGI("skyvertexcount: %d\n", skyVertexCount);
   122→        //glEndList();
   123→}

   148→void LevelRenderer::allChanged()
   149→{
   150→        deleteChunks();
   151→
   152→        Tile::leaves->setFancy(mc->options.fancyGraphics);
   153→        Tile::leaves_carried->setFancy(mc->options.fancyGraphics);
   154→        lastViewDistance = mc->options.viewDistance;
   155→
   156→        int dist = (512 >> 3) << (3 - lastViewDistance);
   157→        if (lastViewDistance <= 2 && mc->isPowerVR())
   158→                dist = (int)((float)dist * 0.8f);
   159→        LOGI("last: %d, power: %d\n", lastViewDistance, mc->isPowerVR());
   160→
   161→        #if defined(RPI)
   162→                dist *= 0.6f;
   163→        #endif
   164→
   165→        if (dist > 400) dist = 400;
   166→        /*
   167→        * if (Minecraft.FLYBY_MODE) { dist = 512 - CHUNK_SIZE * 2; }
   168→        */
   169→        xChunks = (dist / LevelRenderer::CHUNK_SIZE) + 1;
   170→        yChunks = (128 /  LevelRenderer::CHUNK_SIZE);
   171→        zChunks = (dist / LevelRenderer::CHUNK_SIZE) + 1;
   172→        chunksLength = xChunks * yChunks * zChunks;
   173→        LOGI("chunksLength: %d. Distance: %d\n", chunksLength, dist);
   174→
   175→        chunks = new Chunk*[chunksLength];
   176→        sortedChunks = new Chunk*[chunksLength];
   177→
   178→        int id = 0;
   179→        int count = 0;
   180→
   181→        xMinChunk = 0;
   182→        yMinChunk = 0;
   183→        zMinChunk = 0;
   184→        xMaxChunk = xChunks;
   185→        yMaxChunk = yChunks;
   186→        zMaxChunk = zChunks;
   187→        dirtyChunks.clear();
   188→        //renderableTileEntities.clear();
   189→
   190→        for (int x = 0; x < xChunks; x++) {
   191→                for (int y = 0; y < yChunks; y++) {
   192→                        for (int z = 0; z < zChunks; z++) {
   193→                                const int c = getLinearCoord(x, y, z);
   194→                                Chunk* chunk = new Chunk(level, x * CHUNK_SIZE, y * CHUNK_SIZE, z * CHUNK_SIZE, CHUNK_SIZE, chunkLists + id, &chunkBuffers[id]);
   195→
   196→                                if (occlusionCheck) {
   197→                                        chunk->occlusion_id = 0;//occlusionCheckIds.get(count);
   198→                                }
   199→                                chunk->occlusion_querying = false;
   200→                                chunk->occlusion_visible = true;
   201→                                chunk->visible = true;
   202→                                chunk->id = count++;
   203→                                chunk->setDirty();
   204→
   205→                                chunks[c] = chunk;
   206→                                sortedChunks[c] = chunk;
   207→                                dirtyChunks.push_back(chunk);
   208→
   209→                                id += 3;
   210→                        }
   211→                }
   212→        }
   213→
   214→        if (level != NULL) {
   215→                Entity* player = mc->cameraTargetPlayer;
   216→                if (player != NULL) {
   217→                        this->resortChunks(Mth::floor(player->x), Mth::floor(player->y), Mth::floor(player->z));
   218→                        DistanceChunkSorter distanceSorter(player);
   219→                        std::sort(sortedChunks, sortedChunks + chunksLength, distanceSorter);
   220→                }
   221→        }
   222→        noEntityRenderFrames = 2;
   223→}

   225→void LevelRenderer::deleteChunks()
   226→{
   227→        for (int z = 0; z < zChunks; ++z)
   228→        for (int y = 0; y < yChunks; ++y)
   229→        for (int x = 0; x < xChunks; ++x) {
   230→                int c = getLinearCoord(x, y, z);
   231→                delete chunks[c];
   232→        }
   233→
   234→        delete[] chunks;
   235→        chunks = NULL;
   236→
   237→        delete[] sortedChunks;
   238→        sortedChunks = NULL;
   239→}

   241→void LevelRenderer::resortChunks( int xc, int yc, int zc )
   242→{
   243→        xc -= CHUNK_SIZE / 2;
   244→        //yc -= CHUNK_SIZE / 2;
   245→        zc -= CHUNK_SIZE / 2;
   246→        xMinChunk = INT_MAX;
   247→        yMinChunk = INT_MAX;
   248→        zMinChunk = INT_MAX;
   249→        xMaxChunk = INT_MIN;
   250→        yMaxChunk = INT_MIN;
   251→        zMaxChunk = INT_MIN;
   252→
   253→        int dirty = 0;
   254→
   255→        int s2 = xChunks * CHUNK_SIZE;
   256→        int s1 = s2 / 2;
   257→
   258→        for (int x = 0; x < xChunks; x++) {
   259→                int xx = x * CHUNK_SIZE;
   260→
   261→                int xOff = (xx + s1 - xc);
   262→                if (xOff < 0) xOff -= (s2 - 1);
   263→                xOff /= s2;
   264→                xx -= xOff * s2;
   265→
   266→                if (xx < xMinChunk) xMinChunk = xx;
   267→                if (xx > xMaxChunk) xMaxChunk = xx;
   268→
   269→                for (int z = 0; z < zChunks; z++) {
   270→                        int zz = z * CHUNK_SIZE;
   271→                        int zOff = (zz + s1 - zc);
   272→                        if (zOff < 0) zOff -= (s2 - 1);
   273→                        zOff /= s2;
   274→                        zz -= zOff * s2;
   275→
   276→                        if (zz < zMinChunk) zMinChunk = zz;
   277→                        if (zz > zMaxChunk) zMaxChunk = zz;
   278→
   279→                        for (int y = 0; y < yChunks; y++) {
   280→                                int yy = y * CHUNK_SIZE;
   281→                                if (yy < yMinChunk) yMinChunk = yy;
   282→                                if (yy > yMaxChunk) yMaxChunk = yy;
   283→
   284→                                Chunk* chunk = chunks[(z * yChunks + y) * xChunks + x];
   285→                                bool wasDirty = chunk->isDirty();
   286→                                chunk->setPos(xx, yy, zz);
   287→                                if (!wasDirty && chunk->isDirty()) {
   288→                                        dirtyChunks.push_back(chunk);
   289→                                        ++dirty;
   290→                                }
   291→                        }
   292→                }
   293→        }
   294→}

   296→int LevelRenderer::render( Mob* player, int layer, float alpha )
   297→{
   298→        if (mc->options.viewDistance != lastViewDistance) {
   299→                allChanged();
   300→        }
   301→
   302→        TIMER_PUSH("sortchunks");
   303→        for (int i = 0; i < 10; i++) {
   304→                chunkFixOffs = (chunkFixOffs + 1) % chunksLength;
   305→                Chunk* c = chunks[chunkFixOffs];
   306→                if (c->isDirty() && std::find(dirtyChunks.begin(), dirtyChunks.end(), c) == dirtyChunks.end()) {
   307→                        dirtyChunks.push_back(c);
   308→                }
   309→        }
   310→
   311→        if (layer == 0) {
   312→                totalChunks = 0;
   313→                offscreenChunks = 0;
   314→                occludedChunks = 0;
   315→                renderedChunks = 0;
   316→                emptyChunks = 0;
   317→        }
   318→
   319→        float xOff = player->xOld + (player->x - player->xOld) * alpha;
   320→        float yOff = player->yOld + (player->y - player->yOld) * alpha;
   321→        float zOff = player->zOld + (player->z - player->zOld) * alpha;
   322→
   323→        float xd = player->x - xOld;
   324→        float yd = player->y - yOld;
   325→        float zd = player->z - zOld;
   326→        if (xd * xd + yd * yd + zd * zd > 4 * 4) {
   327→                xOld = player->x;
   328→                yOld = player->y;
   329→                zOld = player->z;
   330→
   331→                resortChunks(Mth::floor(player->x), Mth::floor(player->y), Mth::floor(player->z));
   332→                DistanceChunkSorter distanceSorter(player);
   333→                std::sort(sortedChunks, sortedChunks + chunksLength, distanceSorter);
   334→        }
   335→
   336→        int count = 0;
   337→        if (occlusionCheck && !mc->options.anaglyph3d && layer == 0) {
   338→                int from = 0;
   339→                int to = 16;
   340→                //checkQueryResults(from, to);
   341→                for (int i = from; i < to; i++) {
   342→                        sortedChunks[i]->occlusion_visible = true;
   343→                }
   344→
   345→                count += renderChunks(from, to, layer, alpha);
   346→
   347→                do {
   348→                        from = to;
   349→                        to = to * 2;
   350→                        if (to > chunksLength) to = chunksLength;
   351→
   352→                        glDisable2(GL_TEXTURE_2D);
   353→                        glDisable2(GL_LIGHTING);
   354→                        glDisable2(GL_ALPHA_TEST);
   355→                        glDisable2(GL_FOG);
   356→
   357→                        glColorMask(false, false, false, false);
   358→                        glDepthMask(false);
   359→                        //checkQueryResults(from, to);
   360→                        glPushMatrix2();
   361→                        float xo = 0;
   362→                        float yo = 0;
   363→                        float zo = 0;
   364→                        for (int i = from; i < to; i++) {
   365→                                if (sortedChunks[i]->isEmpty()) {
   366→                                        sortedChunks[i]->visible = false;
   367→                                        continue;
   368→                                }
   369→                                if (!sortedChunks[i]->visible) {
   370→                                        sortedChunks[i]->occlusion_visible = true;
   371→                                }
   372→
   373→                                if (sortedChunks[i]->visible && !sortedChunks[i]->occlusion_querying) {
   374→                                        float dist = Mth::sqrt(sortedChunks[i]->distanceToSqr(player));
   375→
   376→                                        int frequency = (int) (1 + dist / 128);
   377→
   378→                                        if (ticks % frequency == i % frequency) {
   379→                                                Chunk* chunk = sortedChunks[i];
   380→                                                float xt = (float) (chunk->x - xOff);
   381→                                                float yt = (float) (chunk->y - yOff);
   382→                                                float zt = (float) (chunk->z - zOff);
   383→                                                float xdd = xt - xo;
   384→                                                float ydd = yt - yo;
   385→                                                float zdd = zt - zo;
   386→
   387→                                                if (xdd != 0 || ydd != 0 || zdd != 0) {
   388→                                                        glTranslatef2(xdd, ydd, zdd);
   389→                                                        xo += xdd;
   390→                                                        yo += ydd;
   391→                                                        zo += zdd;
   392→                                                }
   393→
   394→                                                sortedChunks[i]->renderBB();
   395→                                                sortedChunks[i]->occlusion_querying = true;
   396→                                        }
   397→                                }
   398→                        }
   399→                        glPopMatrix2();
   400→                        glColorMask(true, true, true, true);
   401→                        glDepthMask(true);
   402→                        glEnable2(GL_TEXTURE_2D);
   403→                        glEnable2(GL_ALPHA_TEST);
   404→                        glEnable2(GL_FOG);
   405→
   406→                        count += renderChunks(from, to, layer, alpha);
   407→
   408→                } while (to < chunksLength);
   409→
   410→        } else {
   411→                TIMER_POP_PUSH("render");
   412→                count += renderChunks(0, chunksLength, layer, alpha);
   413→        }
   414→
   415→        TIMER_POP();
   416→        return count;
   417→}

   537→int LevelRenderer::renderChunks( int from, int to, int layer, float alpha )
   538→{
   539→        _renderChunks.clear();
   540→        int count = 0;
   541→        for (int i = from; i < to; i++) {
   542→                if (layer == 0) {
   543→                        totalChunks++;
   544→                        if (sortedChunks[i]->empty[layer]) emptyChunks++;
   545→                        else if (!sortedChunks[i]->visible) offscreenChunks++;
   546→                        else if (occlusionCheck && !sortedChunks[i]->occlusion_visible) occludedChunks++;
   547→                        else renderedChunks++;
   548→                }
   549→
   550→                if (!sortedChunks[i]->empty[layer] && sortedChunks[i]->visible && sortedChunks[i]->occlusion_visible) {
   551→                        int list = sortedChunks[i]->getList(layer);
   552→                        if (list >= 0) {
   553→                                _renderChunks.push_back(sortedChunks[i]);
   554→                                count++;
   555→                        }
   556→                }
   557→        }
   558→
   559→        Mob* player = mc->cameraTargetPlayer;

   593→bool LevelRenderer::updateDirtyChunks( Mob* player, bool force )
   594→{
   595→        bool slow = false;
   596→
   597→        if (slow) {
   598→                DirtyChunkSorter dirtySorter(player);
   599→                std::sort(dirtyChunks.begin(), dirtyChunks.end(), dirtySorter);
   600→                int s = dirtyChunks.size() - 1;
   601→                int amount = dirtyChunks.size();
   602→                for (int i = 0; i < amount; i++) {
   603→                        Chunk* chunk = dirtyChunks[s-i];
   604→                        if (!force) {
   605→                                if (chunk->distanceToSqr(player) > 16 * 16) {
   606→                                        if (chunk->visible) {
   607→                                                if (i >= MAX_VISIBLE_REBUILDS_PER_FRAME) return false;
   608→                                        } else {
   609→                                                if (i >= MAX_INVISIBLE_REBUILDS_PER_FRAME) return false;
   610→                                        }
   611→                                }
   612→                        } else {
   613→                                if (!chunk->visible) continue;
   614→                        }
   615→                        chunk->rebuild();
   616→
   617→                        dirtyChunks.erase( std::find(dirtyChunks.begin(), dirtyChunks.end(), chunk) ); // @q: s-i?
   618→                        chunk->setClean();
   619→                }
   620→
   621→                return dirtyChunks.size() == 0;
   622→        } else {
   623→                const int count = 3;
   624→
   625→                DirtyChunkSorter dirtyChunkSorter(player);
   626→                Chunk* toAdd[count] = {NULL};
   627→                std::vector<Chunk*>* nearChunks = NULL;
   628→
   629→                int pendingChunkSize = dirtyChunks.size();
   630→                int pendingChunkRemoved = 0;
   631→
   632→                for (int i = 0; i < pendingChunkSize; i++) {
   633→                        Chunk* chunk = dirtyChunks[i];
   634→
   635→                        if (!force) {
   636→                                if (chunk->distanceToSqr(player) > 1024.0f) {
   637→                                        int index;
   638→
   639→                                        // is this chunk in the closest <count>?
   640→                                        for (index = 0; index < count; index++) {
   641→                                                if (toAdd[index] != NULL && dirtyChunkSorter(toAdd[index], chunk) == false) {
   642→                                                        break;
   643→                                                }
   644→                                        }
   645→
   646→                                        index--;
   647→
   648→                                        if (index > 0) {
   649→                                                int x = index;
   650→                                                while (--x != 0) {
   651→                                                        toAdd[x - 1] = toAdd[x];
   652→                                                }
   653→                                                toAdd[index] = chunk;
   654→                                        }
   655→
   656→                                        continue;
   657→                                }
   658→                        } else if (!chunk->visible) {
   659→                                continue;
   660→                        }
   661→
   662→                        // chunk is very close -- always render
   663→
   664→                        if (nearChunks == NULL) {
   665→                                nearChunks = new std::vector<Chunk*>();
   666→                        }
   667→
   668→                        pendingChunkRemoved++;
   669→                        nearChunks->push_back(chunk);
   670→                        dirtyChunks[i] = NULL;
   671→                }
   672→
   673→                // if there are nearby chunks that need to be prepared for
   674→                // rendering, sort them and then process them
   675→                static const float MaxFrameTime = 1.0f / 100.0f;
   676→                Stopwatch chunkWatch;
   677→                chunkWatch.start();
   678→
   679→                if (nearChunks != NULL) {
   680→                        if (nearChunks->size() > 1) {
   681→                                std::sort(nearChunks->begin(), nearChunks->end(), dirtyChunkSorter);
   682→                        }
   683→
   684→                        for (int i = nearChunks->size() - 1; i >= 0; i--) {
   685→                                Chunk* chunk = (*nearChunks)[i];
   686→                                chunk->rebuild();
   687→                                chunk->setClean();
   688→                        }
   689→                        delete nearChunks;
   690→                }
   691→
   692→                // render the nearest <count> chunks (farther than 1024 units away)
   693→                int secondaryRemoved = 0;
   694→
   695→                for (int i = count - 1; i >= 0; i--) {
   696→                        Chunk* chunk = toAdd[i];
   697→                        if (chunk != NULL) {
   698→
   699→                                float ttt = chunkWatch.stopContinue();
   700→                                if (ttt >= MaxFrameTime) {
   701→                                        //LOGI("Too much work, I quit2!\n");
   702→                                        break;
   703→                                }
   704→
   705→                                if (!chunk->visible && i != count - 1) {
   706→                                        // escape early if chunks aren't ready
   707→                                        toAdd[i] = NULL;
   708→                                        toAdd[0] = NULL;
   709→                                        break;
   710→                                }
   711→                                toAdd[i]->rebuild();
   712→                                toAdd[i]->setClean();
   713→                                secondaryRemoved++;
   714→                        }
   715→                }
   716→
   717→                // compact by removing nulls
   718→                int cursor = 0;
   719→                int target = 0;
   720→                int arraySize = dirtyChunks.size();
   721→                while (cursor != arraySize) {
   722→                        Chunk* chunk = dirtyChunks[cursor];
   723→                        if (chunk != NULL) {
   724→                                bool remove = false;
   725→                for (int i = 0; i < count && !remove; i++)
   726→                    if (chunk == toAdd[i]) {
   727→                        remove = true;
   728→                    }
   729→
   730→                if (!remove) {
   731→                                //if (chunk == toAdd[0] || chunk == toAdd[1] || chunk == toAdd[2]) {
   732→                                //      ; // this chunk was rendered and should be removed
   733→                                //} else {
   734→                                        if (target != cursor) {
   735→                                                dirtyChunks[target] = chunk;
   736→                                        }
   737→                                        target++;
   738→                                }
   739→                        }
   740→                        cursor++;
   741→                }
   742→
   743→                // trim
   744→                if (cursor > target)
   745→                        dirtyChunks.erase(dirtyChunks.begin() + target, dirtyChunks.end());
   746→
   747→                return pendingChunkSize == (pendingChunkRemoved + secondaryRemoved);
   748→        }
   749→}

   841→void LevelRenderer::setDirty( int x0, int y0, int z0, int x1, int y1, int z1 )
   842→{
   843→        int _x0 = Mth::intFloorDiv(x0, CHUNK_SIZE);
   844→        int _y0 = Mth::intFloorDiv(y0, CHUNK_SIZE);
   845→        int _z0 = Mth::intFloorDiv(z0, CHUNK_SIZE);
   846→        int _x1 = Mth::intFloorDiv(x1, CHUNK_SIZE);
   847→        int _y1 = Mth::intFloorDiv(y1, CHUNK_SIZE);
   848→        int _z1 = Mth::intFloorDiv(z1, CHUNK_SIZE);
   849→
   850→        for (int x = _x0; x <= _x1; x++) {
   851→                int xx = x % xChunks;
   852→                if (xx < 0) xx += xChunks;
   853→                for (int y = _y0; y <= _y1; y++) {
   854→                        int yy = y % yChunks;
   855→                        if (yy < 0) yy += yChunks;
   856→                        for (int z = _z0; z <= _z1; z++) {
   857→                                int zz = z % zChunks;
   858→                                if (zz < 0) zz += zChunks;
   859→
   860→                                int p = ((zz) * yChunks + (yy)) * xChunks + (xx);
   861→                                Chunk* chunk = chunks[p];
   862→                                if (!chunk->isDirty()) {
   863→                                        dirtyChunks.push_back(chunk);
   864→                                        chunk->setDirty();
   865→                                }
   866→                        }
   867→                }
   868→        }
   869→}

   911→void LevelRenderer::renderEntities(Vec3 cam, Culler* culler, float a) {
   912→    if (noEntityRenderFrames > 0) {
   913→        noEntityRenderFrames--;
   914→        return;
   915→    }
   916→
   917→        TIMER_PUSH("prepare");
   918→    TileEntityRenderDispatcher::getInstance()->prepare(level, textures, mc->font, mc->cameraTargetPlayer, a);
   919→    EntityRenderDispatcher::getInstance()->prepare(level, mc->font, mc->cameraTargetPlayer, &mc->options, a);
   920→
   921→    totalEntities = 0;
   922→    renderedEntities = 0;
   923→    culledEntities = 0;
   924→
   925→        Entity* player = mc->cameraTargetPlayer;
   926→    EntityRenderDispatcher::xOff = TileEntityRenderDispatcher::xOff = (player->xOld + (player->x - player->xOld) * a);
   927→    EntityRenderDispatcher::yOff = TileEntityRenderDispatcher::yOff = (player->yOld + (player->y - player->yOld) * a);
   928→    EntityRenderDispatcher::zOff = TileEntityRenderDispatcher::zOff = (player->zOld + (player->z - player->zOld) * a);
   929→
   930→        glEnableClientState2(GL_VERTEX_ARRAY);
   931→        glEnableClientState2(GL_TEXTURE_COORD_ARRAY);
   932→
   933→        TIMER_POP_PUSH("entities");
   934→        const EntityList& entities = level->getAllEntities();
   935→        totalEntities = entities.size();
   936→        if (totalEntities > 0) {
   937→                Entity** toRender = new Entity*[totalEntities];
   938→                for (int i = 0; i < totalEntities; i++) {
   939→                        Entity* entity = entities[i];
   940→
   941→                        if (entity->shouldRender(cam) && culler->isVisible(entity->bb))
   942→                        {
   943→                                if (entity == mc->cameraTargetPlayer && mc->options.thirdPersonView == 0 && mc->cameraTargetPlayer->isPlayer() && !((Player*)mc->cameraTargetPlayer)->isSleeping()) continue;
   944→                                if (entity == mc->cameraTargetPlayer && !mc->options.thirdPersonView)
   945→                                        continue;
   946→                                if (!level->hasChunkAt(Mth::floor(entity->x), Mth::floor(entity->y), Mth::floor(entity->z)))
   947→                                        continue;
   948→
   949→                                toRender[renderedEntities++] = entity;
   950→                                //EntityRenderDispatcher::getInstance()->render(entity, a);
   951→                        }
   952→                }
   953→
   954→                if (renderedEntities > 0) {
   955→                        std::sort(&toRender[0], &toRender[renderedEntities], entityRenderPredicate);
   956→                        for (int i = 0; i < renderedEntities; ++i) {
   957→                                EntityRenderDispatcher* disp = EntityRenderDispatcher::getInstance();
   958→                                disp->render(toRender[i], a);
   959→                        }
   960→                }
   961→
   962→                delete[] toRender;
   963→        }
   964→
   965→    TIMER_POP_PUSH("tileentities");
   966→    for (unsigned int i = 0; i < level->tileEntities.size(); i++) {
   967→        TileEntityRenderDispatcher::getInstance()->render(level->tileEntities[i], a);
   968→    }
   969→
   970→        glDisableClientState2(GL_VERTEX_ARRAY);
   971→        glDisableClientState2(GL_TEXTURE_COORD_ARRAY);
   972→
   973→        TIMER_POP();
   974→}

  1021→void LevelRenderer::renderClouds( float alpha ) {
  1022→        //if (!mc->level->dimension->isNaturalDimension()) return;
  1023→        glEnable2(GL_TEXTURE_2D);
  1024→        glDisable(GL_CULL_FACE);
  1025→        float yOffs = (float) (mc->player->yOld + (mc->player->y - mc->player->yOld) * alpha);
  1026→        int s = 32;
  1027→        int d = 256 / s;
  1028→        Tesselator& t = Tesselator::instance;
  1029→
  1030→        //glBindTexture(GL_TEXTURE_2D, texturesloadTexture("/environment/clouds.png"));
  1031→        textures->loadAndBindTexture("environment/clouds.png");
  1032→        
  1033→        glEnable(GL_BLEND);
  1034→        glBlendFunc(GL_SRC_ALPHA, GL_ONE_MINUS_SRC_ALPHA);
  1035→
  1036→        Vec3 cc = level->getCloudColor(alpha);
  1037→        float cr = (float) cc.x;
  1038→        float cg = (float) cc.y;
  1039→        float cb = (float) cc.z;
  1040→
  1041→        float scale = 1 / 2048.0f;
  1042→
  1043→        float time = (ticks + alpha);
  1044→        float xo = mc->player->xo + (mc->player->x - mc->player->xo) * alpha + time * 0.03f;
  1045→        float zo = mc->player->zo + (mc->player->z - mc->player->zo) * alpha;
  1046→        int xOffs = Mth::floor(xo / 2048);
  1047→        int zOffs = Mth::floor(zo / 2048);
  1048→        xo -= xOffs * 2048;
  1049→        zo -= zOffs * 2048;
  1050→
  1051→        float yy = /*level.dimension.getCloudHeight()*/ 128 - yOffs + 0.33f;//mc->player->y + 1;
  1052→        float uo = (float) (xo * scale);
  1053→        float vo = (float) (zo * scale);
  1054→        t.begin();
  1055→
  1056→        t.color(cr, cg, cb, 0.8f);
  1057→        for (int xx = -s * d; xx < +s * d; xx += s) {
  1058→                for (int zz = -s * d; zz < +s * d; zz += s) {
  1059→                        t.vertexUV((float)xx, yy, (float)zz + s, xx * scale + uo, (zz + s) * scale + vo);
  1060→                        t.vertexUV((float)xx + s, yy, (float)zz + s, (xx + s) * scale + uo, (zz + s) * scale + vo);
  1061→                        t.vertexUV((float)xx + s, yy, (float)zz, (xx + s) * scale + uo, zz * scale + vo);
  1062→                        t.vertexUV((float)xx, yy, (float)zz, xx * scale + uo, zz * scale + vo);
  1063→                }
  1064→        }
  1065→        t.endOverrideAndDraw();
  1066→        glColor4f(1, 1, 1, 1.0f);
  1067→        glDisable(GL_BLEND);
  1068→        glEnable(GL_CULL_FACE);
  1069→}

  1177→void LevelRenderer::renderHitSelect( Player* player, const HitResult& h, int mode, /*ItemInstance*/void* inventoryItem, float a )
  1178→{
  1179→        //if (h.type == TILE) LOGI("type: %s @ (%d, %d, %d)\n", Tile::tiles[level->getTile(h.x, h.y, h.z)]->getDescriptionId().c_str(), h.x, h.y, h.z);
  1180→
  1181→        if (mode == 0) {
  1182→
  1183→                Tesselator& t = Tesselator::instance;
  1184→                glEnable2(GL_BLEND);
  1185→                glDisable2(GL_TEXTURE_2D);
  1186→                glBlendFunc2(GL_SRC_ALPHA, GL_ONE);
  1187→                glBlendFunc2(GL_DST_COLOR, GL_SRC_COLOR);
  1188→                glEnable2(GL_DEPTH_TEST);
  1189→
  1190→                textures->loadAndBindTexture("terrain.png");
  1191→                
  1192→                int tileId = level->getTile(h.x, h.y, h.z);
  1193→                Tile* tile = tileId > 0 ? Tile::tiles[tileId] : NULL;
  1194→                glDisable2(GL_ALPHA_TEST);
  1195→
  1196→                //LOGI("block: %d - %d (%s)\n", tileId, level->getData(h.x, h.y, h.z), tile==NULL?"null" : tile->getDescriptionId().c_str() );
  1197→
  1198→                const float br = 0.65f;
  1199→                glColor4f2(br * 1.0f, br * 1.0f, br * 1.0f, br * 1.0f);
  1200→                glPushMatrix2();
  1201→
  1202→                //glPolygonOffset(-.3f, -.3f);
  1203→                glPolygonOffset(-1.f, -1.f); //Implementation dependent units
  1204→                glEnable2(GL_POLYGON_OFFSET_FILL);
  1205→                float xo = player->xOld + (player->x - player->xOld) * a;
  1206→                float yo = player->yOld + (player->y - player->yOld) * a;
  1207→                float zo = player->zOld + (player->z - player->zOld) * a;
  1208→
  1209→                t.begin();
  1210→                t.offset(-xo, -yo, -zo);
  1211→                t.noColor();
  1212→
  1213→                if (tile == NULL) tile = Tile::rock;
  1214→                tileRenderer->tesselateInWorld(tile, h.x, h.y, h.z);
  1215→
  1216→                t.draw();
  1217→                t.offset(0, 0, 0);
  1218→                glPolygonOffset(0.0f, 0.0f);
  1219→
  1220→                glDisable2(GL_POLYGON_OFFSET_FILL);
  1221→                glEnable2(GL_TEXTURE_2D);
  1222→
  1223→                glDepthMask(true);
  1224→                glPopMatrix2();
  1225→
  1226→                glEnable2(GL_ALPHA_TEST);
  1227→                glDisable2(GL_BLEND);
  1228→        }
