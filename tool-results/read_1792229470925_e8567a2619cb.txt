 10777→#ifndef STANDALONE_SERVER
 10778→    if (action->isRemove()) {
 10779→        if (missTime > 0) return;     // can't destroy during miss cooldown

 10791→        }

 10794→            player->swing();

 10796→            raknetInstance->send(packet);       // network: tell other clients
 10797→            gameMode->attack(player, hitResult.entity);
 10798→        } else if (action->isInteract()) {
 10799→            if (hitResult.entity->interactPreventDefault())

 10801→            InteractPacket packet(InteractPacket::Interact, player->entityId, hitResult.entity->entityId);
 10802→            raknetInstance->send(packet);
 10803→            gameMode->interact(player, hitResult.entity);
 10804→        }
 10805→    } else if (hitResult.type == TILE) {
 10806→        int x = hitResult.x;
 10807→        int y = hitResult.y;
 10808→        int z = hitResult.z;

 10812→        Tile* oldTile = Tile::tiles[oldTileId];

 10814→        if (action->isRemove()) {
 10815→            if (!oldTile) return;

 10817→            gameMode->startDestroyBlock(x, y, z, hitResult.f);    // begin breaking (verified Entry 11)
 10818→        }

 11221→    for (int tt = 0; tt < TEXTURE_COUNT; tt++) {       // 4 texture groups
 11222→        ParticleList& list = particles[tt];

 11241→void ParticleEngine::render(Entity* player, float a) {
 11242→    // Camera-billboard basis vectors for orienting particle quads toward camera
 11243→    float xa = (float) Mth::cos(player->yRot * Mth::PI / 180);
 11244→    float za = (float) Mth::sin(player->yRot * Mth::PI / 180);
 11245→
 11246→    float xa2 = -za * (float) Mth::sin(player->xRot * Mth::PI / 180);
 11247→    float za2 = xa * (float) Mth::sin(player->xRot * Mth::PI / 180);
 11248→    float ya = (float) Mth::cos(player->xRot * Mth::PI / 180);
 11249→
 11250→    Particle::xOff = (player->xOld+(player->x-player->xOld)*a);
 11251→    Particle::yOff = (player->yOld+(player->y-player->yOld)*a);
 11252→    Particle::zOff = (player->zOld+(player->z-player->zOld)*a);
 11253→
 11254→    // Render the first 3 texture groups (MISC, TERRAIN, ITEM)
 11255→    for (int tt = 0; tt < 3; tt++) {
 11256→        if (particles[tt].size() == 0) continue;
 11257→
 11258→        if (tt == MISC_TEXTURE)         textures->loadAndBindTexture("particles.png");
 11259→        else if (tt == TERRAIN_TEXTURE) textures->loadAndBindTexture("terrain.png");
 11260→        else if (tt == ITEM_TEXTURE)    textures->loadAndBindTexture("gui/items.png");
 11261→
 11262→        Tesselator& t = Tesselator::instance;
 11263→        t.begin();
 11264→        for (unsigned int i = 0; i < particles[tt].size(); i++) {
 11265→            Particle* p = particles[tt][i];
 11266→            p->render(t, a, xa, ya, za, xa2, za2);    // particle emits 4 vertices to tesselator
 11267→        }
 11268→        t.draw();
 11269→    }
 11270→
 11271→    renderLit(player, a);    // 4th group: ENTITY_PARTICLE_TEXTURE
 11272→}
 11273→```

 11275→**Billboard vectors:**
 11276→- `xa = cos(yRot)`, `za = sin(yRot)` — camera-right and camera-forward (in world XZ plane)
 11277→- `xa2 = -sin(yRot) * sin(xRot)`, `za2 = cos(yRot) * sin(xRot)`, `ya = cos(xRot)` — camera-up (with pitch tilt)
