   123→bool Level::checkAndHandleWater(const AABB& box, const Material* material, Entity* e) {
   124→    int x0 = Mth::floor(box.x0);
   125→    int x1 = Mth::floor(box.x1 + 1);
   126→
   127→    int y0 = Mth::floor(box.y0);
   128→    int y1 = Mth::floor(box.y1 + 1);
   129→
   130→    int z0 = Mth::floor(box.z0);
   131→    int z1 = Mth::floor(box.z1 + 1);
   132→
   133→    if (!hasChunksAt(x0, y0, z0, x1, y1, z1)) {
   134→        return false;
   135→    }
   136→
   137→    bool ok = false;
   138→    Vec3 current(0,0,0);
   139→    for (int x = x0; x < x1; x++)
   140→        for (int y = y0; y < y1; y++)
   141→            for (int z = z0; z < z1; z++) {
   142→                Tile* tile = Tile::tiles[getTile(x, y, z)];
   143→                if (tile != NULL && tile->material == material) {
   144→                    float yt0 = y + 1 - LiquidTile::getHeight(getData(x, y, z));
   145→                    if (y1 >= yt0) {
   146→                        ok = true;
   147→                        tile->handleEntityInside(this, x, y, z, e, current);
   148→                    }
   149→                }
   150→            }
   151→        float len = current.length();
   152→    if (len > 0) {
   153→        const float pow = 0.004f / len;
   154→        e->xd += current.x * pow;
   155→        e->yd += current.y * pow;
   156→        e->zd += current.z * pow;
   157→    }
   158→    return ok;
   159→}

   167→Player* Level::getNearestPlayer(float x, float y, float z, float maxDist) {
   168→    float maxDistSqr = maxDist * maxDist;
   169→        float best = -1;
   170→    Player* result = NULL;
   171→    for (unsigned int i = 0; i < players.size(); i++) {
   172→        Player* p = players[i];
   173→                if (p->removed) continue;
   174→                float dist = p->distanceToSqr(x, y, z);
   175→        if ((maxDist < 0 || dist < maxDistSqr) && (best == -1 || dist < best)) {
   176→            best = dist;
   177→            result = p;
   178→        }
   179→    }
   180→    return result;
   181→}

   185→        if (!isClientSide && levelData.getSpawnMobs()) {
   186→                static int _mobSpawnTick = 0;
   187→                const int MobSpawnInterval = 2;
   188→                if (++_mobSpawnTick >= MobSpawnInterval) {
   189→                        _mobSpawnTick = 0;
   190→                        TIMER_PUSH("mobSpawner");
   191→                        MobSpawner::tick(this,  _spawnEnemies && difficulty > Difficulty::PEACEFUL,
   192→                                                                        _spawnFriendlies && (levelData.getTime() % 400) < MobSpawnInterval);
   193→                        TIMER_POP();
   194→                }
   195→        }
   196→
   197→        TIMER_PUSH("chunkSource");
   198→        _chunkSource->tick();
   199→
   200→        updateSkyDarken();
   201→        if(_nightMode) {
   202→                long curTime = levelData.getTime();
   203→                if(curTime % TICKS_PER_DAY  != MIDDLE_OF_NIGHT_TIME) {
   204→                        if(curTime % TICKS_PER_DAY < MIDDLE_OF_NIGHT_TIME && (curTime + 20) % TICKS_PER_DAY > MIDDLE_OF_NIGHT_TIME) {
   205→                                curTime = MIDDLE_OF_NIGHT_TIME;
   206→                                SetTimePacket packet(curTime);
   207→                                raknetInstance->send(packet);
   208→                        }
   209→                        else {
   210→                                curTime += 20;
   211→                                if(curTime % 20  == 0) {
   212→                                        SetTimePacket packet(curTime);
   213→                                        raknetInstance->send(packet);
   214→                                }
   215→                        }
   216→                        setTime(curTime%TICKS_PER_DAY);
   217→                }
   218→        }
   219→        else {
   220→                long time = levelData.getTime() + 1;
   221→                //if (time % (saveInterval) == 0) {
   222→                //    save(false, NULL);
   223→                //}
   224→                levelData.setTime(time);
   225→                if ((time & 255) == 0) {
   226→                        SetTimePacket packet(time);
   227→                        raknetInstance->send(packet);
   228→                }
   229→        }
   230→        TIMER_POP_PUSH("tickPending");
   231→        tickPendingTicks(false);
   232→
   233→        TIMER_POP_PUSH("tickTiles");
   234→    tickTiles();
   235→
   236→        TIMER_POP_PUSH("sendEntityData");
   237→        for (unsigned int i = 0; i < entities.size(); ++i) {
   238→                Entity* e = entities[i];
   239→                SynchedEntityData* data = e->getEntityData();
   240→                if (data && data->isDirty()) {
   241→                        SetEntityDataPacket packet(e->entityId, *data);
   242→                        raknetInstance->send(packet);
   243→                }
   244→        }
   245→
   246→        //if (!_pendingEntityData.empty()) {
   247→        //      for (EntityMap::iterator it = _pendingEntityData.begin(); it != _pendingEntityData.end(); ++it) {
   248→        //              SetEntityDataPacket packet(it->first, it->second->getEntityData());
   249→        //              raknetInstance->send(packet);
   250→        //      }
   251→        //      _pendingEntityData.clear();
   252→        //}
   253→        TIMER_POP();
   254→}

   339→bool Level::tickPendingTicks(bool force) {
   340→    int count = _tickNextTickSet.size();
   341→    if (count > MAX_TICK_TILES_PER_TICK) count = MAX_TICK_TILES_PER_TICK;
   342→    for (int i = 0; i < count; i++) {
   343→                TickDataSet::iterator td = _tickNextTickSet.begin();
   344→        if (!force && td->delay > levelData.getTime()) {
   345→            break;
   346→        }
   347→
   348→                int r = 8;
   349→        if (hasChunksAt(td->x - r, td->y - r, td->z - r, td->x + r, td->y + r, td->z + r)) {
   350→            int id = getTile(td->x, td->y, td->z);
   351→            if (id == td->tileId && id > 0) {
   352→                Tile::tiles[id]->tick(this, td->x, td->y, td->z, &random);
   353→            }
   354→        }
   355→        _tickNextTickSet.erase(td);
   356→    }
   357→    return _tickNextTickSet.size() != 0;
   358→}

   378→bool Level::findPath(Path* path, Entity* from, Entity* to, float maxDist, bool canOpenDoors, bool avoidWater) {
   379→    TIMER_PUSH("pathfind");
   380→        int x = Mth::floor(from->x);
   381→    int y = Mth::floor(from->y);
   382→    int z = Mth::floor(from->z);
   383→
   384→    int r = (int) (maxDist + 16);
   385→    int x1 = x - r;
   386→    int y1 = y - r;
   387→    int z1 = z - r;
   388→    int x2 = x + r;
   389→    int y2 = y + r;
   390→    int z2 = z + r;
   391→        //LOGI("trying to move! 1: %ld\n", levelData.getTime());
   392→    Region region(this, x1, y1, z1, x2, y2, z2);
   393→        _pathFinder->setLevelSource(&region);
   394→        _pathFinder->canOpenDoors = canOpenDoors;
   395→        _pathFinder->avoidWater = avoidWater;
   396→        _pathFinder->findPath(path, from, to, maxDist);
   397→        TIMER_POP();
   398→        return true;
   399→}

   429→void Level::setInitialSpawn() {
   430→    isFindingSpawn = true;
   431→    int xSpawn = CHUNK_CACHE_WIDTH * CHUNK_WIDTH / 2; // (Level.MAX_LEVEL_SIZE - 100) * 0;
   432→    int ySpawn = 64;
   433→    int zSpawn = CHUNK_CACHE_WIDTH * CHUNK_DEPTH / 2; // (Level.MAX_LEVEL_SIZE - 100) * 0;
   434→    while (!dimension->isValidSpawn(xSpawn, zSpawn)) {
   435→        xSpawn += random.nextInt(32) - random.nextInt(32);
   436→        zSpawn += random.nextInt(32) - random.nextInt(32);
   437→
   438→                if (xSpawn < 4) xSpawn += 32;
   439→                if (xSpawn >= LEVEL_WIDTH-4) xSpawn -= 32;
   440→                if (zSpawn < 4) zSpawn += 32;
   441→                if (zSpawn >= LEVEL_DEPTH-4) zSpawn -= 32;
   442→    }
   443→    levelData.setSpawn(xSpawn, ySpawn, zSpawn);
   444→    isFindingSpawn = false;
   445→}

   447→/*public*/
   448→void Level::validateSpawn() {
   449→    if (levelData.getYSpawn() <= 0) {
   450→        levelData.setYSpawn(64);
   451→    }
   452→    int xSpawn = levelData.getXSpawn();
   453→    int zSpawn = levelData.getZSpawn();
   454→    while (getTopTile(xSpawn, zSpawn) == 0 || getTopTile(xSpawn, zSpawn) == Tile::invisible_bedrock->id) {
   455→        xSpawn += random.nextInt(8) - random.nextInt(8);
   456→        zSpawn += random.nextInt(8) - random.nextInt(8);
   457→
   458→                if (xSpawn < 4) xSpawn += 8;
   459→                if (xSpawn >= LEVEL_WIDTH-4) xSpawn -= 8;
   460→                if (zSpawn < 4) zSpawn += 8;
   461→                if (zSpawn >= LEVEL_DEPTH-4) zSpawn -= 8;
   462→    }
   463→    levelData.setXSpawn(xSpawn);
   464→    levelData.setZSpawn(zSpawn);
   465→}

   849→HitResult Level::clip(const Vec3& A, const Vec3& b, bool liquid /*= false*/, bool solidOnly /*= false*/) {
   850→    static Stopwatch sw;
   851→    //sw.printEvery(1000, "clip");
   852→    SwStartStopper w(sw);
   853→
   854→    if (A.x != A.x || A.y != A.y || A.z != A.z) return HitResult(); //@kludge This is actually a way to do it
   855→        if (b.x != b.x || b.y != b.y || b.z != b.z) return HitResult();
   856→
   857→        Vec3 a(A);
   858→
   859→    int xTile1 = Mth::floor(b.x);
   860→    int yTile1 = Mth::floor(b.y);
   861→    int zTile1 = Mth::floor(b.z);
   862→
   863→    int xTile0 = Mth::floor(a.x);
   864→    int yTile0 = Mth::floor(a.y);
   865→    int zTile0 = Mth::floor(a.z);
   866→
   867→    int maxIterations = 200;
   868→    while (maxIterations-- >= 0) {
   869→                if (a.x != a.x || a.y != a.y || a.z != a.z)
   870→                        return HitResult();
   871→        if (xTile0 == xTile1 && yTile0 == yTile1 && zTile0 == zTile1)
   872→                        return HitResult();
   873→
   874→        float xClip = 999;
   875→        float yClip = 999;
   876→        float zClip = 999;
   877→
   878→        if (xTile1 > xTile0) xClip = xTile0 + 1.000f;
   879→        if (xTile1 < xTile0) xClip = xTile0 + 0.000f;
   880→
   881→        if (yTile1 > yTile0) yClip = yTile0 + 1.000f;
   882→        if (yTile1 < yTile0) yClip = yTile0 + 0.000f;
   883→
   884→        if (zTile1 > zTile0) zClip = zTile0 + 1.000f;
   885→        if (zTile1 < zTile0) zClip = zTile0 + 0.000f;
   886→
   887→        float xDist = 999;
   888→        float yDist = 999;
   889→        float zDist = 999;
   890→
   891→        float xd = b.x - a.x;
   892→        float yd = b.y - a.y;
   893→        float zd = b.z - a.z;
   894→
   895→        if (xClip != 999) xDist = (xClip - a.x) / xd;
   896→        if (yClip != 999) yDist = (yClip - a.y) / yd;
   897→        if (zClip != 999) zDist = (zClip - a.z) / zd;
   898→
   899→        int face = 0;
   900→        if (xDist < yDist && xDist < zDist) {
   901→            if (xTile1 > xTile0) face = 4;
   902→            else face = 5;
   903→
   904→            a.x = xClip;
   905→            a.y += yd * xDist;
   906→            a.z += zd * xDist;
   907→        } else if (yDist < zDist) {
   908→            if (yTile1 > yTile0) face = 0;
   909→            else face = 1;
   910→
   911→            a.x += xd * yDist;
   912→            a.y = yClip;
   913→            a.z += zd * yDist;
   914→        } else {
   915→            if (zTile1 > zTile0) face = 2;
   916→            else face = 3;
   917→
   918→            a.x += xd * zDist;
   919→            a.y += yd * zDist;
   920→            a.z = zClip;
   921→        }
   922→
   923→        Vec3 tPos(a.x, a.y, a.z);
   924→        xTile0 = (int) (tPos.x = (float)Mth::floor(a.x));
   925→        if (face == 5) {
   926→            xTile0--;
   927→            tPos.x++;
   928→        }
   929→        yTile0 = (int) (tPos.y = (float)Mth::floor(a.y));
   930→        if (face == 1) {
   931→            yTile0--;
   932→            tPos.y++;
   933→        }
   934→        zTile0 = (int) (tPos.z = (float)Mth::floor(a.z));
   935→        if (face == 3) {
   936→            zTile0--;
   937→            tPos.z++;
   938→        }
   939→
   940→        int t = getTile(xTile0, yTile0, zTile0);
   941→        int data = getData(xTile0, yTile0, zTile0);
   942→        Tile* tile = Tile::tiles[t];
   943→
   944→                if (solidOnly && tile != NULL && tile->getAABB(this, xTile0, yTile0, zTile0) == NULL) {
   945→                        // No collision
   946→                } else if (t > 0 && tile->mayPick(data, liquid)) {
   947→                        if(xTile0 >= 0 && zTile0 >= 0 && xTile0 < LEVEL_WIDTH && zTile0 < LEVEL_WIDTH) {
   948→                                HitResult r = tile->clip(this, xTile0, yTile0, zTile0, a, b);
   949→                                if (r.isHit()) return r;
   950→                        }
   951→        }
   952→    }
   953→    return HitResult();
   954→}
