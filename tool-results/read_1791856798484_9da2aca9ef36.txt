   398→HitResult Tile::clip(Level* level, int xt, int yt, int zt, const Vec3& A, const Vec3& B) {
   399→    updateShape(level, xt, yt, zt);
   400→
   401→        //Stopwatch sw;
   402→        //sw.start();
   403→
   404→        Vec3 sub((float)xt, (float)yt, (float)zt);
   405→        Vec3 a = A - sub;//a.add((float)-xt, (float)-yt, (float)-zt);
   406→    Vec3 b = B - sub;//b.add((float)-xt, (float)-yt, (float)-zt);
   407→
   408→    Vec3 xh0, xh1, yh0, yh1, zh0, zh1;
   409→
   410→        bool bxh0 = a.clipX(b, xx0, xh0);
   411→    bool bxh1 = a.clipX(b, xx1, xh1);
   412→
   413→    bool byh0 = a.clipY(b, yy0, yh0);
   414→    bool byh1 = a.clipY(b, yy1, yh1);
   415→
   416→    bool bzh0 = a.clipZ(b, zz0, zh0);
   417→    bool bzh1 = a.clipZ(b, zz1, zh1);
   418→
   419→    //if (!containsX(xh0)) xh0 = NULL;
   420→    if (!bxh0 || !containsX(xh0)) bxh0 = false;
   421→    if (!bxh1 || !containsX(xh1)) bxh1 = false;
   422→    if (!byh0 || !containsY(yh0)) byh0 = false;
   423→    if (!byh1 || !containsY(yh1)) byh1 = false;
   424→    if (!bzh0 || !containsZ(zh0)) bzh0 = false;
   425→    if (!bzh1 || !containsZ(zh1)) bzh1 = false;
   426→    Vec3* closest = NULL;
   427→
   428→    //if (xh0 != NULL && (closest == NULL || a.distanceToSqr(xh0) < a.distanceToSqr(closest))) closest = xh0;
   429→    if (bxh0 && (closest == NULL || a.distanceToSqr(xh0) < a.distanceToSqr(*closest))) closest = &xh0;
   430→    if (bxh1 && (closest == NULL || a.distanceToSqr(xh1) < a.distanceToSqr(*closest))) closest = &xh1;
   431→    if (byh0 && (closest == NULL || a.distanceToSqr(yh0) < a.distanceToSqr(*closest))) closest = &yh0;
   432→    if (byh1 && (closest == NULL || a.distanceToSqr(yh1) < a.distanceToSqr(*closest))) closest = &yh1;
   433→    if (bzh0 && (closest == NULL || a.distanceToSqr(zh0) < a.distanceToSqr(*closest))) closest = &zh0;
   434→    if (bzh1 && (closest == NULL || a.distanceToSqr(zh1) < a.distanceToSqr(*closest))) closest = &zh1;
   435→
   436→    if (closest == NULL)
   437→                return HitResult();
   438→
   439→    int face = -1;
   440→
   441→    if (closest == &xh0) face = 4;
   442→    if (closest == &xh1) face = 5;
   443→    if (closest == &yh0) face = 0;
   444→    if (closest == &yh1) face = 1;
   445→    if (closest == &zh0) face = 2;
   446→    if (closest == &zh1) face = 3;
   447→
   448→        //sw.stop();
   449→        //sw.printEvery(5, ">>> ");
   450→
   451→    return HitResult(xt, yt, zt, face, closest->add((float)xt, (float)yt, (float)zt));
   452→}

   454→/*virtual*/
   455→void Tile::spawnResources(Level* level, int x, int y, int z, int data, float odds) {
   456→    if (level->isClientSide) return;
   457→
   458→    int count = getResourceCount(&level->random);
   459→    for (int i = 0; i < count; i++) {
   460→        if (level->random.nextFloat() > odds) continue;
   461→        int type = getResource(data, &level->random);
   462→        if (type <= 0) continue;
   463→        const float s = 0.7f;
   464→        float xo = level->random.nextFloat() * s + (1 - s) * 0.5f;
   465→        float yo = level->random.nextFloat() * s + (1 - s) * 0.5f;
   466→        float zo = level->random.nextFloat() * s + (1 - s) * 0.5f;
   467→                ItemEntity* item = new ItemEntity(level, x + xo, y + yo, z + zo, ItemInstance(type, 1, getSpawnResourcesAuxValue(data)));
   468→        item->throwTime = 10;
   469→        level->addEntity(item);
   470→    }
   471→}

   511→bool Tile::shouldRenderFace( LevelSource* level, int x, int y, int z, int face )
   512→{
   513→        if (face == 0 && y == -1) return false;
   514→        // For fixed size worlds //@todo: external constants rather than magic numbers
   515→        if (face == 2 && z == -1)  return false;
   516→        if (face == 3 && z == 256) return false;
   517→        if (face == 4 && x == -1)  return false;
   518→        if (face == 5 && x == 256) return false;
   519→        // Common
   520→        if (face == 0 && yy0 > 0) return true;
   521→        if (face == 1 && yy1 < 1) return true;
   522→        if (face == 2 && zz0 > 0) return true;
   523→        if (face == 3 && zz1 < 1) return true;
   524→        if (face == 4 && xx0 > 0) return true;
   525→        if (face == 5 && xx1 < 1) return true;
   526→        Tile* t = Tile::tiles[level->getTile(x, y, z)];
   527→        if (!t) return true;
   528→        if (face == 1 && t->id == topSnow->id) return false;
   529→        return !t->isSolidRender();
   530→        //return (!level->isSolidRenderTile(x, y, z));
   531→}
