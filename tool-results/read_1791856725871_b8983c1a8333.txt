   123→   123→bool Level::checkAndHandleWater(const AABB& box, const Material* material, Entity* e) {
   124→   124→    int x0 = Mth::floor(box.x0);
   125→   125→    int x1 = Mth::floor(box.x1 + 1);
   126→   126→
   127→   127→    int y0 = Mth::floor(box.y0);
   128→   128→    int y1 = Mth::floor(box.y1 + 1);
   129→   129→
   130→   130→    int z0 = Mth::floor(box.z0);
   131→   131→    int z1 = Mth::floor(box.z1 + 1);
   132→   132→
   133→   133→    if (!hasChunksAt(x0, y0, z0, x1, y1, z1)) {
   134→   134→        return false;
   135→   135→    }
   136→   136→
   137→   137→    bool ok = false;
   138→   138→    Vec3 current(0,0,0);
   139→   139→    for (int x = x0; x < x1; x++)
   140→   140→        for (int y = y0; y < y1; y++)
   141→   141→            for (int z = z0; z < z1; z++) {
   142→   142→                Tile* tile = Tile::tiles[getTile(x, y, z)];
   143→   143→                if (tile != NULL && tile->material == material) {
   144→   144→                    float yt0 = y + 1 - LiquidTile::getHeight(getData(x, y, z));
   145→   145→                    if (y1 >= yt0) {
   146→   146→                        ok = true;
   147→   147→                        tile->handleEntityInside(this, x, y, z, e, current);
   148→   148→                    }
   149→   149→                }
   150→   150→            }
   151→   151→        float len = current.length();
   152→   152→    if (len > 0) {
   153→   153→        const float pow = 0.004f / len;
   154→   154→        e->xd += current.x * pow;
   155→   155→        e->yd += current.y * pow;
   156→   156→        e->zd += current.z * pow;
   157→   157→    }
   158→   158→    return ok;
   159→   159→}

   167→   167→Player* Level::getNearestPlayer(float x, float y, float z, float maxDist) {
   168→   168→    float maxDistSqr = maxDist * maxDist;
   169→   169→        float best = -1;
   170→   170→    Player* result = NULL;
   171→   171→    for (unsigned int i = 0; i < players.size(); i++) {
   172→   172→        Player* p = players[i];
   173→   173→                if (p->removed) continue;
   174→   174→                float dist = p->distanceToSqr(x, y, z);
   175→   175→        if ((maxDist < 0 || dist < maxDistSqr) && (best == -1 || dist < best)) {
   176→   176→            best = dist;
   177→   177→            result = p;
   178→   178→        }
   179→   179→    }
   180→   180→    return result;
   181→   181→}

   339→   339→bool Level::tickPendingTicks(bool force) {
   340→   340→    int count = _tickNextTickSet.size();
   341→   341→    if (count > MAX_TICK_TILES_PER_TICK) count = MAX_TICK_TILES_PER_TICK;
   342→   342→    for (int i = 0; i < count; i++) {
   343→   343→                TickDataSet::iterator td = _tickNextTickSet.begin();
   344→   344→        if (!force && td->delay > levelData.getTime()) {
   345→   345→            break;
   346→   346→        }
   347→   347→
   348→   348→                int r = 8;
   349→   349→        if (hasChunksAt(td->x - r, td->y - r, td->z - r, td->x + r, td->y + r, td->z + r)) {
   350→   350→            int id = getTile(td->x, td->y, td->z);
   351→   351→            if (id == td->tileId && id > 0) {
   352→   352→                Tile::tiles[id]->tick(this, td->x, td->y, td->z, &random);
   353→   353→            }
   354→   354→        }
   355→   355→        _tickNextTickSet.erase(td);
   356→   356→    }
   357→   357→    return _tickNextTickSet.size() != 0;
   358→   358→}
