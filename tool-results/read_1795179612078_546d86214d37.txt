    28→def _vscode_active_folder() -> Optional[Path]:
    29→    if sys.platform == "win32":
    30→        result = subprocess.run(
    31→            ["tasklist", "/FI", "IMAGENAME eq Code.exe", "/NH"],
    32→            capture_output=True,
    33→            text=True,
    34→            encoding="utf-8",
    35→            errors="replace",
    36→            timeout=5,
    37→            check=False,
    38→        )
    39→        if result.returncode != 0:
    40→            raise RuntimeError("Could not check whether VS Code is running.")
    41→        if "Code.exe" not in result.stdout:
    42→            return None
    43→
    44→    for storage_path in _vscode_storage_paths():
    45→        if not storage_path.is_file():
    46→            continue
    47→        try:
    48→            data = json.loads(storage_path.read_text(encoding="utf-8", errors="replace"))
    49→        except (OSError, json.JSONDecodeError) as error:
    50→            raise RuntimeError(f"Could not read VS Code workspace state: {error}") from error
    51→
    52→        windows_state = data.get("windowsState") or {}
    53→        uris = []
    54→        last_folder = (windows_state.get("lastActiveWindow") or {}).get("folder")
    55→        if last_folder:
    56→            uris.append(last_folder)
    57→        uris.extend(
    58→            folder
    59→            for window in (windows_state.get("openedWindows") or [])
    60→            if (folder := (window or {}).get("folder"))
    61→        )
    62→
    63→        for uri in uris:
    64→            try:
    65→                parsed = urlparse(uri)
    66→                path = unquote(parsed.path)
    67→                if sys.platform == "win32" and re.match(r"^/[A-Za-z]:/", path):
    68→                    path = path[1:]
    69→                candidate = Path(path).resolve()
    70→            except (OSError, ValueError):
    71→                continue
    72→            if candidate.is_dir():
    73→                return candidate
    74→    return None

    77→def _vscode_storage_paths() -> List[Path]:
    78→    if sys.platform == "win32":
    79→        appdata = os.environ.get("APPDATA")
    80→        if not appdata:
    81→            return []
    82→        return [Path(appdata) / "Code" / "User" / "globalStorage" / "storage.json"]
    83→
    84→    if sys.platform == "linux":
    85→        config = Path(os.environ.get("XDG_CONFIG_HOME", Path.home() / ".config"))
    86→        return [
    87→            config / product / "User" / "globalStorage" / "storage.json"
    88→            for product in ("Code", "Code - Insiders", "VSCodium")
    89→        ]
    90→
    91→    support = Path.home() / "Library" / "Application Support"
    92→    return [
    93→        support / product / "User" / "globalStorage" / "storage.json"
    94→        for product in ("Code", "Code - Insiders", "VSCodium")
    95→    ]

   105→def normalize_target(rel_or_abs: str, root: Path) -> Path:
   106→    path = Path(rel_or_abs or ".")
   107→    if not path.is_absolute():
   108→        path = root / path
   109→    target = path.resolve()
   110→    try:
   111→        target.relative_to(root)
   112→    except ValueError as error:
   113→        raise ValueError(f"Path escapes workspace root: {rel_or_abs}") from error
   114→    return target

   117→def _relative(path: Path, root: Path) -> str:
   118→    return path.relative_to(root).as_posix() or "."

   121→def _iter_files(base: Path) -> Iterable[Path]:
   122→    if base.is_file():
   123→        yield base
   124→        return
   125→    for current, dirs, names in os.walk(base):
   126→        dirs[:] = sorted(
   127→            name for name in dirs
   128→            if name not in IGNORED_DIRECTORIES
   129→        )
   130→        for name in sorted(names):
   131→            yield Path(current) / name

   134→def _run_process(command: List[str], root: Path, timeout: int = 30) -> str:
   135→    result = subprocess.run(
   136→        command,
   137→        cwd=str(root),
   138→        capture_output=True,
   139→        text=True,
   140→        encoding="utf-8",
   141→        errors="replace",
   142→        shell=False,
   143→        timeout=timeout,
   144→        check=False,
   145→    )
   146→    output = result.stdout.strip() or result.stderr.strip() or "(no output)"
   147→    if result.returncode:
   148→        raise RuntimeError(
   149→            f"Command exited with status {result.returncode}:\n{output[:MAX_FILE_CHARS]}"
   150→        )
   151→    return output[:MAX_FILE_CHARS]

   182→def list_dir(path: str = ".") -> List[Dict[str, str]]:
   183→    root = get_root()
   184→    target = normalize_target(path, root)
   185→    if not target.exists():
   186→        raise FileNotFoundError(f"Directory not found: {path}")
   187→    if not target.is_dir():
   188→        raise NotADirectoryError(f"Not a directory: {path}")
   189→    return [
   190→        {
   191→            "name": child.name,
   192→            "path": _relative(child, root),
   193→            "type": "directory" if child.is_dir() else "file",
   194→        }
   195→        for child in sorted(target.iterdir(), key=lambda item: (item.is_file(), item.name.lower()))
   196→    ]

   199→def list_tree(path: str = ".", max_depth: int = 3) -> List[Dict[str, Any]]:
   200→    root = get_root()
   201→    target = normalize_target(path, root)
   202→    if not target.is_dir():
   203→        raise NotADirectoryError(f"Not a directory: {path}")
   204→    depth_limit = max(0, min(int(max_depth), 8))
   205→    result: List[Dict[str, Any]] = []
   206→    for current, dirs, files in os.walk(target):
   207→        current_path = Path(current)
   208→        depth = len(current_path.relative_to(target).parts)
   209→        dirs[:] = sorted(
   210→            name for name in dirs
   211→            if name not in IGNORED_DIRECTORIES
   212→        )
   213→        for name in sorted(dirs) + sorted(files):
   214→            entry_path = current_path / name
   215→            result.append({
   216→                "path": _relative(entry_path, root),
   217→                "type": "directory" if entry_path.is_dir() else "file",
   218→            })
   219→            if len(result) >= MAX_RESULTS:
   220→                return result
   221→        if depth >= depth_limit:
   222→            dirs[:] = []
   223→    return result

   226→def find_files(path: str = ".", pattern: str = "*") -> List[str]:
   227→    root = get_root()
   228→    base = normalize_target(path, root)
   229→    if not base.exists():
   230→        raise FileNotFoundError(f"Path not found: {path}")
   231→    if not base.is_dir():
   232→        raise NotADirectoryError(f"Not a directory: {path}")
   233→    matches = [
   234→        _relative(file_path, root)
   235→        for file_path in _iter_files(base)
   236→        if fnmatch.fnmatch(file_path.name, pattern)
   237→        or fnmatch.fnmatch(_relative(file_path, base), pattern)
   238→    ]
   239→    return matches[:MAX_RESULTS]

   242→def search_files(path: str, query: str, file_pattern: str = "*", case_sensitive: bool = False) -> List[Dict[str, Any]]:
   243→    root = get_root()
   244→    target = normalize_target(path, root)
   245→    if not target.exists():
   246→        raise FileNotFoundError(f"Path not found: {path}")
   247→    if not query.strip():
   248→        raise ValueError("Search query cannot be empty.")
   249→    needle = query if case_sensitive else query.lower()
   250→    matches: List[Dict[str, Any]] = []
   251→    files = [target] if target.is_file() else _iter_files(target)
   252→    for file_path in files:
   253→        if not fnmatch.fnmatch(file_path.name, file_pattern):
   254→            continue
   255→        try:
   256→            text = file_path.read_text(encoding="utf-8", errors="replace")
   257→        except OSError:
   258→            continue
   259→        for line_number, line in enumerate(text.splitlines(), start=1):
   260→            haystack = line if case_sensitive else line.lower()
   261→            if needle in haystack:
   262→                matches.append({
   263→                    "path": _relative(file_path, root),
   264→                    "line": line_number,
   265→                    "text": line[:500],
   266→                })
   267→                if len(matches) >= MAX_RESULTS:
   268→                    return matches
   269→    return matches

   289→def read_file_lines(path: str, start_line: int = 1, end_line: int = 200) -> str:
   290→    if start_line < 1 or end_line < start_line:
   291→        raise ValueError("Line numbers must be positive, and end_line must be >= start_line.")
   292→    if end_line - start_line > 999:
   293→        raise ValueError("Read at most 1000 lines per request.")
   294→    lines = read_file(path).splitlines()
   295→    return "\n".join(
   296→        f"{number}: {lines[number - 1]}"
   297→        for number in range(start_line, min(end_line, len(lines)) + 1)
   298→    )

   316→def get_file_outline(path: str) -> List[Dict[str, Any]]:
   317→    content = read_file(path)
   318→    patterns = (
   319→        r"^\s*(?:async\s+)?def\s+([A-Za-z_]\w*)",
   320→        r"^\s*class\s+([A-Za-z_]\w*)",
   321→        r"^\s*(?:export\s+)?(?:async\s+)?function\s+([A-Za-z_$][\w$]*)",
   322→        r"^\s*(?:export\s+)?(?:const|let|var)\s+([A-Za-z_$][\w$]*)\s*=\s*(?:async\s*)?(?:function|\([^)]*\)\s*=>|[A-Za-z_$][\w$]*\s*=>)",
   323→    )
   324→    result = []
   325→    for line_number, line in enumerate(content.splitlines(), start=1):
   326→        for pattern in patterns:
   327→            match = re.search(pattern, line)
   328→            if match:
   329→                result.append({"name": match.group(1), "line": line_number})
   330→                break
   331→    return result[:MAX_RESULTS]

   354→def edit_file(path: str, old_text: str, new_text: str, replace_all: bool = False) -> str:
   355→    if not old_text:
   356→        raise ValueError("old_text cannot be empty.")
   357→    root = get_root()
   358→    target = normalize_target(path, root)
   359→    if not target.is_file():
   360→        raise FileNotFoundError(f"File not found: {path}")
   361→    content = target.read_text(encoding="utf-8", errors="replace")
   362→    occurrences = content.count(old_text)
   363→    if occurrences == 0:
   364→        raise ValueError("The exact old_text was not found.")
   365→    if not replace_all and occurrences != 1:
   366→        raise ValueError(f"old_text occurs {occurrences} times; provide a unique snippet or set replace_all.")
   367→    updated = content.replace(old_text, new_text) if replace_all else content.replace(old_text, new_text, 1)
   368→    target.write_text(updated, encoding="utf-8")
   369→    return f"Updated {_relative(target, root)} ({occurrences if replace_all else 1} replacement(s))"

   432→    root = get_root()
   433→    target = normalize_target(path, root)
   434→    if target == root or ".git" in target.relative_to(root).parts:
   435→        raise ValueError(f"Refusing to delete protected path: {path}")

   437→        raise IsADirectoryError(f"Not a file (directories are not deleted): {path}")

   444→def apply_patch(path: str, edits: List[Dict[str, Any]]) -> str:
   445→    if not isinstance(edits, list) or not edits:
   446→        raise ValueError("edits must be a non-empty list.")
   447→    root = get_root()
   448→    target = normalize_target(path, root)
   449→    if not target.is_file():
   450→        raise FileNotFoundError(f"File not found: {path}")
   451→    content = target.read_text(encoding="utf-8", errors="replace")
   452→    for index, edit in enumerate(edits, start=1):
   453→        if not isinstance(edit, dict):
   454→            raise ValueError(f"Edit {index} must be an object with old_text and new_text.")
   455→        old_text = edit.get("old_text")
   456→        new_text = edit.get("new_text")
   457→        if not isinstance(old_text, str) or not old_text or not isinstance(new_text, str):
   458→            raise ValueError(f"Edit {index} needs a non-empty old_text and a string new_text.")
   459→        occurrences = content.count(old_text)
   460→        if occurrences != 1:
   461→            raise ValueError(f"Edit {index}: old_text occurs {occurrences} times; it must match exactly once. No changes were written.")
   462→        content = content.replace(old_text, new_text, 1)
   463→    target.write_text(content, encoding="utf-8")
   464→    return f"Patched {_relative(target, root)} ({len(edits)} edit(s))"

   484→def run_command(command: str, args: Optional[List[str]] = None) -> str:
   485→    if not command.strip():
   486→        raise ValueError("Command cannot be empty.")
   487→    root = get_root()
   488→    extra_args = args or []
   489→    if os.name == "nt":
   490→        return _run_process(["cmd.exe", "/c", command, *extra_args], root, timeout=30)
   491→    return _run_process([command, *extra_args], root, timeout=30)

   494→def _package_script_command(script: str, root: Path) -> List[str]:
   495→    if script not in SAFE_PROJECT_SCRIPTS:
   496→        raise ValueError(f"Script must be one of: {', '.join(sorted(SAFE_PROJECT_SCRIPTS))}.")
   497→    package_path = root / "package.json"
   498→    if not package_path.is_file():
   499→        raise FileNotFoundError("No package.json found in the workspace root.")
   500→    try:
   501→        package = json.loads(package_path.read_text(encoding="utf-8"))
   502→    except (OSError, json.JSONDecodeError) as error:
   503→        raise RuntimeError(f"Could not read package.json: {error}") from error
   504→    if script not in (package.get("scripts") or {}):
   505→        raise ValueError(f"package.json does not define the '{script}' script.")
   506→    manager = "npm"
   507→    if (root / "pnpm-lock.yaml").exists():
   508→        manager = "pnpm"
   509→    elif (root / "yarn.lock").exists():
   510→        manager = "yarn"
   511→    elif (root / "bun.lock").exists() or (root / "bun.lockb").exists():
   512→        manager = "bun"
   513→    if os.name == "nt":
   514→        return ["cmd.exe", "/c", manager, "run", script]
   515→    return [manager, "run", script]

   523→def run_tests() -> str:
   524→    root = get_root()
   525→    if (root / "package.json").is_file():
   526→        return run_project_script("test")
   527→    if (root / "pyproject.toml").is_file() or (root / "pytest.ini").is_file():
   528→        return _run_process([sys.executable, "-m", "pytest"], root, timeout=300)
   529→    if any(root.glob("*.sln")) or any(root.glob("*.csproj")):
   530→        return _run_process(["dotnet", "test"], root, timeout=300)
   531→    raise FileNotFoundError("No supported test setup found (package.json, pytest, or .NET project).")

   599→        "create_directory": lambda: create_directory(args["path"]),

   602→        "git_status": lambda: git_status(),
   603→        "git_diff": lambda: git_diff(args.get("path", "")),
   604→        "git_log": lambda: git_log(args.get("limit", 10)),
   605→        "git_branches": lambda: git_branches(),

   620→def send_message(message: Dict[str, Any]) -> None:
   621→    sys.stdout.write(json.dumps(message, separators=(",", ":"), ensure_ascii=False) + "\n")
   622→    sys.stdout.flush()

   625→def handle_request(request: Dict[str, Any]) -> None:
   626→    method = request.get("method")
   627→    request_id = request.get("id")
   628→    params = request.get("params") or {}
   629→
   630→    if method in ("initialize", "ping"):
   631→        result = {
   632→            "protocolVersion": "2024-11-05",
   633→            "capabilities": {"tools": {"listChanged": False}},
   634→            "serverInfo": {"name": "vscode-mcp", "version": "2.0.0"},
   635→        } if method == "initialize" else {}
   636→        send_message({"jsonrpc": "2.0", "id": request_id, "result": result})
   637→        return
   638→
   639→    if method == "tools/list":
   640→        tools = TOOLS if _vscode_active_folder() is not None else []
   641→        send_message({"jsonrpc": "2.0", "id": request_id, "result": {"tools": tools}})
   642→        return
   643→
   644→    if method == "tools/call":
   645→        try:
   646→            output = tool_dispatch(params.get("name", ""), params.get("arguments") or {})
   647→            text = output if isinstance(output, str) else json.dumps(output, ensure_ascii=False)
   648→            send_message({
   649→                "jsonrpc": "2.0",
   650→                "id": request_id,
   651→                "result": {
   652→                    "content": [{"type": "text", "text": text}],
   653→                    "structuredContent": {"result": output},
   654→                },
   655→            })
   656→        except Exception as error:
   657→            send_message({
   658→                "jsonrpc": "2.0",
   659→                "id": request_id,
   660→                "error": {"code": -32603, "message": str(error)},
   661→            })
   662→        return
   663→
   664→    if method == "notifications/initialized":
   665→        return
   666→    send_message({
   667→        "jsonrpc": "2.0",
   668→        "id": request_id,
   669→        "error": {"code": -32601, "message": f"Method not found: {method}"},
   670→    })
   671→
   672→
   673→if __name__ == "__main__":
   674→    for line in sys.stdin:
   675→        stripped = line.strip()
   676→        if not stripped:
   677→            continue
   678→        try:
   679→            request = json.loads(stripped)
   680→        except json.JSONDecodeError:
   681→            continue
   682→        if isinstance(request, dict):
   683→            handle_request(request)
