    70→void renderCursor(float x, float y, Minecraft* minecraft) {
    71→        Tesselator& t = Tesselator::instance;
    72→
    73→        minecraft->textures->loadAndBindTexture("gui/cursor.png");
    74→        glEnable(GL_BLEND);
    75→
    76→        const float s = 32;
    77→        const float width = 16;
    78→        const float height = 16;
    79→        t.begin();
    80→        t.color(0xffffffff);
    81→        t.vertexUV(x, y + (float)height, 0, 0, 1);
    82→        t.vertexUV(x + (float)width, y + (float)height, 0, 1, 1);
    83→        t.vertexUV(x + (float)width, y, 0, 1, 0);
    84→        t.vertexUV(x, y, 0, 0, 0);
    85→        t.draw();
    86→
    87→        glDisable(GL_BLEND);

    91→void GameRenderer::setupCamera(float a, int eye) {
    92→    renderDistance = (float) (16 * 16 >> (mc->options.viewDistance));
    93→#if defined(ANDROID)
    94→    if (mc->isPowerVR() && mc->options.viewDistance <= 2)
    95→                renderDistance *= 0.8f;
    96→#endif
    97→
    98→        glMatrixMode(GL_PROJECTION);
    99→    glLoadIdentity2();
   100→
   101→    float stereoScale = 0.07f;
   102→    if (mc->options.anaglyph3d) glTranslatef2(-(eye * 2 - 1) * stereoScale, 0, 0);
   103→    if (zoom != 1) {
   104→        glTranslatef2((float) zoom_x, (float) -zoom_y, 0);
   105→                glScalef2(zoom, zoom, 1);
   106→        gluPerspective(_setupCameraFov = getFov(a, true), mc->width / (float) mc->height, 0.05f, renderDistance);
   107→    } else {
   108→        gluPerspective(_setupCameraFov = getFov(a, true), mc->width / (float) mc->height, 0.05f, renderDistance);
   109→    }
   110→
   111→    glMatrixMode(GL_MODELVIEW);
   112→    glLoadIdentity2();
   113→    if (mc->options.anaglyph3d) glTranslatef2((eye * 2 - 1) * 0.10f, 0, 0);
   114→
   115→    bobHurt(a);
   116→    if (mc->options.bobView) bobView(a);
   117→         
   118→        moveCameraToPlayer(a);
   119→}

   123→/*public*/
   124→void GameRenderer::render(float a) {
   125→        TIMER_PUSH("mouse");
   126→        if (mc->player && mc->mouseGrabbed) {
   127→        mc->mouseHandler.poll();
   128→        //printf("Controller.x,y : %f,%f\n", Controller::getX(0), Controller::getY(0));
   129→
   130→        float ss = mc->options.sensitivity * 0.6f + 0.2f;
   131→        float sens = (ss * ss * ss) * 8;
   132→        float xo = mc->mouseHandler.xd * sens * 4.f;
   133→        float yo = mc->mouseHandler.yd * sens * 4.f;
   134→
   135→                const float now = _tick + a;
   136→                float deltaT = now - _lastTickT;
   137→                if (deltaT > 3.0f) deltaT = 3.0f;
   138→                _lastTickT = now;
   139→
   140→                _rotX += xo;
   141→                _rotY += yo;
   142→
   143→        int yAxis = -1;
   144→        if (mc->options.invertYMouse) yAxis = 1;
   145→
   146→                bool screenCovering = mc->screen && !mc->screen->passEvents;
   147→                if (!screenCovering)
   148→                {
   149→                        mc->player->turn(deltaT * _rotXlast, deltaT * _rotYlast * yAxis);
   150→                }
   151→    }
   152→
   153→        int xMouse = (int)(Mouse::getX() * Gui::InvGuiScale);
   154→        int yMouse = (int)(Mouse::getY() * Gui::InvGuiScale);
   155→        if (mc->useTouchscreen()) {
   156→                const int pid = Multitouch::getFirstActivePointerIdExThisUpdate();
   157→                if (pid >= 0) {
   158→                        xMouse = (int)(Multitouch::getX(pid) * Gui::InvGuiScale);
   159→                        yMouse = (int)(Multitouch::getY(pid) * Gui::InvGuiScale);
   160→                } else {
   161→                        xMouse = -9999;
   162→                        yMouse = -9999;
   163→                }
   164→        }
   165→        TIMER_POP();
   166→
   167→        bool hasClearedColorBuffer = false;
   168→        bool hasSetupGuiScreen = false;
   169→        useScreenScissor = false;
   170→        if (mc->isLevelGenerated()) {
   171→
   172→                TIMER_PUSH("level");
   173→                if (_t_keepPic < 0) {
   174→                if (!(mc->screen && !mc->screen->renderGameBehind())) {
   175→
   176→                        if (mc->screen && mc->screen->hasClippingArea(screenScissorArea))
   177→                                useScreenScissor = true;
   178→
   179→                        renderLevel(a);
   180→                        hasClearedColorBuffer = true;
   181→
   182→                        if (!mc->options.hideGui) {
   183→                                TIMER_POP_PUSH("gui");
   184→                                setupGuiScreen(false);
   185→                                hasSetupGuiScreen = true;
   186→                                mc->gui.render(a, mc->screen != NULL, xMouse, yMouse);
   187→                        }
   188→                }}
   189→                TIMER_POP();
   190→
   191→        } else {
   192→        glViewport(0, 0, mc->width, mc->height);
   193→        glMatrixMode(GL_PROJECTION);
   194→        glLoadIdentity2();
   195→        glMatrixMode(GL_MODELVIEW);
   196→        glLoadIdentity2();
   197→        setupGuiScreen(true);
   198→                hasSetupGuiScreen = true;
   199→                hasClearedColorBuffer = true;
   200→    }
   201→        //@todo
   202→        if (!hasSetupGuiScreen)
   203→                setupGuiScreen(!hasClearedColorBuffer);
   204→
   205→        if (mc->player && mc->screen == NULL) {
   206→                if (mc->inputHolder) mc->inputHolder->render(a);
   207→                if (mc->player->input) mc->player->input->render(a);
   208→        }
   209→
   210→    if (mc->screen != NULL) {
   211→                if (useScreenScissor)
   212→                        glDisable2(GL_SCISSOR_TEST);
   213→
   214→                mc->screen->render(xMouse, yMouse, a);
   215→#ifdef RPI
   216→                renderCursor(xMouse, yMouse, mc);
   217→#endif
   218→                // Screen might have been removed, so check it again
   219→                if (mc->screen && !mc->screen->isInGameScreen())
   220→                        sleepMs(15);
   221→    }
   222→}

   389→void GameRenderer::tickFov() {
   390→        if (mc->cameraTargetPlayer != mc->player)
   391→                return;
   392→
   393→        oFov = fov;
   394→        fov += (mc->player->getFieldOfViewModifier() - fov) * 0.5f;
   395→}

   397→/*private*/
   398→float GameRenderer::getFov(float a, bool applyEffects) {
   399→    Mob* player = mc->cameraTargetPlayer;
   400→    float fov = 70;
   401→
   402→        if (applyEffects)
   403→                fov *= this->oFov + (this->fov - this->oFov) * a;
   404→
   405→    if (player->isUnderLiquid(Material::water)) fov = 60;
   406→    if (player->health <= 0) {
   407→        float duration = player->deathTime + a;
   408→
   409→        fov /= ((1 - 500 / (duration + 500)) * 2.0f + 1);
   410→    }
   411→    return fov + fovOffsetO + (fovOffset - fovOffsetO) * a;
   412→}

   414→/*private*/
   415→void GameRenderer::moveCameraToPlayer(float a) {
   416→    Entity* player = mc->cameraTargetPlayer;
   417→
   418→    float heightOffset = player->heightOffset - 1.62f;
   419→
   420→    float x = player->xo + (player->x - player->xo) * a;
   421→    float y = player->yo + (player->y - player->yo) * a - heightOffset;
   422→        //printf("camera y: %f\n", y);
   423→    float z = player->zo + (player->z - player->zo) * a;
   424→
   425→        //printf("rot: %f %f\n", cameraRollO, cameraRoll);
   426→    glRotatef2(cameraRollO + (cameraRoll - cameraRollO) * a, 0, 0, 1);
   427→
   428→        //LOGI("player. alive, removed: %d, %d\n", player->isAlive(), player->removed);
   429→        if(player->isPlayer() && ((Player*)player)->isSleeping()) {
   430→                heightOffset += 1.0;
   431→                glTranslatef(0.0f, 0.3f, 0);
   432→                if (!mc->options.fixedCamera) {
   433→                        int t = mc->level->getTile(Mth::floor(player->x), Mth::floor(player->y), Mth::floor(player->z));
   434→                        if (t == Tile::bed->id) {
   435→                                int data = mc->level->getData(Mth::floor(player->x), Mth::floor(player->y), Mth::floor(player->z));
   436→
   437→                                int direction = data & 3;
   438→                                glRotatef(float(direction * 90), 0, 1, 0);
   439→                        }
   440→                        glRotatef(player->yRotO + (player->yRot - player->yRotO) * a + 180, 0, -1, 0);
   441→                        glRotatef(player->xRotO + (player->xRot - player->xRotO) * a, -1, 0, 0);
   442→                }
   443→        } else if (mc->options.thirdPersonView/* || (player->isPlayer() && !player->isAlive())*/) {
   444→        float cameraDist = thirdDistanceO + (thirdDistance - thirdDistanceO) * a;
   445→
   446→        if (mc->options.fixedCamera) {
   447→
   448→            float rotationY = thirdRotationO + (thirdRotation - thirdRotationO) * a;
   449→            float xRot = thirdTiltO + (thirdTilt - thirdTiltO) * a;
   450→
   451→            glTranslatef2(0, 0, (float) -cameraDist);
   452→            glRotatef2(xRot, 1, 0, 0);
   453→            glRotatef2(rotationY, 0, 1, 0);
   454→        } else {
   455→            float yRot = player->yRot;
   456→            float xRot = player->xRot/* + 180.0f*/;
   457→            float xd = -Mth::sin(yRot / 180 * Mth::PI) * Mth::cos(xRot / 180 * Mth::PI) * cameraDist;
   458→            float zd = Mth::cos(yRot / 180 * Mth::PI) * Mth::cos(xRot / 180 * Mth::PI) * cameraDist;
   459→            float yd = -Mth::sin(xRot / 180 * Mth::PI) * cameraDist;
   460→
   461→            for (int i = 0; i < 8; i++) {
   462→                float xo = (float)((i & 1) * 2 - 1);
   463→                float yo = (float)(((i >> 1) & 1) * 2 - 1);
   464→                float zo = (float)(((i >> 2) & 1) * 2 - 1);
   465→
   466→                xo *= 0.1f;
   467→                yo *= 0.1f;
   468→                zo *= 0.1f;
   469→
   470→                HitResult hr = mc->level->clip(Vec3(x + xo, y + yo, z + zo), Vec3(x - xd + xo + zo, y - yd + yo, z - zd + zo)); // newTemp
   471→                                if (hr.type != NO_HIT) {
   472→                    float dist = hr.pos.distanceTo(Vec3(x, y, z)); // newTemp
   473→                    if (dist < cameraDist) cameraDist = dist;
   474→                }
   475→            }
   476→
   477→                        //glRotatef2(180, 0, 1, 0);
   478→
   479→                        glRotatef2(player->xRot - xRot, 1, 0, 0);
   480→            glRotatef2(player->yRot - yRot, 0, 1, 0);
   481→            glTranslatef2(0, 0, (float) -cameraDist);
   482→            glRotatef2(yRot - player->yRot, 0, 1, 0);
   483→            glRotatef2(xRot - player->xRot, 1, 0, 0);
   484→        }
   485→    } else {
   486→        glTranslatef2(0, 0, -0.1f);
   487→    }
   488→
   489→    if (!mc->options.fixedCamera) {
   490→        glRotatef2(player->xRotO + (player->xRot - player->xRotO) * a, 1.0f, 0.0f, 0.0f);
   491→        glRotatef2(player->yRotO + (player->yRot - player->yRotO) * a + 180, 0, 1, 0);
   492→                //if (_t_keepPic > 0)
   493→        }
   494→    glTranslatef2(0, heightOffset, 0);
   495→}

   497→/*private*/
   498→void GameRenderer::bobHurt(float a) {
   499→    Mob* player = mc->cameraTargetPlayer;
   500→
   501→    float hurt = player->hurtTime - a;
   502→
   503→    if (player->health <= 0) {
   504→        float duration = player->deathTime + a;
   505→        glRotatef2(40 - (40 * 200) / (duration + 200), 0, 0, 1);
   506→    }
   507→
   508→    if (player->hurtTime <= 0) return;
   509→
   510→        hurt /= player->hurtDuration;
   511→    hurt = (float) Mth::sin(hurt * hurt * hurt * hurt * Mth::PI);
   512→
   513→    float rr = player->hurtDir;
   514→
   515→    glRotatef2(-rr, 0, 1, 0);
   516→    glRotatef2(-hurt * 14, 0, 0, 1);
   517→    glRotatef2(+rr, 0, 1, 0);
   518→}

   520→/*private*/
   521→void GameRenderer::bobView(float a) {
   522→    //if (mc->options.thirdPersonView) return;
   523→        if (!(mc->cameraTargetPlayer->isPlayer())) {
   524→        return;
   525→    }
   526→    Player* player = (Player*) mc->cameraTargetPlayer;
   527→
   528→    float wda = player->walkDist - player->walkDistO;
   529→    float b = -(player->walkDist + wda * a);
   530→    float bob = player->oBob + (player->bob - player->oBob) * a;
   531→    float tilt = player->oTilt + (player->tilt - player->oTilt) * a;
   532→    glTranslatef2((float) Mth::sin(b * Mth::PI) * bob * 0.5f, -(float) std::abs(Mth::cos(b * Mth::PI) * bob), 0);
   533→    glRotatef2((float) Mth::sin(b * Mth::PI) * bob * 3, 0, 0, 1);
   534→    glRotatef2((float) std::abs(Mth::cos(b * Mth::PI - 0.2f) * bob) * 5, 1, 0, 0);
   535→    glRotatef2((float) tilt, 1, 0, 0);
   536→}

   601→bool GameRenderer::updateFreeformPickDirection(float a, Vec3& outDir) {
   602→
   603→    if (!mc->inputHolder->allowPicking()) {
   604→        _shTicks = 1;
   605→        return false;
   606→    }
   607→
   608→    Vec3 c = mc->cameraTargetPlayer->getPos(a);
   609→
   610→    bool firstPerson = !mc->options.thirdPersonView;
   611→    const float PickingDistance = firstPerson? 6.0f : 12.0f;
   612→
   613→    _shTicks = -1;
   614→
   615→    int vp[4] = {0, 0, mc->width, mc->height};
   616→    float pt[3];
   617→    float x = mc->inputHolder->mousex;
   618→    float y = mc->height - mc->inputHolder->mousey;
   619→
   620→    //sw.start();
   621→
   622→    if (!glhUnProjectf(x, y, 1, lastModelMatrix, lastProjMatrix, vp, pt)) {
   623→        return false;
   624→    }
   625→    Vec3 p1(pt[0] + c.x, pt[1] + c.y, pt[2] + c.z);
   626→
   627→    glhUnProjectf(x, y, 0, lastModelMatrix, lastProjMatrix, vp, pt);
   628→    Vec3 p0(pt[0] + c.x, pt[1] + c.y, pt[2] + c.z);
   629→
   630→    outDir = (p1 - p0).normalized();
   631→    p1 = p0 + outDir * PickingDistance;
   632→
   633→    //sw.stop();
   634→    //sw.printEvery(30, "unproject ");
   635→
   636→    const HitResult& hit = mc->hitResult = mc->level->clip(p0, p1, false);
   637→
   638→    // If in 3rd person view - verify that the hit target is within range
   639→    if (!firstPerson && hit.isHit()) {
   640→        const float MaxSqrDist = PickingDistance*PickingDistance;
   641→        if (mc->cameraTargetPlayer->distanceToSqr((float)hit.x, (float)hit.y, (float)hit.z) > MaxSqrDist)
   642→            mc->hitResult.type = NO_HIT;
   643→    }
   644→    return true;
   645→}

   647→/*public*/
   648→void GameRenderer::pick(float a) {
   649→        if (mc->level == NULL) return;
   650→    if (mc->cameraTargetPlayer == NULL) return;
   651→        if (!mc->cameraTargetPlayer->isAlive()) return;
   652→
   653→    float range = mc->gameMode->getPickRange();
   654→    bool isPicking = true;
   655→
   656→    bool freeform = mc->useTouchscreen()  && !mc->options.isJoyTouchArea;
   657→    if (freeform) {
   658→        isPicking = updateFreeformPickDirection(a, pickDirection);
   659→    } else {
   660→        mc->hitResult = mc->cameraTargetPlayer->pick(range, a);
   661→        pickDirection = mc->cameraTargetPlayer->getViewVector(a);
   662→    }
   663→
   664→    Vec3  from = mc->cameraTargetPlayer->getPos(a);
   665→    float dist = range;
   666→        if (mc->hitResult.isHit()) {
   667→        dist = mc->hitResult.pos.distanceTo(from);
   668→    }
   669→
   670→        if (mc->gameMode->isCreativeType()) {
   671→        /*dist =*/ range = 12;
   672→    } else {
   673→        if (dist > 3) dist = 3;
   674→        range = dist;
   675→    }
   676→
   677→    Vec3 pv = (pickDirection * range);
   678→    Vec3 to  = from + pv;
   679→    mc->cameraTargetPlayer->aimDirection = pickDirection;
   680→
   681→    Entity* hovered = NULL;
   682→    const float g = 1;
   683→    AABB aabb = mc->cameraTargetPlayer->bb.expand(pv.x, pv.y, pv.z).grow(g, g, g);
   684→        EntityList& objects = mc->level->getEntities(mc->cameraTargetPlayer, aabb);
   685→    float nearest = 0;
   686→    for (unsigned int i = 0; i < objects.size(); i++) {
   687→        Entity* e = objects[i];
   688→        if (!e->isPickable()) continue;
   689→
   690→        float rr = e->getPickRadius();
   691→        AABB bb = e->bb.grow(rr, rr, rr);
   692→        HitResult p = bb.clip(from, to);
   693→                //printf("Clip Hitresult %d (%d)\n", p.type, p.isHit());
   694→
   695→        if (bb.contains(from)) {
   696→            //@todo: hovered = e; break; ?
   697→            if (nearest >= 0) {
   698→                hovered = e;
   699→                nearest = 0;
   700→            }
   701→                } else if (p.isHit()) {
   702→            float dd = from.distanceTo(p.pos);
   703→            if (dd < nearest || nearest == 0) {
   704→                hovered = e;
   705→                nearest = dd;
   706→            }
   707→        }
   708→    }
   709→
   710→    if (hovered != NULL) {
   711→                if(nearest < dist) {
   712→                        mc->hitResult = HitResult(hovered);
   713→                }
   714→    }
   715→        else if (isPicking && !mc->hitResult.isHit()) {
   716→                // if we don't have a hit result, attempt to hit the edge of the block we are standing on
   717→                // (this is an pocket edition simplification to help building floors)
   718→        //LOGI("hovered : %d (%f)\n", mc->hitResult.type, viewVec.y);
   719→                if (pickDirection.y < -.7f) {
   720→                        // looking down by more than roughly 45 degrees, fetch a hit to the block standing on
   721→                        Vec3 to = from.add(0, -2.0f, 0);
   722→
   723→                        HitResult downHitResult = mc->level->clip(from, to);
   724→                        if (downHitResult.isHit()) {
   725→                                mc->hitResult = downHitResult;
   726→                                mc->hitResult.indirectHit = true;
   727→                                // change face (not up)
   728→                                if (std::abs(pickDirection.x) > std::abs(pickDirection.z)) {
   729→                    mc->hitResult.f = (pickDirection.x < 0)? 4 : 5;
   730→                                } else {
   731→                    mc->hitResult.f = (pickDirection.z < 0)? 2 : 3;
   732→                                }
   733→                        }
   734→                }
   735→        }
   736→}
   737→/*public*/
   738→void GameRenderer::tick(int nTick, int maxTick) {
   739→        --_t_keepPic;
   740→
   741→        if (!mc->player)
   742→        {
   743→                return;
   744→        }
   745→
   746→        if (--_shTicks == 0)
   747→                mc->hitResult.type = NO_HIT;
   748→
   749→        //_rotXlast = _rotX;
   750→        //_rotYlast = _rotY;
   751→
   752→        //LOGI("x: %f\n", _rotX);
   753→
   754→    if (nTick == maxTick) {
   755→        const float tickMult = 1.0f / (float)(1 + maxTick);
   756→        _rotXlast = 0.4f * std::pow(std::abs(_rotX), 1.2f) * tickMult;
   757→        if (_rotX < 0) _rotXlast = -_rotXlast;
   758→
   759→        _rotYlast = 0.4f * std::pow(std::abs(_rotY), 1.2f) * tickMult;
   760→        if (_rotY < 0) _rotYlast = -_rotYlast;
   761→
   762→        _rotX = 0;
   763→        _rotY = 0;
   764→    }
   765→
   766→    fogBrO = fogBr;
   767→    thirdDistanceO = thirdDistance;
   768→    thirdRotationO = thirdRotation;
   769→    thirdTiltO = thirdTilt;
   770→    fovOffsetO = fovOffset;
   771→    cameraRollO = cameraRoll;
   772→
   773→    if (mc->cameraTargetPlayer == NULL) {
   774→        mc->cameraTargetPlayer = mc->player;
   775→    }
   776→
   777→        tickFov();
   778→
   779→    float brr = mc->level->getBrightness(       Mth::floor(mc->cameraTargetPlayer->x),
   780→                                                                                        Mth::floor(mc->cameraTargetPlayer->y),
   781→                                                                                        Mth::floor(mc->cameraTargetPlayer->z));
   782→
   783→        float whiteness = (3 - mc->options.viewDistance) / 3.0f;
   784→    float fogBrT = brr * (1 - whiteness) + whiteness;
   785→    fogBr += (fogBrT - fogBr) * 0.1f;
   786→
   787→    _tick++;
   788→
   789→    itemInHandRenderer->tick();
   790→//    if (mc->isRaining) tickRain();
   791→}

   793→/*private*/
   794→void GameRenderer::setupClearColor(float a) {
   795→    Level* level = mc->level;
   796→    Mob* player = mc->cameraTargetPlayer;
   797→
   798→    float whiteness = 1.0f / (4 - mc->options.viewDistance);
   799→    whiteness = 1 - (float) pow(whiteness, 0.25f);
   800→
   801→    Vec3 skyColor = level->getSkyColor(mc->cameraTargetPlayer, a);
   802→    float sr = (float) skyColor.x;
   803→    float sg = (float) skyColor.y;
   804→    float sb = (float) skyColor.z;
   805→
   806→    Vec3 fogColor = level->getFogColor(a);
   807→    fr = (float) fogColor.x;
   808→    fg = (float) fogColor.y;
   809→    fb = (float) fogColor.z;
   810→
   811→    fr += (sr - fr) * whiteness;
   812→    fg += (sg - fg) * whiteness;
   813→    fb += (sb - fb) * whiteness;
   814→
   815→    if (player->isUnderLiquid(Material::water)) {
   816→        fr = 0.02f;
   817→        fg = 0.02f;
   818→        fb = 0.2f;
   819→    } else if (player->isUnderLiquid(Material::lava)) {
   820→        fr = 0.6f;
   821→        fg = 0.1f;
   822→        fb = 0.00f;
   823→    }
   824→
   825→    float brr = fogBrO + (fogBr - fogBrO) * a;
   826→    fr *= brr;
   827→    fg *= brr;
   828→    fb *= brr;
   829→
   830→    if (mc->options.anaglyph3d) {
   831→        float frr = (fr * 30 + fg * 59 + fb * 11) / 100;
   832→        float fgg = (fr * 30 + fg * 70) / (100);
   833→        float fbb = (fr * 30 + fb * 70) / (100);
   834→
   835→        fr = frr;
   836→        fg = fgg;
   837→        fb = fbb;
   838→    }
   839→
   840→    glClearColor(fr, fg, fb, 1.0f);
   841→}
